71     private async connect(target: DbTarget = 'plan'): Promise<mysql.Connection> {
72         const { host, user, password, database } = this.requireConnection(target)
73         let lastError: unknown
74         for (let attempt = 0; attempt < 3; attempt++) {
75             try {
76                 return await mysql.createConnection({
77                     host,
78                     user,
79                     password,
80                     database,
81                     port: 3306,
82                     connectTimeout: 30_000,
83                 })
84             } catch (error) {
85                 lastError = error
86             }
87         }
88         throw lastError
89     }

91     private requireConnection(target: DbTarget): {
92         host: string
93         user: string
94         password: string
95         database: string
96     } {
97         const { user, password, host } = this.config
98         const database =
99             target === 'email' ? this.config.emailName : this.config.planName
100         const prefix = this.config.product.toUpperCase()
101         const missing: string[] = []
102         if (!user) missing.push('DB_USER')
103         if (!password) missing.push('DB_PASSWORD')
104         if (!host) missing.push('DB_HOST')
105         if (!database) {
106             missing.push(
107                 target === 'email' ? `${prefix}_DB_EMAIL` : `${prefix}_DB_PLAN`
108             )
109         }
110         if (missing.length > 0 || !user || !password || !host || !database) {
111             throw new Error(
112                 `Database credentials are not set for ${this.config.product}. Missing: ${missing.join(', ')}.`
113             )
114         }
115 
116         return { host, user, password, database }
117     }

312     async getSubscriptionRetryState(email: string): Promise<{
313         row: Record<string, unknown> | null
314         status: string | null
315         retryAttempt: number | null
316         nextAmount: number | null
317         stripeCustomerId: string | null
318     }> {
319         const row = await this.getUserRow(email)
320         if (!row) {
321             return {
322                 row: null,
323                 status: null,
324                 retryAttempt: null,
325                 nextAmount: null,
326                 stripeCustomerId: null,
327             }
328         }
329 
330         const pick = (...keys: string[]): unknown => {
331             for (const key of keys) {
332                 if (key in row && row[key] != null && row[key] !== '') {
333                     return row[key]
334                 }
335             }
336             return null
337         }
338 
339         const statusRaw = pick(
340             'subscription_status',
341             'plan_status',
342             'status',
343             'sub_status'
344         )
345         const retryRaw = pick(
346             'retry_attempt',
347             'retry_count',
348             'dunning_attempt',
349             'payment_retry_attempt',
350             'stripe_retry_attempt'
351         )
352         const amountRaw = pick(
353             'next_renewal_amount',
354             'next_charge_amount',
355             'plan_price',
356             'renewal_price',
357             'price'
358         )
359         const stripeRaw = pick(
360             'stripe_customer_id',
361             'stripe_id',
362             'customer_id',
363             'stripe_cus_id'
364         )
365 
366         return {
367             row,
368             status: statusRaw != null ? String(statusRaw) : null,
369             retryAttempt:
370                 retryRaw != null && !Number.isNaN(Number(retryRaw))
371                     ? Number(retryRaw)
372                     : null,
373             nextAmount:
374                 amountRaw != null && !Number.isNaN(Number(amountRaw))
375                     ? Number(amountRaw)
376                     : null,
377             stripeCustomerId: stripeRaw != null ? String(stripeRaw) : null,
378         }

381     /** Orphan a QR (`user_id` null, datetime one month ago) before any scan. */
382     async orphanQrForArchivedPricing(uid: string): Promise<void> {
383         const shape = await this.resolveQrTableShape()
384         if (!shape.datetimeColumn) {
385             throw new Error(
386                 `No datetime column on ${shape.table} to backdate QR "${uid}"`
387             )
388         }
389 
390         const assignments = [
391             `\`${shape.userIdColumn}\` = NULL`,
392             `\`${shape.datetimeColumn}\` = DATE_SUB(NOW(), INTERVAL 1 MONTH)`,
393         ]
394         if (shape.lastDatetimeColumn) {
395             assignments.push(
396                 `\`${shape.lastDatetimeColumn}\` = DATE_SUB(NOW(), INTERVAL 1 MONTH)`
397             )
398         }
399 
400         const write = await this.connect()
401         try {
402             const [result] = await write.query(
403                 `UPDATE \`${shape.table}\` SET ${assignments.join(', ')} WHERE \`${shape.uidColumn}\` = ? LIMIT 1`,
404                 [uid]
405             )
406             const affected = (result as { affectedRows?: number }).affectedRows
407             if (!affected) {
408                 throw new Error(
409                     `No QR row updated for uid "${uid}" on ${shape.table}.${shape.uidColumn}`
410                 )
411             }
412         } finally {
413             await write.end()
414         }
415 
416         await this.waitUntilQrIsOrphaned(uid)
417     }

446     private async waitUntilQrIsOrphaned(uid: string): Promise<void> {
447         const deadline = Date.now() + 20_000
448         let last: QrOrphanRow | null | undefined
449         while (Date.now() < deadline) {
450             last = await this.getQrOrphanRow(uid)
451             if (
452                 last &&
453                 last.user_id === null &&
454                 isAtLeastDaysAgo(last.datetime, 20)
455             ) {
456                 return
457             }
458             await new Promise((resolve) => setTimeout(resolve, 500))
459         }
460         throw new Error(
461             `QR uid "${uid}" not orphaned after waiting for replica (user_id=${String(last?.user_id)}, datetime=${String(last?.datetime)})`
462         )
463     }

465     private async resolveQrTableShape(): Promise<QrTableShape> {
466         if (this.qrTableShape) return this.qrTableShape
467 
468         const schema = this.requireConnection('plan').database
469         const conn = await this.connect()
470         try {
471             const [rows] = await conn.query(
472                 `SELECT TABLE_NAME, COLUMN_NAME
473                  FROM INFORMATION_SCHEMA.COLUMNS
474                  WHERE TABLE_SCHEMA = ?
475                    AND COLUMN_NAME IN (${QR_COLUMN_CANDIDATES.map(() => '?').join(', ')})`,
476                 [schema, ...QR_COLUMN_CANDIDATES]
477             )
478 
479             const byTable = new Map<string, Set<string>>()
480             for (const row of rows as Array<{
481                 TABLE_NAME: string
482                 COLUMN_NAME: string
483             }>) {
484                 const columns = byTable.get(row.TABLE_NAME) ?? new Set<string>()
485                 columns.add(row.COLUMN_NAME)
486                 byTable.set(row.TABLE_NAME, columns)
487             }
488 
489             const matches: QrTableShape[] = []
490             for (const [table, columns] of byTable) {
491                 const uidColumn = findColumn(columns, UID_COLUMN_CANDIDATES)
492                 const userIdColumn = findColumn(
493                     columns,
494                     USER_ID_COLUMN_CANDIDATES
495                 )
496                 if (uidColumn && userIdColumn) {
497                     matches.push({
498                         table,
499                         uidColumn,
500                         userIdColumn,
501                         datetimeColumn: findColumn(
502                             columns,
503                             DATETIME_COLUMN_CANDIDATES
504                         ),
505                         lastDatetimeColumn: findColumn(
506                             columns,
507                             LAST_DATETIME_COLUMN_CANDIDATES
508                         ),
509                     })
510                 }
511             }
512 
513             if (matches.length === 0) {
514                 throw new Error(
515                     `No table in ${schema} has both a uid column (${UID_COLUMN_CANDIDATES.join(', ')}) and user_id`
516                 )
517             }
518 
519             const preferred =
520                 matches.find(
521                     (shape) => shape.table.toLowerCase() === 'qr_codes'
522                 ) ??
523                 matches.find(
524                     (shape) =>
525                         /qr/i.test(shape.table) && !/^view_/i.test(shape.table)
526                 ) ??
527                 matches[0]
528             this.qrTableShape = preferred
529             return preferred
530         } finally {
531             await conn.end()
532         }
533     }

536 function userField(
537     row: Record<string, unknown> | null,
538     column: string
539 ): string {
540     if (!row) return ''
541     const byLower = new Map(
542         Object.keys(row).map((key) => [key.toLowerCase(), key])
543     )
544     const actual = byLower.get(column.toLowerCase())
545     if (!actual || row[actual] == null) return ''
546     return String(row[actual]).trim()
547 }
548 
549 function findColumn(
550     columns: Set<string>,
551     candidates: string[]
552 ): string | undefined {
553     const byLower = new Map(
554         [...columns].map((name) => [name.toLowerCase(), name])
555     )
556     for (const candidate of candidates) {
557         const actual = byLower.get(candidate.toLowerCase())
558         if (actual) return actual
559     }
560     return undefined
561 }

563 function isAtLeastDaysAgo(value: Date | string | null, days: number): boolean {
564     if (value == null) return false
565     const date = value instanceof Date ? value : new Date(value)
566     if (Number.isNaN(date.getTime())) return false
567     return date.getTime() <= Date.now() - days * 24 * 60 * 60 * 1000
568 }
