Bundled source snapshot. Credential-like literals are omitted. Line numbers correspond to the audited repository file; this page is for reading and is not executable test code.

1 import { APIRequestContext, APIResponse } from '@playwright/test'
2 
3 export class BaseAPI {
4     protected request: APIRequestContext
5     /** Always ends with `/` when set, so callers can append `api/...` safely. */
6     protected baseURL: string
7 
8     constructor(request: APIRequestContext, baseURL?: string) {
9         this.request = request
10         const raw = baseURL || process.env.BASE_URL || ''
11         this.baseURL = raw ? `${raw.replace(/\/+$/, '')}/` : ''
12     }
13 
14     protected async get(
15         endpoint: string,
16         params?: Record<string, any>
17     ): Promise<APIResponse> {
18         return this.request.get(`${this.baseURL}${endpoint}`, {
19             params,
20             headers: await this.getHeaders(),
21         })
22     }
23 
24     protected async post(endpoint: string, data?: any): Promise<APIResponse> {
25         return this.request.post(`${this.baseURL}${endpoint}`, {
26             data,
27             headers: await this.getHeaders(),
28         })
29     }
30 
31     protected async postForm(
32         endpoint: string,
33         form?: Record<string, string | number | boolean>,
34         headers: Record<string, string> = {}
35     ): Promise<APIResponse> {
36         return this.request.post(`${this.baseURL}${endpoint}`, {
37             form,
38             headers: {
39                 'User-Agent': 'QR-Automation-Test',
40                 Accept: 'application/json',
41                 ...headers,
42             },
43         })
44     }
45 
46     protected async put(endpoint: string, data?: any): Promise<APIResponse> {
47         return this.request.put(`${this.baseURL}${endpoint}`, {
48             data,
49             headers: await this.getHeaders(),
50         })
51     }
52 
53     protected async patch(endpoint: string, data?: any): Promise<APIResponse> {
54         return this.request.patch(`${this.baseURL}${endpoint}`, {
55             data,
56             headers: await this.getHeaders(),
57         })
58     }
59 
60     protected async delete(endpoint: string): Promise<APIResponse> {
61         return this.request.delete(`${this.baseURL}${endpoint}`, {
62             headers: await this.getHeaders(),
63         })
64     }
65 
66     protected async getHeaders(): Promise<Record<string, string>> {
67         return {
68             'Content-Type': 'application/json',
69             Accept: 'application/json',
70             'User-Agent': 'QR-Automation-Test',
71         }
72     }
73 
74     protected async validateResponse(
75         response: APIResponse,
76         expectedStatus: number = 200
77     ): Promise<any> {
78         if (response.status() !== expectedStatus) {
79             const errorBody = await response.text()
80             throw new Error(
81                 `Expected status ${expectedStatus}, got ${response.status()}. Response: ${errorBody}`
82             )
83         }
84         return response.json()
85     }
86 }
