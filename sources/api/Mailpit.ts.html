4  * Mailpit email-testing client.

42 export class MailpitApi {
43     private contextPromise: Promise<APIRequestContext>
44 
45     constructor() {
46         const baseURL = process.env.MAILPIT_BASE_URL
47         if (!baseURL) {
48             throw new Error(
49                 'Missing required env variable: MAILPIT_BASE_URL (Mailpit HTTP API base, e.g. http://host:8025)'
50             )
51         }
52         const user = process.env.MAILPIT_USER
53         const pass = process.env.MAILPIT_PASSWORD ?? ''
54         const extraHTTPHeaders: Record<string, string> = {
55             Accept: 'application/json',
56         }
57         if (user) {
58             extraHTTPHeaders.Authorization =
59                 "<literal omitted>" + Buffer.from(`${user}:${pass}`).toString('base64')
60         }
61         this.contextPromise = request.newContext({ baseURL, extraHTTPHeaders })
62     }
63 
64     private async searchByRecipient(to: string): Promise<MailpitListItem[]> {
65         const ctx = await this.contextPromise
66         const needle = to.toLowerCase()
67         const res = await ctx.get('/api/v1/search', {
68             params: { query: `to:${to}`, limit: 50 },
69         })
70         if (res.ok()) {
71             const data = (await res.json()) as { messages?: MailpitListItem[] }
72             if (data.messages?.length) {
73                 return data.messages
74             }
75         }
76         const list = await ctx.get('/api/v1/messages', {
77             params: { limit: 50 },
78         })
79         if (!list.ok()) {
80             return []
81         }
82         const data = (await list.json()) as { messages?: MailpitListItem[] }
83         return (data.messages ?? []).filter((item) =>
84             (item.To ?? []).some(
85                 (recipient) => recipient.Address?.toLowerCase() === needle
86             )
87         )
88     }

90     private static parseStamp(raw: string): number {
91         if (!raw) {
92             return Number.NaN
93         }
94         const iso = /Z|[+-]\d{2}:?\d{2}$/.test(raw) ? raw : `${raw}Z`
95         return new Date(iso).getTime()
96     }
97 
98     private static matches(
99         item: MailpitListItem,
100         to: string,
101         afterTimestamp: number,
102         subjectIncludes?: string
103     ): boolean {
104         const addressed = (item.To ?? []).some(
105             (t) => t.Address?.toLowerCase() === to.toLowerCase()
106         )
107         if (!addressed) {
108             return false
109         }
110         const stamp = MailpitApi.parseStamp(item.Created ?? item.Date ?? '')
111         // Mailpit Created is often UTC without `Z`; local parse then fails the
112         // afterTimestamp filter. Allow 2 min skew; skip the check if unparseable.
113         if (Number.isFinite(stamp) && stamp + 120_000 < afterTimestamp) {
114             return false
115         }
116         if (!subjectIncludes) {
117             return true
118         }
119         return (item.Subject ?? '')
120             .toLowerCase()
121             .includes(subjectIncludes.toLowerCase())
122     }

124     async waitForEmail(
125         to: string,
126         afterTimestamp: number,
127         timeout = 60_000,
128         interval = 5_000,
129         subjectIncludes?: string
130     ): Promise<MailpitMessage> {
131         const subjectHint = subjectIncludes
132             ? ` with subject containing "${subjectIncludes}"`
133             : ''
134         await expect
135             .poll(
136                 async () => {
137                     const messages = await this.searchByRecipient(to)
138                     return messages.some((m) =>
139                         MailpitApi.matches(
140                             m,
141                             to,
142                             afterTimestamp,
143                             subjectIncludes
144                         )
145                     )
146                 },
147                 {
148                     timeout,
149                     intervals: [interval],
150                     message: `Expected email to ${to}${subjectHint} to be received within ${timeout}ms`,
151                 }
152             )
153             .toBeTruthy()
154 
155         const messages = await this.searchByRecipient(to)
156         const message = messages.find((m) =>
157             MailpitApi.matches(m, to, afterTimestamp, subjectIncludes)
158         )
159 
160         expect(
161             message,
162             `Email to ${to}${subjectHint} was detected by poll, but not found on final fetch`
163         ).toBeTruthy()
164 
165         const matchedTo =
166             (message!.To ?? []).find(
167                 (t) => t.Address?.toLowerCase() === to.toLowerCase()
168             )?.Address ?? to
169 
170         return {
171             id: message!.ID,
172             to_email: matchedTo,
173             subject: message!.Subject ?? '',
174             sent_at: message!.Created ?? message!.Date ?? '',
175         }
176     }

221     /**
222      * Counts messages for a recipient received after `afterTimestamp`.
223      * Optionally filters by subject substring (case-insensitive).
224      */
225     async countEmails(
226         to: string,
227         afterTimestamp: number,
228         subjectIncludes?: string
229     ): Promise<number> {
230         const messages = await this.searchByRecipient(to)
231         return messages.filter((m) =>
232             MailpitApi.matches(m, to, afterTimestamp, subjectIncludes)
233         ).length
234     }
235 
236     /**
237      * Asserts no matching email arrives within `timeout` ms.
238      * Fails immediately if a matching message appears; otherwise waits the full window.
239      * @param to - Recipient address
240      * @param afterTimestamp - Only count messages newer than this (ms epoch)
241      * @param timeout - How long to keep confirming absence (default 15s)
242      * @param subjectIncludes - Optional subject substring filter
243      */
244     async expectNoEmail(
245         to: string,
246         afterTimestamp: number,
247         timeout = 15_000,
248         subjectIncludes?: string
249     ): Promise<void> {
250         const subjectHint = subjectIncludes
251             ? ` with subject containing "${subjectIncludes}"`
252             : ''
253         const started = Date.now()
254         await expect
255             .poll(
256                 async () => {
257                     const count = await this.countEmails(
258                         to,
259                         afterTimestamp,
260                         subjectIncludes
261                     )
262                     if (count > 0) return 'arrived'
263                     return Date.now() - started >= timeout
264                         ? 'absent'
265                         : 'waiting'
266                 },
267                 {
268                     timeout: timeout + 5_000,
269                     intervals: [2_000],
270                     message: `Expected no email to ${to}${subjectHint} within ${timeout}ms`,
271                 }
272             )
273             .toBe('absent')
274     }
