62 function seedGuardReason(testInfo: TestInfo): string | null {
63     if (process.env.EVENT_VALIDATION_SEED !== 'true') {
64         return 'EVENT_VALIDATION_SEED=true is required for this persistent data-seeding suite'
65     }
66     if (!process.env.EVENT_VALIDATION_RUN_ID?.trim()) {
67         return 'EVENT_VALIDATION_RUN_ID is required for resumable persistent seeding'
68     }
69     const account = process.env.EVENT_VALIDATION_ACCOUNT?.trim() || 'local'
70     if (!['local', 'default', 'dedicated'].includes(account)) {
71         return 'EVENT_VALIDATION_ACCOUNT must be local, default, or dedicated'
72     }
73     if (account !== 'dedicated') {
74         const loginVariable =
75             account === 'default' ? 'DEFAULT_USER_LOGIN' : "<literal omitted>"
76         const passwordVariable =
77             account === 'default'
78                 ? 'DEFAULT_USER_PASSWORD'
79                 : "<literal omitted>"
80         if (
81             !process.env[loginVariable]?.trim() ||
82             !process.env[passwordVariable]?.trim()
83         ) {
84             return `${loginVariable} and ${passwordVariable} are required; disposable-user fallback is prohibited`
85         }
86     }
87 
88     const requestedFields = commaSeparatedEnv('EVENT_VALIDATION_FIELDS')
89     const unknownFields = requestedFields
90         ? [...requestedFields].filter(
91               (field) =>
92                   !EVENT_VALIDATION_FIELDS.includes(
93                       field as EventValidationField
94                   )
95           )
96         : []
97     if (unknownFields.length > 0) {
98         return `Unknown EVENT_VALIDATION_FIELDS: ${unknownFields.join(', ')}`
99     }
100 
101     const baseURL = String(testInfo.project.use.baseURL ?? '')
102     try {
103         const hostname = new URL(baseURL).hostname
104         const isPrivateTarget =
105             hostname === 'localhost' ||
106             hostname === '127.0.0.1' ||
107             /^10\./.test(hostname) ||
108             /^192\.168\./.test(hostname) ||
109             /^172\.(1[6-9]|2\d|3[01])\./.test(hostname)
110         const isApprovedDevTarget = hostname === 'oqg.devii.test-qr.com'
111         if (!isPrivateTarget && !isApprovedDevTarget) {
112             return `Persistent Event seeding requires a local/private target or oqg.devii.test-qr.com; selected host is ${hostname || '(missing)'}`
113         }
114     } catch {
115         return 'Persistent Event seeding requires a valid local/private BASE_URL'
116     }
117 
118     return null
119 }

216 async function visibleText(locator: Locator): Promise<string | null> {
217     const count = await locator.count()
218     const messages: string[] = []
219     for (let index = 0; index < count; index++) {
220         const item = locator.nth(index)
221         if (await item.isVisible().catch(() => false)) {
222             const text = (await item.textContent())?.trim()
223             if (text) messages.push(text)
224         }
225     }
226     return messages.length ? [...new Set(messages)].join(' | ') : null

606     currentRunId: string
607 ): Promise<ScenarioObservation> {

613             return (
614                 (await pageManager.myQrCodesPage

623     ) {
624         return {
625             accepted: true,
626             observedValue: scenario.input,
627             validationText: null,
628             retainedQrName: qrName,
629             folderName: scenario.input,
630             folderSource: 'dashboard_fallback',
631             dashboardUrl: page.url(),
632             dashboardUid:
633                 await pageManager.myQrCodesPage.getQrCardIdByName(qrName),
634             resumedExistingArtifact: true,
635             unsafeExecution: false,
636             note: 'Existing folder and QR reused for this run ID',
637         }
638     }
639

641     const step2FolderPresent =
642         (await pageManager.qrCreationPage.locator.qrFolderButton.count()) > 0 &&
643         (await pageManager.qrCreationPage.locator.qrFolderButton
644             .isVisible()
645             .catch(() => false))
646 
647     if (!step2FolderPresent) {
648         const dashboardResult = exactFolderExists
649             ? {
650                   accepted: true,
651                   observedValue: scenario.input,
652                   savedFolderName: scenario.input,
653                   observedMaxLength: null,

674                 unsafeExecution: dashboardResult.unsafeExecution,

701             folderName: savedFolderName,
702             ...published,
703             folderSource: 'dashboard_fallback',
704             resumedExistingArtifact: exactFolderExists,
705             unsafeExecution: false,
706             note: 'Folder validated through dashboard; Event Step 2 folder UI absent',
707         }
708     }
709 
710     await pageManager.step2Page.fillEventRequiredFields(BASELINE_TITLE)
711     await pageManager.step2Page.setEventQrName(qrName)
712     let assignedFolderName = scenario.input
713     if (exactFolderExists) {
714         await pageManager.qrCreationPage.selectExistingFolder(scenario.input)
715     } else {
716         await pageManager.qrCreationPage.openFolderAccordion()
717         await pageManager.qrCreationPage.locator.noQrFolderButton.click()
718         await pageManager.qrCreationPage.locator.createNewFolderButton.click()
719         const input = pageManager.qrCreationPage.locator.folderNameInput
720         const dialogs: string[] = []
721         const dismissDialog = (dialog: Dialog): void => {
722             dialogs.push(dialog.message())
723             void dialog.dismiss()
724         }
725         page.on('dialog', dismissDialog)
726         await input.fill(scenario.input)
727         const observedValue = await input.inputValue()
728         const maxLengthAttribute = await input.getAttribute('maxlength')
729         const createFolderButton =
730             pageManager.qrCreationPage.locator.createFolderButton
731         if (!(await createFolderButton.isEnabled())) {
732             page.off('dialog', dismissDialog)
733             const validationText = await visibleText(
734                 fieldValidationLocator(page, pageManager, 'folder')
735             )
736             await page.keyboard.press('Escape')
737             return {
738                 accepted: false,
739                 observedValue,
740                 observedMaxLength:
741                     maxLengthAttribute === null
742                         ? null
743                         : Number(maxLengthAttribute),
744                 validationText,
745                 retainedQrName: null,
746                 folderName: null,
747                 folderSource: 'event_step2',
748                 dashboardUrl: null,
749                 dashboardUid: null,
750                 resumedExistingArtifact: false,
751                 unsafeExecution: dialogs.length > 0,
752                 note: 'Folder create remained disabled after input',
753             }
754         }
755         await createFolderButton.click()
756         const accepted = await input
757             .waitFor({ state: 'hidden', timeout: 10_000 })
758             .then(() => true)
759             .catch(() => false)
760         page.off('dialog', dismissDialog)
761         if (!accepted || dialogs.length > 0) {
762             return {
763                 accepted: false,
764                 observedValue,
765                 observedMaxLength:
766                     maxLengthAttribute === null
767                         ? null
768                         : Number(maxLengthAttribute),
769                 validationText:
770                     dialogs.length > 0
771                         ? `Dialog executed: ${dialogs.map(displaySafe).join(' | ')}`
772                         : await visibleText(
773                               fieldValidationLocator(
774                                   page,
775                                   pageManager,
776                                   'folder'
777                               )
778                           ),
779                 retainedQrName: null,
780                 folderName: null,
781                 folderSource: 'event_step2',
782                 dashboardUrl: null,
783                 dashboardUid: null,
784                 resumedExistingArtifact: false,
785                 unsafeExecution: dialogs.length > 0,
786                 note:
787                     dialogs.length > 0
788                         ? 'Publication blocked because folder input executed a dialog'
789                         : null,
790             }
791         }
792         const selectedFolderName = (
793             (await pageManager.qrCreationPage.locator.noQrFolderButton.textContent()) ??
794             observedValue
795         ).trim()
796         assignedFolderName = selectedFolderName
797     }
798 
799     await pageManager.qrCreationPage.clickStep2NextButton()
800     await page.waitForURL(/[?&]step=3\b/, { timeout: 60_000 })
801     const published = await publishFromStep3(
802         page,
803         pageManager,
804         qrName,
805         assignedFolderName,
806         true
807     )
808     return {
809         accepted: true,
810         observedValue: assignedFolderName,
811         validationText: null,
812         retainedQrName: qrName,
813         folderName: assignedFolderName,
814         ...published,
815         resumedExistingArtifact: exactFolderExists,
816         unsafeExecution: false,
817         note: null,
818     }
819 }

821 function classifyOutcome(
822     scenario: EventValidationScenario,
823     observation: ScenarioObservation
824 ): EventValidationOutcome {
825     if (observation.unsafeExecution) return 'unsafe_execution'
826     if (observation.accepted !== scenario.expectedPass) {
827         return 'expectation_mismatch'
828     }
829     if (!observation.accepted) return 'rejected'
830     if (
831         observation.observedValue !== null &&
832         observation.observedValue !== scenario.input
833     ) {
834         return 'sanitized'
835     }
836     return 'accepted'
837 }

839 async function runField(
840     page: Page,
841     pageManager: PageManager,
842     testInfo: TestInfo,
843     field: EventValidationField
844 ): Promise<void> {
845     const currentRunId = runId()
846     const manifest = new EventValidationManifest(currentRunId, field)
847     await manifest.load()
848 
849     try {
850         await loginSeedOwner(page, pageManager)
851         if (
852             !(await pageManager.step1Page.locator.eventQrType
853                 .isVisible()
854                 .catch(() => false))
855         ) {
856             await openCreateWizardLoggedIn(page, pageManager)
857         }
858         await expect(pageManager.step1Page.locator.eventQrType).toBeVisible()
859 
860         const scenarios = requestedScenarios(field)
861         expect(
862             scenarios.length,
863             `No ${field} scenarios matched EVENT_VALIDATION_CASE_IDS`
864         ).toBeGreaterThan(0)
865         for (const scenario of scenarios) {
866             let observation: ScenarioObservation
867             try {
868                 observation =
869                     field === 'folder'
870                         ? await executeFolderScenario(
871                               page,
872                               pageManager,
873                               scenario,
874                               currentRunId
875                           )
876                         : await executeNonFolderScenario(
877                               page,
878                               pageManager,
879                               scenario,
880                               currentRunId
881                           )
882             } catch (error) {
883                 observation = {
884                     accepted: false,
885                     observedValue: null,
886                     validationText: null,
887                     retainedQrName: null,
888                     folderName: null,
889                     folderSource: null,
890                     dashboardUrl: page.url(),
891                     dashboardUid: null,
892                     resumedExistingArtifact: false,
893                     unsafeExecution: false,
894                     note: `Operational error: ${
895                         error instanceof Error
896                             ? displaySafe(error.message)
897                             : 'unknown error'
898                     }`,
899                 }
900             }
901 
902             const outcome = observation.note?.startsWith('Operational error:')
903                 ? 'operational_error'
904                 : classifyOutcome(scenario, observation)
905             await manifest.add({
906                 sourceTestId: scenario.sourceTestId,
907                 targetField: field,
908                 source: scenario.source,
909                 sourceFieldId: scenario.sourceFieldId,
910                 rawInput: scenario.input,
911                 displaySafeInput: displaySafe(scenario.input),
912                 expected: scenario.expected,
913                 expectedError: scenario.expectedError,
914                 observedValue: observation.observedValue,
915                 observedMaxLength: observation.observedMaxLength ?? null,
916                 actualValidationText: observation.validationText,
917                 outcome,
918                 retainedQrName: observation.retainedQrName,
919                 folderName: observation.folderName,
920                 folderSource: observation.folderSource,
921                 dashboardUrl: observation.dashboardUrl,
922                 dashboardUid: observation.dashboardUid,
923                 resumedExistingArtifact: observation.resumedExistingArtifact,
924                 note: observation.note,
925             })
926 
927             expect
928                 .soft(
929                     observation.accepted,
930                     `${field}/${scenario.sourceTestId}: expected ${scenario.expected}; actual validation: ${observation.validationText ?? '(none)'}`
931                 )
932                 .toBe(scenario.expectedPass)
933             expect
934                 .soft(
935                     outcome,
936                     `${field}/${scenario.sourceTestId} encountered an operational error`
937                 )
938                 .not.toBe('operational_error')
939             expect
940                 .soft(
941                     outcome,
942                     `${field}/${scenario.sourceTestId} executed an unsafe dialog`
943                 )
944                 .not.toBe('unsafe_execution')
945         }
946     } finally {
947         await manifest.attach(testInfo)
948         console.info(
949             `Event validation ${field} summary: ${JSON.stringify(manifest.summary())}`
950         )
951     }
952 }
