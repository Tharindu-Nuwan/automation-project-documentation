41 export function displaySafe(value: string): string {
42     return Array.from(value)
43         .map((character) => {
44             const code = character.charCodeAt(0)
45             if (!((code >= 0 && code <= 31) || (code >= 127 && code <= 159))) {
46                 return character
47             }
48             const named: Record<number, string> = {
49                 9: '\\t',
50                 10: '\\n',
51                 13: '\\r',
52             }
53             return named[code] ?? `\\u${code.toString(16).padStart(4, '0')}`
54         })
55         .join('')
56         .replace(/\u202e/g, '\\u202e')
57         .replace(/\u200b/g, '\\u200b')
58         .replace(/\u200d/g, '\\u200d')
59 }

61 function csvCell(value: unknown): string {
62     const text =
63         value === null || value === undefined
64             ? ''
65             : typeof value === 'boolean'
66               ? String(value)
67               : String(value)
68     return `"${text.replace(/"/g, '""')}"`
69 }

80         if (!/^[A-Za-z0-9._-]+$/.test(runId)) {

91             '.auth',

97     async add(record: EventValidationManifestRecord): Promise<void> {
98         const existingIndex = this.records.findIndex(
99             (existing) =>
100                 existing.sourceTestId === record.sourceTestId &&
101                 existing.source === record.source
102         )
103         if (existingIndex >= 0) {
104             this.records[existingIndex] = record
105         } else {
106             this.records.push(record)
107         }
108         await this.write()
109     }
110 
111     async load(): Promise<void> {
112         for (const existingPath of [this.checkpointJsonPath, this.jsonPath]) {
113             try {
114                 const existing = JSON.parse(
115                     await readFile(existingPath, 'utf8')
116                 ) as {
117                     records?: EventValidationManifestRecord[]
118                 }
119                 this.records.splice(
120                     0,
121                     this.records.length,
122                     ...(existing.records ?? [])
123                 )
124                 return
125             } catch {
126                 // Try the next resume source.
127             }
128         }
129     }
130 
131     async attach(testInfo: TestInfo): Promise<void> {
132         await this.write()
133         await testInfo.attach(`event-validation-${this.field}.json`, {
134             path: this.jsonPath,
135             contentType: 'application/json',
136         })
137         await testInfo.attach(`event-validation-${this.field}.csv`, {
138             path: this.csvPath,
139             contentType: 'text/csv',
140         })
141     }

143     summary(): Record<EventValidationOutcome, number> {
144         const counts: Record<EventValidationOutcome, number> = {
145             accepted: 0,
146             rejected: 0,
147             sanitized: 0,
148             expectation_mismatch: 0,
149             unsafe_execution: 0,
150             operational_error: 0,
151         }
152         for (const record of this.records) {
153             counts[record.outcome]++
154         }
155         return counts

174     private async write(): Promise<void> {
175         await Promise.all([
176             mkdir(this.outputDirectory, { recursive: true }),
177             mkdir(this.checkpointDirectory, { recursive: true }),
178         ])
179         const json = `${JSON.stringify(
180             {
181                 runId: this.runId,
182                 field: this.field,
183                 generatedAt: new Date().toISOString(),
184                 records: this.records,
185             },
186             null,
187             2
188         )}\n`
189         const csv = this.toCsv()
190         await Promise.all([
191             writeFile(this.jsonPath, json, 'utf8'),
192             writeFile(this.csvPath, csv, 'utf8'),
193             writeFile(this.checkpointJsonPath, json, 'utf8'),
194             writeFile(this.checkpointCsvPath, csv, 'utf8'),
195         ])
196     }

198     private toCsv(): string {
199         const headers: (keyof EventValidationManifestRecord)[] = [
200             'sourceTestId',
201             'targetField',
202             'source',
203             'sourceFieldId',
204             'rawInput',
205             'displaySafeInput',
206             'expected',
207             'expectedError',
208             'observedValue',
209             'observedMaxLength',
210             'actualValidationText',
211             'outcome',
212             'retainedQrName',
213             'folderName',
214             'folderSource',
215             'dashboardUrl',
216             'dashboardUid',
217             'resumedExistingArtifact',
218             'note',
219         ]
220         return [
221             headers.map(csvCell).join(','),
222             ...this.records.map((record) =>
223                 headers.map((header) => csvCell(record[header])).join(',')
