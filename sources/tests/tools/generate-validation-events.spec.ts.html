tests/tools/generate-validation-events.spec.ts

Bundled source snapshot. Credential-like literals are omitted. Line numbers correspond to the audited repository file; this page is for reading and is not executable test code.

1 /**
2  * GENERATOR — this is not a test suite. It creates data for manual review.
3  *
4  * For every text type in `validationPool.json` (52 distinct values drawn from
5  * qr_field_validation_master_dataset.json) it creates one Event QR per target
6  * field, filling only the mandatory fields plus the field under test.
7  *
8  * Every QR is named `VAL-<FIELD>-<Vnn>` so it can be found in the dashboard and
9  * traced back to its input via `validationPool.md`.
10  *
11  * Nothing is asserted about validation behaviour — each attempt is simply
12  * recorded as CREATED or BLOCKED (with whatever message the UI showed) into
13  * `validation_results_<FIELD>.json`.
14  *
15  * Reuses `.auth/validation_account.json` (legacy: repo-root
16  * `validation_account.json`); the first run registers one if neither exists.
17  *
18  * Run one field at a time, or all with --workers=3:
19  *   npx playwright test tests/tools/generate-validation-events.spec.ts \
20  *     --project=oqg-chromium --workers=3 --retries=0
21  */
22

76 /** Opens Step 2 of a fresh Event QR. Retries the card click once. */
77 async function openFreshEventStep2(
78     page: Page,
79     pageManager: PageManager
80 ): Promise<void> {
81     await page.goto('/qr-codes')
82     await page.getByTestId('sidebar-create-qr-link').click()
83     await pageManager.step1Page.locator.eventQrType.waitFor({
84         state: 'visible',
85         timeout: 60_000,
86     })
87     for (let attempt = 0; attempt < 2; attempt++) {
88         await pageManager.step1Page.selectEventQrType()
89         const onStep2 = await page
90             .waitForURL(/step=2/, { timeout: 15_000 })
91             .then(() => true)
92             .catch(() => false)
93         if (onStep2) break
94     }
95     // The help popup only shows on this account's first visit to Step 2.
96     await pageManager.step2Page.ensureHelpClosedBeforeUpload().catch(() => {})
97     await pageManager.step2Page.locator.eventTitleInput.waitFor({
98         state: 'visible',
99         timeout: 30_000,
100     })
101 }

103 /** Collects any validation text currently on screen. */
104 async function collectMessages(page: Page): Promise<string[]> {
105     return page.evaluate(() => {
106         const out = new Set<string>()
107         document.querySelectorAll('p,span,div,label,small').forEach((el) => {
108             // Leaf nodes only: a section wrapper concatenates every label it
109             // contains, which drowns the actual validation message.
110             if (el.querySelector('*')) return
111             const t = (el.textContent || '').trim()
112             if (
113                 t &&
114                 t.length < 140 &&
115                 /required|invalid|must |cannot|not allowed|maximum|minimum|exceed|valid |too long|characters/i.test(
116                     t
117                 )
118             ) {
119                 out.add(t)
120             }
121         })
122         return [...out].slice(0, 10)
123     })
124 }

126 /** Fills the field under test. Never asserts — bad input must be allowed in. */
127 async function fillTargetField(
128     pageManager: PageManager,
129     field: Field,
130     value: string
131 ): Promise<void> {
132     const step2 = pageManager.step2Page
133     const loc = step2.locator
134     switch (field) {
135         case 'TITLE':
136             await loc.eventTitleInput.fill(value)
137             break
138         case 'SUMMARY':
139             await loc.eventSummaryTextarea.fill(value)
140             break
141         case 'CONTACT':
142             await step2.expandEventAccordion(loc.eventContactAccordionTrigger)
143             await loc.eventContactNameInput.fill(value)
144             break
145         case 'PHONE':
146             await step2.expandEventAccordion(loc.eventContactAccordionTrigger)
147             await loc.eventPhoneAddButton.click()
148             await loc.eventPhone0Input.fill(value)
149             break
150         case 'EMAIL':
151             await step2.expandEventAccordion(loc.eventContactAccordionTrigger)
152             await loc.eventEmailAddButton.click()
153             await loc.eventEmail0Input.fill(value)
154             break
155         case 'WEBSITE':
156             await step2.expandEventAccordion(loc.eventContactAccordionTrigger)
157             await loc.eventWebsiteAddButton.click()
158             await loc.eventWebsite0UrlInput.fill(value)
159             break
160     }
161 }

163 /** Reads back what the field actually kept (truncation / sanitisation shows here). */
164 async function readStoredValue(
165     pageManager: PageManager,
166     field: Field
167 ): Promise<string> {
168     const loc = pageManager.step2Page.locator
169     const map: Record<Field, () => Promise<string>> = {
170         TITLE: () => loc.eventTitleInput.inputValue(),
171         SUMMARY: () => loc.eventSummaryTextarea.inputValue(),
172         CONTACT: () => loc.eventContactNameInput.inputValue(),
173         PHONE: () => loc.eventPhone0Input.inputValue(),
174         EMAIL: () => loc.eventEmail0Input.inputValue(),
175         WEBSITE: () => loc.eventWebsite0UrlInput.inputValue(),
176     }
177     return map[field]().catch(() => '')
178 }

180 async function createOne(
181     page: Page,
182     pageManager: PageManager,
183     field: Field,
184     entry: PoolEntry
185 ): Promise<Outcome> {
186     const qrName = `VAL-${field}-${entry.vid}`
187     const base: Outcome = {
188         vid: entry.vid,
189         field,
190         qrName,
191         sourceIds: entry.ids,
192         datasetExpectation: entry.expected,
193         valuePreview: preview(entry.value),
194         result: 'ERROR',
195         messages: [],
196     }
197 
198     await openFreshEventStep2(page, pageManager)
199     const step2 = pageManager.step2Page
200 
201     // Mandatory fields only: title + both dates (All day on). The date setters
202     // default to tomorrow, which is all this needs.
203     if (field !== 'TITLE') {
204         await step2.locator.eventTitleInput.fill(qrName)
205     }
206     await step2.setEventStartDate()
207     await step2.setEventEndDate()
208     await step2.toggleEventAllDay()
209 
210     await fillTargetField(pageManager, field, entry.value)
211     base.storedValue = preview(await readStoredValue(pageManager, field))
212 
213     // Name the QR so it is findable in the dashboard.
214     await step2.locator.qrCodeNameTab.click()
215     await step2.locator.qrCodeNameInput.fill(qrName)
216 
217     await pageManager.qrCreationPage.clickStep2NextButton({
218         waitForEnabled: false,
219     })
220     const reachedStep3 = await page
221         .waitForURL(/step=3/, { timeout: 25_000 })
222         .then(() => true)
223         .catch(() => false)
224 
225     if (!reachedStep3) {
226         base.result = 'BLOCKED'
227         base.messages = await collectMessages(page)
228         return base
229     }
230 
231     await pageManager.qrCreationPage.clickStep3CreateButton()
232     await pageManager.qrCreationPage.closeDownloadModalIfPresent()
233     base.result = 'CREATED'
234     return base
235 }

237 /** VAL_FIELDS=TITLE,EMAIL restricts the run to those fields. */
238 const SELECTED = (process.env.VAL_FIELDS || '')
239     .split(',')
240     .map((f) => f.trim().toUpperCase())
241     .filter(Boolean)
242 const ACTIVE_FIELDS = SELECTED.length
243     ? FIELDS.filter((f) => SELECTED.includes(f))
244     : FIELDS
245 
246 for (const field of ACTIVE_FIELDS) {
247     test(`generate Event QRs for ${field}`, async ({ page, pageManager }) => {
248         test.setTimeout(0) // 52 create-and-publish cycles
249 
250         await loginValidationAccount(page, pageManager)
251         const results: Outcome[] = []
252         const outFile = `validation_results_${field}.json`
253 
254         for (const entry of POOL) {
255             try {
256                 const outcome = await createOne(page, pageManager, field, entry)
257                 results.push(outcome)
258                 console.log(
259                     `[${field}] ${entry.vid} -> ${outcome.result}` +
260                         (outcome.messages.length
261                             ? ` :: ${outcome.messages[0]}`
262                             : '')
263                 )
264             } catch (error) {
265                 results.push({
266                     vid: entry.vid,
267                     field,
268                     qrName: `VAL-${field}-${entry.vid}`,
269                     sourceIds: entry.ids,
270                     datasetExpectation: entry.expected,
271                     valuePreview: preview(entry.value),
272                     result: 'ERROR',
273                     messages: [String(error).slice(0, 200)],
274                 })
275                 console.log(
276                     `[${field}] ${entry.vid} -> ERROR :: ${String(error).slice(0, 120)}`
277                 )
278             }
279             // Write after every item so a crash never loses the whole run.
280             writeFileSync(outFile, JSON.stringify(results, null, 1))
281         }
282 
283         const created = results.filter((r) => r.result === 'CREATED').length
284         console.log(
285             `[${field}] DONE created=${created} blocked=${
286                 results.filter((r) => r.result === 'BLOCKED').length
287             } errored=${results.filter((r) => r.result === 'ERROR').length}`
288         )
289         expect(results).toHaveLength(POOL.length)
290     })
291 }
