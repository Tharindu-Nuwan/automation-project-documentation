1 import {
2     closeSync,
3     existsSync,
4     mkdirSync,
5     openSync,
6     readFileSync,
7     unlinkSync,
8     writeFileSync,
9 } from 'node:fs'
10 import path from 'node:path'
11 
12 import type { Page } from '@playwright/test'
13 
14 import type { PageManager } from '../../base/PageManager'
15 
16 export type ValidationAccount = { email: string; password: string }
17 
18 const AUTH_DIR = path.resolve('.auth')
19 const ACCOUNT_PATH = path.join(AUTH_DIR, 'validation_account.json')
20 const LEGACY_ACCOUNT_PATH = path.resolve('validation_account.json')
21 const LOCK_PATH = path.join(AUTH_DIR, 'validation_account.lock')
22 
23 function parseAccount(filePath: string): ValidationAccount | null {
24     if (!existsSync(filePath)) return null
25     try {
26         const parsed = JSON.parse(
27             readFileSync(filePath, 'utf-8')
28         ) as Partial<ValidationAccount>
29         if (parsed.email && parsed.password) {
30             return { email: parsed.email, password: parsed.password }
31         }
32     } catch {
33         // Missing, empty, or mid-write.
34     }
35     return null
36 }
37 
38 function readStoredAccount(): ValidationAccount | null {
39     return parseAccount(ACCOUNT_PATH) ?? parseAccount(LEGACY_ACCOUNT_PATH)
40 }
41 
42 function persistAccount(account: ValidationAccount): void {
43     mkdirSync(AUTH_DIR, { recursive: true })
44     writeFileSync(
45         ACCOUNT_PATH,
46         `${JSON.stringify(account, null, 2)}\n`,
47         'utf-8'
48     )
49 }
50 
51 async function delay(ms: number): Promise<void> {
52     await new Promise((resolve) => setTimeout(resolve, ms))
53 }
54 
55 async function waitForStoredAccount(
56     timeoutMs = 120_000
57 ): Promise<ValidationAccount | null> {
58     const deadline = Date.now() + timeoutMs
59     while (Date.now() < deadline) {
60         const account = readStoredAccount()
61         if (account) return account
62         await delay(200)
63     }
64     return null
65 }
66 
67 async function loginWith(
68     page: Page,
69     pageManager: PageManager,
70     account: ValidationAccount
71 ): Promise<void> {
72     await page.goto('/login')
73     await pageManager.loginPage.logIn(account.email, account.password)
74     await page.waitForURL(/qr-codes/, { timeout: 60_000 })
75 }

77 /**
78  * Logs into the shared validation-generator account.
79  *
80  * Reuses `.auth/validation_account.json` (or a legacy repo-root
81  * `validation_account.json`) when present; otherwise registers a user and
82  * persists it so later workers and runs share the same dashboard.
83  */
84 export async function loginValidationAccount(
85     page: Page,
86     pageManager: PageManager
87 ): Promise<ValidationAccount> {
88     const existing = readStoredAccount()
89     if (existing) {
90         await loginWith(page, pageManager, existing)
91         return existing
92     }
93 
94     mkdirSync(AUTH_DIR, { recursive: true })
95     let lockFd: number | undefined
96     try {
97         lockFd = openSync(LOCK_PATH, 'wx')
98     } catch (error) {
99         if ((error as NodeJS.ErrnoException).code !== 'EEXIST') throw error
100         const account = await waitForStoredAccount()
101         if (!account) {
102             throw new Error(
103                 'Timed out waiting for .auth/validation_account.json to be created'
104             )
105         }
106         await loginWith(page, pageManager, account)
107         return account
108     }
109 
110     try {
111         const raced = readStoredAccount()
112         if (raced) {
113             await loginWith(page, pageManager, raced)
114             return raced
115         }
116         const account = await pageManager.registerPage.authorizeUser()
117         persistAccount(account)
118         return account
119     } finally {
120         if (lockFd !== undefined) closeSync(lockFd)
121         try {
122             unlinkSync(LOCK_PATH)
123         } catch {
124             // Another worker may have already removed it.
125         }
126     }
127 }
