77   test('Capture QR list across languages and screen sizes', async ({ pageManager, page }) => {
78     const email    = process.env.TEST_ACCOUNT_EMAIL!
79     const password = process.env.TEST_ACCOUNT_PASSWORD!
80 
81     // 1. Log in and wait until redirect to /qr-codes completes
82     await page.goto('/login')
83     await pageManager.loginPage.logIn(email, password)
84     await page.waitForURL('**/qr-codes**', { timeout: 30_000 })
85     await page.waitForLoadState('domcontentloaded')
86     await waitForQrListReady(page, pageManager)
87 
88     // 2. Go to account page to start language loop
89     await page.setViewportSize({ width: 1920, height: 1080 })
90     await page.goto('/account')
91     await page.waitForLoadState('domcontentloaded')
92     await waitForAccountLanguageReady(page, pageManager)
93 
94     for (const lang of LANGUAGES_TO_RUN) {
95 
96       // 3. Switch language and save
97       await pageManager.accountPage.locator.languageSelect.selectOption({ value: lang.code })
98       await pageManager.accountPage.locator.languageUpdateButton.click()
99       await page.waitForLoadState('domcontentloaded')
100       await waitForAccountLanguageReady(page, pageManager)
101 
102       // 4. Go to QR codes list
103       await page.goto('/qr-codes')
104       await page.waitForLoadState('domcontentloaded')
105       await waitForQrListReady(page, pageManager)
106 
107       // 5. Set quantity to 100
108       await pageManager.myQrCodesPage.locator.quantityOfQrCodesLocator.click()
109       await pageManager.myQrCodesPage.locator.perPageOption100.click()
110       await page.waitForLoadState('domcontentloaded')
111       await waitForQrListReady(page, pageManager)
112 
113       // 6. Take Percy screenshot at each viewport
114       for (const vp of VIEWPORTS) {

120       }
121 
122       // 7. Reset to desktop and go back to account for next language
123       await page.setViewportSize({ width: 1920, height: 1080 })
124       await waitForVisualReady(page)
