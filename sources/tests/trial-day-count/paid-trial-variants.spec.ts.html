160     test(
161         "OQG3-2142-TC28: '14 days remaining' trial text is localised after URL language switch",
162         { tag: ['@Regression', '@Medium'] },
163         async ({ page, pageManager }) => {
164             qase.id(0)
165             qase.title(
166                 "OQG3-2142-TC28: '14 days remaining' trial text is localised after URL language switch"
167             )
168             qase.suite(SUITE)
169 
170             const daysRemaining14 = formatDaysRemaining(
171                 LOCALIZED_TRIAL.daysRemainingTemplate,
172                 14
173             )
174             // date_diff %d on a fresh 14-day trial often renders as 13 full days.
175             const daysRemaining13 = formatDaysRemaining(
176                 LOCALIZED_TRIAL.daysRemainingTemplate,
177                 13
178             )
179 
180             await test.step('TC28: start DPF under /es URL language and pay', async () => {
181                 await completeDpfPaidFullAccess(
182                     page,
183                     pageManager,
184                     userData.email,
185                     {
186                         entryPath: dpfCreateEntryForLocale(
187                             LOCALIZED_TRIAL.locale
188                         ),
189                     }
190                 )
191             })
192 
193             await test.step('TC28: dashboard keeps the localised language after pay', async () => {
194                 await expect(page).toHaveURL(/qr-codes/)
195                 // Logged-in routes often drop the /es/ prefix; language stays on the user.
196                 await expect(page.locator('html')).toHaveAttribute(
197                     'lang',
198                     new RegExp(`^${LOCALIZED_TRIAL.locale}`, 'i')
199                 )
200                 await pageManager.myQrCodesPage.expectPaidTrialDownloadModalWithoutFreeBanners()
201                 await expectPaidTrialStateOnDashboard(page, pageManager)
202             })
203 
204             await test.step('TC28: Account shows localised 14-day plan; dashboard has no EN free countdown', async () => {
205                 // Prefer locale-prefixed account URL; fall back if the app strips it.
206                 await page.goto(
207                     `/${LOCALIZED_TRIAL.locale}/account#subscriptions`
208                 )
209                 if (!page.url().includes('account')) {
210                     await page.goto('/account#subscriptions')
211                 }
212                 await expect(
213                     pageManager.paymentPage.locator.planActiveLabel
214                 ).toBeVisible({ timeout: 30_000 })
215                 // data-qa-id stays English; visible label uses the website language pack.
216                 await expect(
217                     pageManager.paymentPage.locator.days14FullAccessPlanLabel
218                 ).toContainText(LOCALIZED_TRIAL.plan14FullAccess)
219                 await expect(
220                     pageManager.paymentPage.locator.subscriptionsPane
221                 ).not.toContainText('10-day free trial')
222                 await expect(
223                     pageManager.paymentPage.locator.subscriptionsPane
224                 ).not.toContainText('days remaining')
225 
226                 const accountText = await page.locator('body').innerText()
227                 expect(
228                     accountText.includes(daysRemaining14) ||
229                         accountText.includes(daysRemaining13) ||
230                         accountText.includes(LOCALIZED_TRIAL.plan14FullAccess),
231                     `Expected localised trial text (${daysRemaining14} / ${daysRemaining13} or ${LOCALIZED_TRIAL.plan14FullAccess})`
232                 ).toBe(true)
233 
234                 await page.goto('/qr-codes')
235                 await expect(page.locator('html')).toHaveAttribute(
236                     'lang',
237                     new RegExp(`^${LOCALIZED_TRIAL.locale}`, 'i')
238                 )
239                 await expectPaidTrialStateOnDashboard(page, pageManager)
240 
241                 const dashboardText = await page.locator('body').innerText()
242                 expect(dashboardText).not.toContain(
243                     formatDaysRemaining(
244                         PAID_TRIAL_LOCALE_COPY.en.daysRemainingTemplate,
245                         10
246                     )
247                 )
248                 // Free-trial English countdown must not appear for a paid localised user.
249                 expect(dashboardText).not.toContain('days remaining')
250 
251                 // Paid users hide the free-trial sidebar countdown (TC03). When the
252                 // countdown is present it must use the website-pack translation.
253                 const countdown = page.getByTestId(
254                     'sidebar-plan-upgrade-countdown'
255                 )
256                 if (await countdown.isVisible().catch(() => false)) {
257                     const countdownText = await countdown.innerText()
258                     expect(
259                         countdownText.includes(daysRemaining14) ||
260                             countdownText.includes(daysRemaining13),
261                         `Countdown must match ${daysRemaining14} or ${daysRemaining13}`
262                     ).toBe(true)
263                 }
264             })
265         }
