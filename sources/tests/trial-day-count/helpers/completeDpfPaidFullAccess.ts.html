39 /**
40  * Dismiss Step 3 help popup (X closes the whole tour; Continue only advances),
41  * then click Create and wait for the DPF email form.
42  */
43 async function clickStep3CreateForDpf(
44     page: Page,
45     pageManager: PageManager
46 ): Promise<void> {
47     const popup = pageManager.qrCreationPage.locator.step3OnboardingCarousel
48     const appeared = await popup
49         .waitFor({ state: 'visible', timeout: 15_000 })
50         .then(() => true)
51         .catch(() => false)
52 
53     if (appeared) {
54         const closeIcon =
55             pageManager.qrCreationPage.locator.closeCarouselButtonSt3
56         for (let attempt = 0; attempt < 3; attempt++) {
57             if (!(await popup.isVisible().catch(() => false))) break
58             await closeIcon.click({ timeout: 5_000 }).catch(async () => {
59                 await page.keyboard.press('Escape').catch(() => undefined)
60             })
61             await popup
62                 .waitFor({ state: 'hidden', timeout: 5_000 })
63                 .catch(() => undefined)
64         }
65         await pageManager.qrCreationPage.locator.qrBackdrop
66             .waitFor({ state: 'hidden', timeout: 5_000 })
67             .catch(() => undefined)
68     }
69 
70     await pageManager.qrCreationPage.locator.createButton.click()
71     await expect(pageManager.qrCreationPage.locator.emailInput).toBeVisible({
72         timeout: 60_000,
73     })
74 }

76 /**
77  * Submits the DPF email form and waits for the plan step. The submit handler is
78  * bound after the form paints, so an early click is silently dropped and the
79  * page stays on the email step — retry until the plan step renders.
80  */
81 async function submitDpfEmailAndReachPlan(
82     page: Page,
83     pageManager: PageManager,
84     email: string
85 ): Promise<void> {
86     const emailInput = pageManager.qrCreationPage.locator.emailInput
87     const submitButton = pageManager.qrCreationPage.locator.dpfFormSubmitButton
88     const planContinue =
89         pageManager.qrCreationPage.locator.continueUserPlanButton
90 
91     for (let attempt = 0; attempt < 3; attempt++) {
92         if (await planContinue.isVisible().catch(() => false)) return
93 
94         if (await emailInput.isVisible().catch(() => false)) {
95             const current = await emailInput.inputValue().catch(() => '')
96             if (current !== email) {
97                 await emailInput.fill(email)
98             }
99             await submitButton.click({ timeout: 30_000 }).catch(() => undefined)
100         }
101 
102         const reached = await planContinue
103             .waitFor({ state: 'visible', timeout: 45_000 })
104             .then(() => true)
105             .catch(() => false)
106         if (reached) return
107     }
108 
109     await expect(planContinue).toBeVisible({ timeout: 30_000 })
110 }

129 async function createQrForDpf(
130     page: Page,
131     pageManager: PageManager,
132     qrType: DpfQrType,
133     entryPath?: string
134 ): Promise<void> {
135     if (isGenerateEntry(entryPath)) {
136         // Generate entry only has a dedicated website CTA in current markup;
137         // other QR types still go through /create step1 cards.
138         if (qrType !== 'website') {
139             throw new Error(
140                 `DPF /generate entry currently supports qrType=website only (got ${qrType})`
141             )
142         }
143         await createWebsiteQrViaGenerate(page, pageManager)
144         return
145     }
146 
147     switch (qrType) {
148         case 'wifi':
149             await pageManager.qrCreationPage.wifiQrCreate()
150             break
151         case 'instagram':
152             await pageManager.qrCreationPage.instagramQrCreate()
153             break
154         case 'website':
155         default:
156             await pageManager.qrCreationPage.websiteQrCreate()
157             break
158     }
159     await pageManager.qrCreationPage.clickStep2NextButton()
160 }

180 /**
181  * Runs DPF create → email → plan → Stripe payment fill + submit.
182  * Stops after submit (does not wait for thank-you).
183  */
184 export async function completeDpfThroughPayment(
185     page: Page,
186     pageManager: PageManager,
187     email: string,
188     options?: DpfFlowOptions
189 ): Promise<void> {
190     const cardNumber = options?.cardNumber ?? stripeVisaValidCard.cardNumber
191     const expDate = options?.expDate ?? stripeVisaValidCard.expDate
192     const cvc = options?.cvc ?? stripeVisaValidCard.cvc
193     const submitAsEntered = options?.submitAsEntered ?? false
194     const doubleSubmit = options?.doubleSubmit ?? false
195     const plan = options?.plan ?? 'full'
196     const qrType = options?.qrType ?? 'website'
197 
198     if (options?.entryPath) {
199         await page.goto(options.entryPath)
200     } else {
201         await openMainPageDpf(page)
202     }
203 
204     await createQrForDpf(page, pageManager, qrType, options?.entryPath)
205     await clickStep3CreateForDpf(page, pageManager)
206 
207     await submitDpfEmailAndReachPlan(page, pageManager, email)
208 
209     await selectDpfPlan(pageManager, plan)
210 
211     // Select country/zip before card so Stripe PE does not remount mid-fill.
212     await pageManager.paymentPage.selectCountryAndZipInPaymentFrame()
213     await pageManager.paymentPage.makePayment(cardNumber, expDate, cvc)
214 
215     if (submitAsEntered) {
216         await pageManager.paymentPage.makePayment(cardNumber, expDate, cvc)
217         const submitAsEnteredBtn =
218             pageManager.paymentPage.locator.submitPaymentButton
219         if (doubleSubmit) {
220             await submitAsEnteredBtn.dblclick({ force: true })
221         } else {
222             await submitAsEnteredBtn.click({ force: true })
223         }
224         return
225     }
226 
227     const submit = pageManager.paymentPage.locator.submitPaymentButton
228     const enabled = await submit
229         .waitFor({ state: 'visible', timeout: 15_000 })
230         .then(() => submit.isEnabled())
231         .catch(() => false)
232     if (enabled) {
233         if (doubleSubmit) {
234             // Rapid double-submit must still create at most one trial/charge.
235             await submit.dblclick({ force: true })
236         } else {
237             await submit.click()
238         }
239     }
240     await expect(page).toHaveURL(/pay-thank-you-dpf/, { timeout: 90_000 })
241 }
