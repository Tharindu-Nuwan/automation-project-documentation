116 /**
117  * Asserts an email landed inside `slaMs`. Elapsed time is Mailpit
118  * `sent_at` minus the trigger submit; wall-clock is the fallback when
119  * `sent_at` is unparseable or behind the runner clock.
120  */
121 function assertWithinSla(
122     label: string,
123     triggeredAt: number,
124     sentAt: string,
125     slaMs: number
126 ): void {
127     expect(sentAt, `${label}: Mailpit did not return sent_at`).toBeTruthy()
128 
129     const wallClockMs = Date.now() - triggeredAt
130     const sentAtMs = Date.parse(sentAt)
131     const fromSentAt = sentAtMs - triggeredAt
132     const elapsedMs =
133         Number.isNaN(sentAtMs) || fromSentAt < 0 ? wallClockMs : fromSentAt
134 
135     expect(
136         elapsedMs,
137         `${label} took ${elapsedMs}ms (wall-clock ${wallClockMs}ms, Mailpit sent_at=${sentAt}); SLA is ${slaMs}ms`
138     ).toBeLessThan(slaMs)
139 }

141 /**
142  * OQG3-2290-TC42 — Mailpit HTML, plain-text, and raw MIME must not
143  * contain auth tokens or secrets. `password` is the account secret
144  * used at register (distinct from the recipient address).
145  */
146 async function assertNoSensitiveDataInEmail(
147     mailpit: MailpitApi,
148     messageId: string,
149     label: string,
150     password?: string
151 ): Promise<void> {
152     const html = await mailpit.getMessageHtml(messageId)
153     const text = await mailpit.getMessageText(messageId)
154     const raw = await mailpit.getMessageRaw(messageId)
155     const parts: [string, string][] = [
156         ['HTML', html],
157         ['text', text],
158         ['raw', raw],
159     ]
160 
161     for (const [part, body] of parts) {
162         const leaks = findSensitiveDataLeaks(body)
163         expect(
164             leaks,
165             `${label} ${part} must not leak secrets (${leaks.join(', ')})`
166         ).toEqual([])
167     }
168 
169     if (!password) {
170         return
171     }
172 
173     const encoded = encodeURIComponent(password)
174     for (const [part, body] of parts) {
175         expect(
176             body.includes(password),
177             `${label} ${part} must not contain the account password`
178         ).toBe(false)
179         if (encoded !== password) {
180             expect(
181                 body.includes(encoded),
182                 `${label} ${part} must not contain a URL-encoded account password`
183             ).toBe(false)
184         }
185     }
186 }
