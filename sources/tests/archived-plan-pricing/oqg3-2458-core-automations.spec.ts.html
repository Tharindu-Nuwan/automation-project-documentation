30 test.describe('OQG3-2458 DPF trial user buys Annual via Archived P&P', () => {
31     let emailsToDelete: string[]
32 
33     test.beforeEach(({ product }) => {
34         test.skip(product !== 'oqg', 'OQG3-2458 is implemented for OQG first')
35         emailsToDelete = []
36     })
37 
38     test.afterEach(async ({ apiManager }) => {
39         if (emailsToDelete.length > 0) {
40             await apiManager.userApi
41                 .deleteUser(emailsToDelete)
42                 .catch(() => undefined)
43         }
44     })
45 
46     test(
47         'DPF 14-day trial user buying Annual via Archived P&P creates one Stripe customer',
48         { tag: ['@High', '@Smoke'] },
49         async ({ page, pageManager, apiManager, browser, env, product }) => {
50             qase.id(0)
51             qase.title(
52                 'DPF 14-day trial user buying Annual via Archived P&P creates one Stripe customer'
53             )
54             qase.suite(SUITE)
55 
56             const seedEmail = pageManager.registerPage.createUserData().email
57             const buyer = pageManager.registerPage.createUserData(true)
58             emailsToDelete.push(seedEmail, buyer.email)
59 
60             const abandoned =
61                 await test.step('Create an abandoned QR', async () => {
62                     return createAbandonedQr({
63                         browser,
64                         apiManager,
65                         product,
66                         baseURL: env.baseURL,
67                         scanBaseURL: env.scanBaseURL,
68                         seedEmail,
69                     })
70                 })
71             emailsToDelete.push(abandoned.tempEmail)
72 
73             await test.step('TC15: DPF 14-day full-access trial (bug-path buyer)', async () => {
74                 await completeDpfPaidFullAccess(page, pageManager, buyer.email)
75                 await pageManager.qrCreationPage.locator.closeDownloadModalButton
76                     .click()
77                     .catch(() => undefined)
78             })
79 
80             await test.step('TC01: DPF trial payment sets onboarding_funnel to 4', async () => {
81                 await assertDpfFunnelOnCustomer(apiManager, buyer.email)
82             })
83 
84             const sentBefore = Date.now()
85 
86             await test.step('TC10/TC11: buy Annual via Archived P&P', async () => {
87                 await openArchivedPlanAndPricing(
88                     page,
89                     pageManager,
90                     abandoned.scanUrl
91                 )
92                 await buyOnArchivedPlanAndPricing(page, pageManager, {
93                     plan: 'Annually',
94                 })
95             })
96 
97             const customer =
98                 await test.step('TC01/TC03-TC08/TC31/TC52: one customer, one payment, funnel 5, Annual active', async () => {
99                     return assertSingleStripeCustomerAfterArchivedPurchase({
100                         apiManager,
101                         realEmail: buyer.email,
102                         tempEmail: abandoned.tempEmail,
103                         expectedPlan: 'Annually',
104                     })
105                 })
106 
107             await test.step('TC29: mapping stays single after refresh', async () => {
108                 await page.reload({ waitUntil: 'domcontentloaded' })
109                 const again = await apiManager.stripeApi.listCustomersByEmail(
110                     buyer.email
111                 )
112                 expect(again).toHaveLength(1)
113                 expect(again[0].id).toBe(customer.id)
114             })
115 
116             await test.step('TC30: mapping stays single after re-login', async () => {
117                 const password = pageManager.accountPage.fakePassword()
118                 await page.goto('/account', { waitUntil: 'domcontentloaded' })
119                 await pageManager.accountPage.locator.passwordInput.waitFor({
120                     state: 'visible',
121                 })
122                 await pageManager.accountPage.setNewPasswordSuccessfully(
123                     password
124                 )
125                 await pageManager.accountPage.logout()
126                 await page.waitForURL(/login|register|\//)
127                 await page.goto('/login', { waitUntil: 'domcontentloaded' })
128                 await pageManager.loginPage.logIn(buyer.email, password)
129                 await page.waitForURL(/qr-codes/, { timeout: 60_000 })
130                 const again = await apiManager.stripeApi.listCustomersByEmail(
131                     buyer.email
132                 )
133                 expect(again).toHaveLength(1)
134                 expect(again[0].id).toBe(customer.id)
135             })
136 
137             await test.step('TC45: purchase confirmation email is sent once', async () => {
138                 if (!process.env.MAILPIT_BASE_URL) {
139                     return
140                 }
141                 const purchaseSubject = getExpectedEmailSubjects(product)['18']
142                 const mailpit = new MailpitApi()
143                 const message = await mailpit.waitForEmail(
144                     buyer.email,
145                     sentBefore,
146                     90_000,
147                     5_000,
148                     purchaseSubject
149                 )
150                 const count = await mailpit.countEmails(
151                     buyer.email,
152                     sentBefore,
153                     purchaseSubject
154                 )
155                 expect(count, 'Expected exactly one purchase email').toBe(1)
156                 const raw = await mailpit.getMessageRaw(message.id)
157                 const templateId = extractTemplateIdFromRaw(raw)
158                 expect(templateId).toBe(
159                     expectedEmailTemplates['18'].templateId
160                 )
161             })
162         }
163     )
