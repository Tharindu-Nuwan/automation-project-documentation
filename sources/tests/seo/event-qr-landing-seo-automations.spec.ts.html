tests/seo/event-qr-landing-seo-automations.spec.ts

Bundled source snapshot. Credential-like literals are omitted. Line numbers correspond to the audited repository file; this page is for reading and is not executable test code.

1 /**
2  * Jira: OQG3-2390 (Event QR coverage)
3  * Feature: SEO / HTML / a11y audit of the Event QR surfaces
4  *
5  * The static page matrix (`seo-audit-page-matrix.spec.ts`) audits fixed public
6  * routes only. Event has no such route — it is reached through
7  * `/create?step=1` — so it needs its own spec. Two surfaces matter:
8  *
9  *   1. The Event **wizard Step 2** page (guest-reachable today) — catches
10  *      heading/alt/landmark regressions in the Event form itself.
11  *   2. The published Event **landing (scan) page** — the actually-indexable
12  *      surface, named in `docs/SEO_Audit_Report.md` as the highest-value SEO
13  *      target still untested.
14  *
15  * History: Event Step 2 save was returning HTTP 422 (DEV-EVENT-STEP3-422)
16  * earlier on 2026-09-01 and now succeeds on devii, so this suite publishes a
17  * real Event QR. The published QR currently resolves to a `/no-data`
18  * placeholder rather than an Event landing page; the test asserts that
19  * explicitly so the audit can never silently score an error screen.
20  */
21 
22 import { expect } from '@playwright/test'
23 import { qase } from 'playwright-qase-reporter'
24 
25 import { test } from '../../fixtures/baseFixture'
26 import { openEventStep2 } from '../helpers/eventLocalSetup'
27 import {
28     auditCurrentPage,
29     auditUrl,
30     attachReport,
31     describeReport,
32 } from '../helpers/seoAudit'
33 
34 const SUITE = 'SEO audit — Event QR'
35 
36 // Test cases covered by this spec: EVQR-SEO-01 (wizard Step 2),
37 //   EVQR-SEO-02 (published landing page)
38 test.describe(SUITE, { tag: ['@SEO', '@Regression'] }, () => {
39     /**
40      * @covers EVQR-SEO-01
41      */
42     test(
43         'Event QR wizard Step 2 passes the SEO/HTML audit',
44         {
45             tag: ['@SEO', '@Regression', '@High'],
46             annotation: [{ type: 'testId', description: 'EVQR-SEO-01' }],
47         },
48         async ({ page, pageManager }, testInfo) => {
49             qase.id(0)
50             qase.title('Event QR wizard Step 2 passes the SEO/HTML audit')
51             qase.suite(SUITE)
52 
53             await openEventStep2(page, pageManager)
54 
55             const report = await auditCurrentPage(page)
56             await attachReport(testInfo, report)
57             expect(report.criticalFailures, describeReport(report)).toEqual([])
58         }
59     )
60 
61     /**
62      * @covers EVQR-SEO-02
63      */
64     test(
65         'published Event QR landing page passes the SEO/HTML audit',
66         {
67             tag: ['@SEO', '@Regression', '@High'],
68             annotation: [{ type: 'testId', description: 'EVQR-SEO-02' }],
69         },
70         async ({ page, pageManager, apiManager }, testInfo) => {
71             qase.id(0)
72             qase.title(
73                 'published Event QR landing page passes the SEO/HTML audit'
74             )
75             qase.suite(SUITE)
76 
77             const userData = await pageManager.registerPage.authorizeUser()
78             try {
79                 await page.goto('/qr-codes')
80                 await pageManager.menuPage.goToCreateQrCode()
81 
82                 // Same entry sequence the Event Step 3 suite uses: a fresh
83                 // user's first click can select the card without navigating.
84                 const eventCard = pageManager.step1Page.locator.eventQrType
85                 await eventCard.waitFor({ state: 'visible' })
86                 await eventCard.click()
87                 const reached = await page
88                     .waitForURL(/step=2/, { timeout: 15_000 })
89                     .then(() => true)
90                     .catch(() => false)
91                 if (!reached) {
92                     await eventCard.click()
93                     await expect(page).toHaveURL(/step=2/)
94                 }
95                 await pageManager.step2Page.ensureHelpClosedBeforeUpload()
96 
97                 const title = `SEO Event ${Date.now()}` // event title, not QR name
98                 await pageManager.step2Page.fillEventRequiredFields(title)
99                 await pageManager.qrCreationPage.clickStep2NextButton()
100                 await page.waitForURL('**step=3**', { timeout: 60_000 })
101                 await pageManager.qrCreationPage.clickStep3CreateButton()
102                 await pageManager.qrCreationPage.closeDownloadModalIfPresent()
103 
104                 // Resolve the public landing URL from the dashboard card.
105                 // The dashboard shows the QR *name*, which is a separate field
106                 // from the event title, so match on position instead: this user
107                 // was just registered and owns exactly one QR.
108                 await pageManager.myQrCodesPage.goToRootMyQrCodes()
109                 const scanLink = page
110                     .locator('a:has(span.overflow-link)')
111                     .first()
112                 await expect(scanLink).toBeVisible({ timeout: 15_000 })
113                 const landingUrl = await scanLink.getAttribute('href')
114                 expect(
115                     landingUrl,
116                     'the published Event QR must expose a public landing link'
117                 ).toBeTruthy()
118                 // Guard: a `/no-data` (or similar) placeholder means the QR did
119                 // not publish its Event content. Auditing that page would score
120                 // an error screen and report a misleading pass/fail, so fail
121                 // loudly here instead.
122                 expect(
123                     landingUrl,
124                     `published Event QR resolved to a placeholder page (${landingUrl}) instead of its Event landing page`
125                 ).not.toMatch(/\/(no-data|not-found|404)\b/)
126 
127                 // Audit as an anonymous visitor — what a crawler actually sees.
128                 const browser = page.context().browser()
129                 const anonContext = await browser!.newContext()
130                 try {
131                     const anonPage = await anonContext.newPage()
132                     const report = await auditUrl(
133                         anonPage,
134                         landingUrl as string
135                     )
136                     await attachReport(testInfo, report)
137                     expect(
138                         report.criticalFailures,
139                         describeReport(report)
140                     ).toEqual([])
141                 } finally {
142                     await anonContext.close()
143                 }
144             } finally {
145                 await apiManager.userApi
146                     .deleteUser([userData.email])
147                     .catch(() => {})
148             }
149         }
150     )
151 })
