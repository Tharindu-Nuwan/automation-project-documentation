Bundled source snapshot. Credential-like literals are omitted. Line numbers correspond to the audited repository file; this page is for reading and is not executable test code.

1 import { expect } from '@playwright/test'
2 import { qase } from 'playwright-qase-reporter'
3 import { test } from '../../fixtures/baseFixture'
4 import { auditUrl, attachReport, describeReport } from '../helpers/seoAudit'
5 
6 /**
7  * Site-wide SEO / HTML / a11y audit across the public + QR-type pages on the
8  * configured BASE_URL (defaults to dev). SEO is per-page, not home-only, so
9  * every indexable route is audited.
10  *
11  * Event QR has no fixed public route — it is reached through `/create?step=1`,
12  * already audited here as `/en/create`. Its indexable surface is the published
13  * landing (scan) page, which only exists after a QR is created, so it is
14  * audited separately in `event-qr-landing-seo-automations.spec.ts`.
15  */
16 
17 interface SeoPage {
18     name: string
19     path: string
20     qaseId: number
21 }
22 
23 // Public / marketing / legal / auth pages (no auth required).
24 const PAGES: SeoPage[] = [
25     { name: 'home', path: '/en/', qaseId: 9230 },
26     { name: 'create (QR type selection)', path: '/en/create', qaseId: 9231 },
27     { name: 'plans and prices', path: '/en/plans-and-prices', qaseId: 9232 },
28     { name: 'faq', path: '/en/faq', qaseId: 9233 },

40 test.describe('SEO audit — site pages', () => {
41     PAGES.forEach((p) => {
42         test(
43             `SEO audit: ${p.name}`,
44             { tag: ['@SEO', '@Regression'] },
45             async ({ page }, testInfo) => {
46                 qase.id(p.qaseId)
47                 qase.title(`SEO audit: ${p.name}`)
48                 qase.suite('SEO')
49                 const report = await auditUrl(page, p.path)
50                 await attachReport(testInfo, report)
51                 expect(report.criticalFailures, describeReport(report)).toEqual(
52                     []

54             }
