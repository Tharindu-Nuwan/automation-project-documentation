26 /** DPF funnel from entry through registration, landing on the DPF Plans & Pricing. */
27 async function dpfFunnelToPandP(
28     page: Page,
29     pageManager: PageManager,
30     email: string
31 ): Promise<void> {
32     await openMainPageDpf(page) // /create?step=1&qr_onboarding=active_dpf
33     await pageManager.qrCreationPage.websiteQrCreate()
34     await pageManager.qrCreationPage.clickStep2NextButton()
35     await pageManager.qrCreationPage.clickStep3CreateButton()
36     await pageManager.qrCreationPage.locator.emailInput.fill(email)
37     await pageManager.qrCreationPage.locator.dpfFormSubmitButton.click()
38     // Wait for the DPF Plans & Pricing to be fully interactive before returning:
39     // its CFF-conversion handler (back/logo/close) only binds once the page has
40     // loaded, so an early back/logo click just navigates without converting.
41     // (Event-based replacement for the legacy waitForTimeout in funnels/cff.)
42     await pageManager.qrCreationPage.locator.continueUserPlanButton.waitFor({
43         state: 'visible',
44     })
45 }
