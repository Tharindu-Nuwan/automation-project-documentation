12 /**
13  * Reads a date row's actual submitted value (never the placeholder).
14  *
15  * The visible date lives in an <input> (as `value` once set, or as
16  * `placeholder` before that). A transparent, full-size
17  * `<button title="open date picker">` is layered on top of it, so reading the
18  * button's text returns "" even when a date is clearly on screen.
19  * @covers OQG3-2366-AC02
20  * @covers OQG3-2366-AC03
21  */
22 async function eventDateShownValue(
23     pageManager: PageManager,
24     edge: 'start' | 'end'
25 ): Promise<string> {
26     return pageManager.step2Page.page.evaluate((which) => {
27         const trigger = document.querySelector(
28             `[data-qa-id="step2-event-date-0-${which}-date-trigger"]`
29         )
30         let scope: HTMLElement | null = trigger as HTMLElement | null
31         for (let i = 0; i < 4 && scope?.parentElement; i++) {
32             scope = scope.parentElement
33         }
34         const input = scope?.querySelector(
35             'input:not([data-qa-id])'
36         ) as HTMLInputElement | null
37         // Deliberately the value only. Placeholder text looks like a date on
38         // screen but is not submitted, so it is not a default.
39         return input ? input.value : ''
40     }, edge)
41 }

44     eventDateShownValue(pageManager, 'start')
45 const eventEndDateValue = (pageManager: PageManager) =>
