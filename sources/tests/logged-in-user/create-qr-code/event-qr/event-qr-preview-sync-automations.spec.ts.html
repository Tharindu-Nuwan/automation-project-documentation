1 /**
2  * Jira: OQG3-2367 / OQG3-2368 / OQG3-2371 / OQG3-2372 / OQG3-2376
3  * Feature: Event Step 2 — real-time mobile mockup preview
4  *
5  * Every one of these stories ends with a "Scanner mobile display behavior"
6  * block whose first sub-criterion is the same promise: what the creator types
7  * in Step 2 shows up in the mobile mockup immediately. Nothing asserted that.
8  * The existing "reflects immediately" tests check the *input* they just filled,
9  * which cannot fail, so the preview was never actually exercised.
10  *
11  * The preview is an iframe that re-renders a beat behind the keystroke, so
12  * every assertion polls it rather than reading it once.
13  */

24 /** How long the mockup gets to catch up with a Step 2 edit. */
25 const PREVIEW_TIMEOUT = 30_000
26 
27 /** Opens Step 2 of a fresh Event QR with the required fields already filled. */
28 async function openEventStep2(
29     page: Page,
30     pageManager: PageManager,
31     title: string
32 ): Promise<void> {
33     await page.goto('/qr-codes')
34     await pageManager.menuPage.goToCreateQrCode()
35     await pageManager.step1Page.locator.eventQrType.waitFor({
36         state: 'visible',
37         timeout: 30_000,
38     })
39     await pageManager.step1Page.selectEventQrType()
40     await pageManager.step2Page.ensureHelpClosedBeforeUpload().catch(() => {})
41     await pageManager.step2Page.fillEventRequiredFields(title)
42 }
43 
44 /** Current text of the mobile mockup preview, or '' while it is unavailable. */
45 async function previewText(pageManager: PageManager): Promise<string> {
46     return pageManager.step2Page.locator.frameView
47         .locator('body')
48         .innerText()
49         .catch(() => '')
50 }
51 
52 /** Waits for `needle` to reach the mockup preview. */
53 async function expectInPreview(
54     pageManager: PageManager,
55     needle: string,
56     because: string
57 ): Promise<void> {
58     await expect
59         .poll(() => previewText(pageManager), {
60             timeout: PREVIEW_TIMEOUT,
61             message: because,
62         })
63         .toContain(needle)
64 }
