1 /**
2  * Jira: OQG3-2393 — View and Manage Completed Event QR Code
3  * Feature: a published Event QR on My QR Codes
4  *
5  * Covers what the dashboard offers once an Event QR exists: how the card is
6  * listed, the actions reachable from it, downloading the code, and its presence
7  * in Analytics. The temp-user save path (sign-up after Step 3) is covered too,
8  * since it is a distinct save flow rather than a variation of the logged-in one.
9  */
10 
11 import { expect } from '@playwright/test'
12 import { qase } from 'playwright-qase-reporter'
13 import fs from 'node:fs'
14 import os from 'node:os'
15 import path from 'node:path'
16 
17 import { test } from '../../../../fixtures/baseFixture'
18 import { loginCreateAndPublishEventQr } from '../../../helpers/eventLocalSetup'
19 
20 const SUITE = 'Event QR Dashboard Management'
21 
22 // Acceptance criteria covered by this spec: OQG3-2393-AC02, AC03, AC04, AC12,
23 //   AC17, AC18, AC20.
24 // AC14/AC15 are held by EVQR-400 in qr-code-details/event-qr-edit-e2e.spec.ts;
25 // AC25/AC26 by EVQR-G06 in event-qr-resilience.spec.ts.
26 test.describe(SUITE, { tag: ['@High'] }, () => {
27     let userData: { email: string; password: string }
28

30         if (userData?.email) {
31             await apiManager.userApi
32                 .deleteUser([userData.email])
33                 .catch(() => {})
34         }
35     })

37     /**
38      * @covers OQG3-2393-AC12
39      * @covers OQG3-2393-AC04
40      * @covers OQG3-2393-AC12.01
41      * @covers OQG3-2393-AC12.02
42      * @covers OQG3-2393-AC12.03
43      * @covers OQG3-2393-AC12.04
44      */
45     test(
46         'OQG3-2393-AC12/AC04 the card lists the QR with its name, Event type and thumbnail, and offers its actions',
47         {
48             tag: ['@Regression', '@High'],
49             annotation: [
50                 { type: 'testId', description: 'OQG3-2393-AC12' },
51                 { type: 'testId', description: 'OQG3-2393-AC04' },
52                 { type: 'testId', description: 'OQG3-2393-AC12.01' },
53                 { type: 'testId', description: 'OQG3-2393-AC12.02' },
54                 { type: 'testId', description: 'OQG3-2393-AC12.03' },
55                 { type: 'testId', description: 'OQG3-2393-AC12.04' },
56             ],
57         },
58         async ({ page, pageManager }) => {
59             qase.id(0)
60             qase.title(
61                 'OQG3-2393-AC12/AC04 the card lists the QR with its name, Event type and thumbnail, and offers its actions'
62             )
63             qase.suite(SUITE)
64             test.slow()
65 
66             const published = await loginCreateAndPublishEventQr(
67                 page,
68                 pageManager,
69                 { setQrName: true }
70             )
71             userData = published.user
72             const myQrCodes = pageManager.myQrCodesPage
73 
74             // AC12 — name, type label, creation date and a thumbnail.
75             await expect(
76                 myQrCodes.getCreatedQrCodeName(published.qrName),
77                 'the card must show the QR name entered in Step 2'
78             ).toBeVisible({ timeout: 30_000 })
79             await expect(
80                 page.getByText('Event', { exact: true }).first(),
81                 'the card must label the QR type as Event'
82             ).toBeVisible()
83 
84             const listing = await page.evaluate(() => {
85                 const body = document.body.innerText
86                 return {
87                     // The card renders a full month name, e.g. "September 9, 2026".
88                     hasDate:
89                         /[A-Z][a-z]{2,8}\s+\d{1,2},\s*\d{4}|\d{1,2}[./-]\d{1,2}[./-]\d{2,4}|\d{4}-\d{2}-\d{2}/.test(
90                             body
91                         ),
92                     qrImages: Array.from(document.querySelectorAll('img'))
93                         .map((i) => i.getAttribute('src') || '')
94                         .filter((src) => /qr|code/i.test(src)).length,
95                 }
96             })
97             expect(listing.hasDate, 'the card must show a creation date').toBe(
98                 true
99             )
100             expect(
101                 listing.qrImages,
102                 'the card must show a QR thumbnail'
103             ).toBeGreaterThan(0)
104 
105             // AC04 — edit, share and delete are reachable from the card.
106             await myQrCodes.locator.qrKebabButton.click()
107             const menu = page.locator('div.dropdown-menu').filter({
108                 has: page.getByText('Edit QR Content'),
109             })
110             await expect(menu).toBeVisible()
111             for (const action of [
112                 'Edit QR Content',
113                 'Edit QR Design',
114                 'Send to',
115                 'Delete',
116             ]) {
117                 await expect(
118                     menu.getByText(action, { exact: true }),
119                     `the card menu must offer "${action}"`
120                 ).toBeVisible()
121             }
122             await page.keyboard.press('Escape')
123 
124             // ...and Analytics is reachable from the dashboard chrome.
125             await expect(
126                 page.getByTestId('sidebar-analytics-link'),
127                 'Analytics must be reachable from the dashboard'
128             ).toBeVisible()
129         }
130     )
131 
132     /**
133      * @covers OQG3-2393-AC17
134      * @covers OQG3-2393-AC18
135      */
136     test(
137         'OQG3-2393-AC17/AC18 the published Event QR downloads as a file',
138         {
139             tag: ['@Regression', '@High'],
140             annotation: [
141                 { type: 'testId', description: 'OQG3-2393-AC17' },
142                 { type: 'testId', description: 'OQG3-2393-AC18' },
143             ],
144         },
145         async ({ page, pageManager }) => {
146             qase.id(0)
147             qase.title(
148                 'OQG3-2393-AC17/AC18 the published Event QR downloads as a file'
149             )
150             qase.suite(SUITE)
151             test.slow()
152 
153             const published = await loginCreateAndPublishEventQr(
154                 page,
155                 pageManager,
156                 { setQrName: true }
157             )
158             userData = published.user
159 
160             const downloadDir = fs.mkdtempSync(
161                 path.join(os.tmpdir(), 'event-qr-download-')
162             )
163             try {
164                 const filePath =
165                     await pageManager.myQrCodesPage.performQrDownload(
166                         downloadDir,
167                         'png',
168                         'default'
169                     )
170                 const stats = fs.statSync(filePath)
171                 expect(
172                     stats.size,
173                     'the downloaded QR file must not be empty'
174                 ).toBeGreaterThan(0)
175                 expect(
176                     path.basename(filePath),
177                     'the download must be a PNG'
178                 ).toMatch(/\.png$/i)
179             } finally {
180                 fs.rmSync(downloadDir, { recursive: true, force: true })
181             }
182         }
183     )
184 
185     /** @covers OQG3-2393-AC20 */
186     test(
187         'OQG3-2393-AC20 the published Event QR is listed in the Analytics section',
188         {
189             tag: ['@Regression', '@Medium'],
190             annotation: [{ type: 'testId', description: 'OQG3-2393-AC20' }],
191         },
192         async ({ page, pageManager }) => {
193             qase.id(0)
194             qase.title(
195                 'OQG3-2393-AC20 the published Event QR is listed in the Analytics section'
196             )
197             qase.suite(SUITE)
198             test.slow()
199 
200             const published = await loginCreateAndPublishEventQr(
201                 page,
202                 pageManager,
203                 { setQrName: true }
204             )
205             userData = published.user

213                 'the published Event QR must appear in Analytics'

215         }
