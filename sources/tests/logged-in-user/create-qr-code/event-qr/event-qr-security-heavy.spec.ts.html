19 /**
20  * Event QR — HEAVY security suite (Step 2 create UI, guest funnel).
21  *
22  * Goal: try to break the site. Every payload is fed to a real field and the
23  * test passes ONLY if nothing executes. The primary signal is a dialog spy —
24  * an alert()/confirm() firing means an XSS payload actually ran. A secondary
25  * DOM sweep confirms no executable markup landed in the editor or the live
26  * preview iframe.
27  *
28  * A payload that executes, or an input that breaks the save flow, is a real
29  * PRODUCT bug — document it and quarantine the test; never weaken the assertion.
30  */
31 
32 /** Attaches a dialog spy. Any captured message = script executed. */
33 function trackDialogs(page: Page): string[] {
34     const seen: string[] = []
35     page.on('dialog', async (d) => {
36         seen.push(d.message())
37         await d.dismiss().catch(() => {})
38     })
39     return seen
40 }

42 /** No executable markup in the editor page nor the preview iframe. */
43 async function expectInert(page: Page, frameView: Locator): Promise<void> {
44     const badPage = page.locator(
45         'script:has-text("alert(1)"), [onerror*="alert"], svg[onload], [onload*="alert"], iframe[src^="javascript"]'
46     )
47     await expect(badPage).toHaveCount(0)
48     const badFrame = frameView.locator(
49         'script:has-text("alert(1)"), [onerror*="alert"], svg[onload], iframe[src^="javascript"]'
50     )
51     await expect(badFrame).toHaveCount(0)
52 }
