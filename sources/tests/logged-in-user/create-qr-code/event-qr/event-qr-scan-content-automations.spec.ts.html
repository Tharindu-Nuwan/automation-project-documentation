74 /**
75  * Document order of the landing page's title, summary, image and link, so the
76  * arrangement OQG3-2367-AC05.02 specifies can be asserted rather than assumed.
77  */
78 async function landingOutline(scan: Page): Promise<string[]> {
79     return scan.evaluate(() =>
80         Array.from(document.querySelectorAll('h1,p,img,a'))
81             .map((el) => {
82                 const text = (el.textContent || '').trim()
83                 if (el.tagName === 'IMG') return 'IMG'
84                 if (el.tagName === 'H1') return `H1:${text}`
85                 if (el.tagName === 'A') return `A:${text}`
86                 return `P:${text}`
87             })
88             .filter(Boolean)
89     )
90 }
