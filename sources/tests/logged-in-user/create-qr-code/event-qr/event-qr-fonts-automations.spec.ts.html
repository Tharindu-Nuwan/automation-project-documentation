270     /** @covers TC-2381-014 @covers TC-2381-017 @covers OQG3-2381-AC07 */
271     test(
272         'TC-2381-014/017 the Title font drives the event title and section headings',
273         {
274             tag: ['@Regression', '@High'],
275             annotation: [
276                 { type: 'testId', description: 'TC-2381-014' },
277                 { type: 'testId', description: 'TC-2381-017' },
278                 { type: 'testId', description: 'OQG3-2381-AC07' },
279             ],
280         },
281         async ({ pageManager }) => {
282             qase.id(0)
283             qase.title(
284                 'TC-2381-014/017 the Title font drives the event title and section headings'
285             )
286             qase.suite(SUITE)
287             // Product behaviour differs from the sheet (OQG3-2562,
288             // devii, verified 2 Sep 2026). The Title selector changes only the
289             // <h1> event title. The section headings the sheet names — "Date",
290             // "Contact", "Our Social Networks" — follow the *Texts* font
291             // instead. Asserting the documented requirement so this flips to
292             // failing if the product is changed to match, or the sheet is
293             // corrected and this marker removed.
294             test.fail()
295 
296             const step2 = pageManager.step2Page
297             await step2.setEventTitle('Title Font Event')
298             await step2.setEventContactName('Jane Organizer')
299 
300             await step2.selectEventFont('title', TITLE_FONT)
301             await step2.selectEventFont('text', TEXTS_FONT)
302 
303             const fonts = await step2.getPreviewFontFamilies()
304             expect(
305                 new Set(fonts.headings),
306                 'the event title must use the Title font'
307             ).toEqual(new Set([TITLE_FONT]))
308 
309             // The sheet lists the section headings as Title-font text too.
310             const headingText = await step2.locator.frameView
311                 .locator('body')
312                 .evaluate((body: HTMLElement) =>
313                     Array.from(body.querySelectorAll('p,span'))
314                         .filter((el) =>
315                             ['Date', 'Contact', 'Our Social Networks'].includes(
316                                 (el.textContent || '').trim()
317                             )
318                         )
319                         .map((el) =>
320                             (el as HTMLElement).style.fontFamily.replace(
321                                 /["']/g,
322                                 ''
323                             )
324                         )
325                 )
326             expect(
327                 headingText.length,
328                 'the preview must render at least one section heading'
329             ).toBeGreaterThan(0)
330             expect(
331                 new Set(headingText),
332                 'section headings must use the Title font'
333             ).toEqual(new Set([TITLE_FONT]))
334         }
335     )
