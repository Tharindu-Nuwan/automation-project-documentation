1 /**
2  * Jira: OQG3-2390 (Event QR coverage)
3  * Feature: Event Step 2 — multi-date ("Add New Date") rows
4  *
5  * These cases were previously skipped as "FE gap: 'Add New Date' multi-row UI
6  * is not reliably automatable per project rules". That reason is stale: on
7  * devii the control exists as `step2-event-date-add-button`, and each row is
8  * addressable via indexed test-ids (`step2-event-date-{n}-…`, with row 0 the
9  * primary row). They are pure Step 2 UI, so they do not depend on the blocked
10  * Event save (DEV-EVENT-STEP3-422).
11  */

173     /** @covers EVQR-095 @covers OQG3-2423-AC04 */
174     test(
175         'EVQR-095 at most 20 date rows can be added',
176         {
177             tag: ['@Regression', '@Medium'],
178             annotation: [
179                 { type: 'testId', description: 'EVQR-095' },
180                 { type: 'testId', description: 'OQG3-2423-AC04' },
181             ],
182         },
183         async ({ pageManager }) => {
184             qase.id(0)
185             qase.title('EVQR-095 at most 20 date rows can be added')
186             qase.suite(SUITE)
187             test.slow()
188 
189             const step2 = pageManager.step2Page
190             while ((await step2.getEventDateRowCount()) < MAX_DATE_ROWS) {
191                 const before = await step2.getEventDateRowCount()
192                 await step2.locator.eventDateAddButton.click()
193                 const grew = await step2.page
194                     .locator(
195                         '[data-qa-id^="step2-event-date-"][data-qa-id$="-start-date-trigger"]'
196                     )
197                     .nth(before)
198                     .waitFor({ state: 'visible', timeout: 10_000 })
199                     .then(() => true)
200                     .catch(() => false)
201                 if (!grew) {
202                     break
203                 }
204             }
205 
206             // The cap must hold: 20 rows, and adding again must not exceed it.
207             expect(await step2.getEventDateRowCount()).toBe(MAX_DATE_ROWS)
208             await step2.locator.eventDateAddButton.click().catch(() => {})
209             await expect
210                 .poll(() => step2.getEventDateRowCount())
211                 .toBe(MAX_DATE_ROWS)
212         }
213     )
