27 // Test cases covered by this spec: OQG3-2211-TC41, TC26
28 // Help-tour history is not the ordinary footer-Preview Back path.
29 // Confirmed on Pixel 5 / dev: the tour does not open the phone mockup, and
30 // footer-preview-open-button is intercepted by div.absolute.inset-0.z-10
31 // while the modal is open. Do not force-click through that overlay.
32 // Automatable oracle: Back closes the help modal, stays on the current
33 // funnel step, and leaves form data unchanged.
34 test.describe(
35     `${OQG3_2211_SUITE} — help tour`,
36     { tag: ['@High', '@Smoke', '@OQG3-2211-Mobile'] },
37     () => {
38         let userData: { email: string; password: string } | undefined
39 
40         test.beforeEach(async ({ pageManager }, testInfo) => {
41             skipUnlessOqgOrQciChromium(testInfo)
42             userData = await pageManager.registerPage.authorizeUser()
43         })
44 
45         test.afterEach(async ({ apiManager }) => {
46             if (userData) {
47                 await apiManager.userApi.deleteUser([userData.email])
48             }
49         })
50 
51         test(
52             'OQG3-2211-TC41 / TC26: browser Back closes the Step 2 help modal and keeps form data',
53             { tag: ['@High', '@Smoke'] },
54             async ({ page, pageManager }) => {
55                 // Product bug: navbar Help does not push a history entry.
56                 // Back leaves Step 2 → Step 1 instead of closing the modal.
57                 // Reproduced on oqg.dev (Pixel 5). Expected: stay on Step 2,
58                 // help closed, form data unchanged.
59                 test.fail()
60                 const title =
61                     'OQG3-2211-TC41 / TC26: browser Back closes the Step 2 help modal and keeps form data'
62                 annotateOqg32211(title)
63                 await openQrTypeOnStep2(page, pageManager, vcard)
64                 const markerValue = await fillMarker(
65                     pageManager,
66                     vcard,
67                     'HelpTour'
68                 )
69                 const step2Url = page.url()
70 
71                 await pageManager.step2Page.openHelpTour()
72                 await pageManager.step2Page.expectHelpModalOpen()
73                 await pageManager.step2Page.expectPhonePreviewClosed()
74 
75                 await page.goBack()
76 
77                 await pageManager.step2Page.expectHelpModalClosed()
78                 await pageManager.step2Page.expectPhonePreviewClosed()
79                 await pageManager.step2Page.expectFunnelContextPreserved(
80                     2,
81                     pageManager.step2Page.locator.vcardFirstNameInput,
82                     markerValue
83                 )
84                 await expect(page).toHaveURL(step2Url)
85                 await expect(page).toHaveURL(QR_FUNNEL_URL)
86             }
87         )
88 
89         test(
90             'OQG3-2211-TC41: browser Back closes the Step 3 help modal and keeps the user on Step 3',
91             { tag: ['@High', '@Smoke'] },
92             async ({ page, pageManager }) => {
93                 // Product bug: navbar Help does not push a history entry.
94                 // Back leaves Step 3 → Step 2 instead of closing the modal.
95                 // Reproduced on oqg.dev (Pixel 5). Expected: stay on Step 3.
96                 test.fail()
97                 const title =
98                     'OQG3-2211-TC41: browser Back closes the Step 3 help modal and keeps the user on Step 3'
99                 annotateOqg32211(title)
100                 await openQrTypeOnStep2(page, pageManager, vcard)
101                 const markerValue = await goToStep3(
102                     page,
103                     pageManager,
104                     vcard,
105                     'HelpSt3'
106                 )
107                 const step3Url = page.url()
108 
109                 await pageManager.step2Page.openHelpTour()
110                 await pageManager.step2Page.expectHelpModalOpen()
111                 await page.goBack()
112 
113                 await pageManager.step2Page.expectHelpModalClosed()
114                 await pageManager.step2Page.expectPhonePreviewClosed()
115                 await pageManager.step2Page.expectOnStep(3)
116                 await expect(page).toHaveURL(step3Url)
117                 await page.goBack()
118                 await pageManager.step2Page.expectOnStep(2)
119                 await expect(
120                     pageManager.step2Page.locator.vcardFirstNameInput
121                 ).toHaveValue(markerValue)
122                 await pageManager.step2Page.expectNoApplicationError()
123             }
124         )
125     }
