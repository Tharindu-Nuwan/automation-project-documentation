26 /**
27  * OQG3-1996 — "Prevent execution of files disguised as images" (upload security).
28  *
29  * Ticket: https://increast.atlassian.net/browse/OQG3-1996  (fixVersion 2.6.0)
30  *   AC1 — block execution of executable/script files disguised as images.
31  *   AC2 — validate the real MIME type / magic bytes match the extension.
32  * Fix ships on dev (oqg-dev-v2.6.0-9): the content service validates
33  * extension + MIME + magic numbers (PDF also page-level); the FE shows an
34  * invalid-file error. Reported repro path was the Facebook QR background image.
35  *
36  * Coverage is in three layers:
37  *   1. Upload-rejection matrix — disguised / mismatched files must be blocked at
38  *      the control (a blocked file never reaches a published QR — that IS the
39  *      security guarantee, so no full path is needed for them).
40  *   2. Full end-to-end (@E2E) — a legitimate upload must survive the whole
41  *      funnel: preview → create → open the public/scanned QR → the content is
42  *      actually SERVED and (for images) actually RENDERS (not a broken/URL-only
43  *      slide). This is the "verify until the scanned QR" requirement.
44  *   3. Real-file format & media matrix — the actual files added under tests/data.
45  *
46  * Nothing here weakens an assertion: a payload that is ACCEPTED, a script that
47  * fires a dialog, or a served image that does not render is a real finding to
48  * triage per CLAUDE.md §11. Three such findings are open and quarantined in
49  * §7 below — UPL-1996-01 (declared-MIME trust bypass), UPL-1996-02 (the MP3
50  * cover-image control rejects every valid image) and UPL-1996-03 (an encrypted
51  * PDF is stored with no preview and no error) — see
52  * docs/Upload_Security_OQG3-1996_Bugs.md.
53  */
54 
55 type UploadFile = string | UploadAttack

101 /**
102  * Wait until the upload control reaches a definite state — the preview renders
103  * the image, or the field error appears — and report which one happened.
104  *
105  * `frameImageBlock.or(imageSelectionErrorMessage)` cannot be used for this:
106  * the preview lives in the mockup iframe and the error in the editor page, and
107  * `locator.or()` resolves its argument inside the FIRST locator's frame, so the
108  * error would be searched for inside the iframe and never found.
109  */
110 async function waitForDefiniteImageOutcome(
111     pageManager: PageManager
112 ): Promise<'accepted' | 'rejected'> {
113     const preview = pageManager.step2Page.locator.frameImageBlock.first()
114     const error =
115         pageManager.step2Page.locator.imageSelectionErrorMessage.first()
116     let outcome: 'accepted' | 'rejected' | 'pending' = 'pending'
117 
118     await expect
119         .poll(
120             async () => {
121                 if (await preview.isVisible().catch(() => false)) {
122                     outcome = 'accepted'
123                 } else if (await error.isVisible().catch(() => false)) {
124                     outcome = 'rejected'
125                 }
126                 return outcome
127             },
128             {
129                 timeout: 20_000,
130                 message:
131                     'the upload control must reach a definite state — a rendered preview or a visible error — and never hang silently',
132             }
133         )
134         .not.toBe('pending')
135 
136     return outcome
137 }

201 /**
202  * Assert at least one served content image actually rendered — polled, because
203  * the public page loads its images from S3 asynchronously after the body shows.
204  * Accepts a loaded `<img>` (naturalWidth) OR an element with a real
205  * `background-image: url(https…)`. A steady failure here = a genuine "content
206  * does not render on the live QR" finding.
207  */
208 async function expectServedImageRenders(scanned: Page): Promise<void> {
209     await expect
210         .poll(
211             async () =>
212                 scanned.evaluate(() => {
213                     const imgs = Array.from(
214                         document.querySelectorAll('.App img')
215                     ) as HTMLImageElement[]
216                     if (
217                         imgs.some((img) => img.complete && img.naturalWidth > 1)
218                     )
219                         return true
220                     return Array.from(document.querySelectorAll('.App *')).some(
221                         (el) => {
222                             const bg = getComputedStyle(el).backgroundImage
223                             return /url\(["']?https?:/.test(bg)
224                         }
225                     )
226                 }),
227             {
228                 timeout: 25_000,
229                 message:
230                     'a served content image must actually render (not a broken/URL-only slide)',
231             }
232         )
233         .toBe(true)
234 }
