86 async function avatarFacts(avatar: Locator): Promise<AvatarFacts> {
87     return avatar.evaluate((el: HTMLImageElement) => {
88         const b = el.getBoundingClientRect()
89         return {
90             fit: getComputedStyle(el).objectFit,
91             nw: el.naturalWidth,
92             nh: el.naturalHeight,
93             complete: el.complete,
94             boxW: Math.round(b.width),
95             boxH: Math.round(b.height),
96         }
97     })
98 }

136 /**
137  * THE distortion detector (why the old CSS checks miss the bug).
138  *
139  * The backend processes a vCard avatar into a 500x500 square, so on the page the
140  * served `<img>` is already square with `object-fit: cover` — the CSS checks
141  * (cover + square box + naturalWidth>0) are ALL trivially true whether the server
142  * cropped correctly or SQUEEZED the image. The distortion is baked into the
143  * pixels and is invisible to CSS.
144  *
145  * This measures the pixels instead. It fetches the served image (server-side via
146  * `page.request`, so no CORS taint), then in the browser draws — from the SAME
147  * original that was uploaded — both a proportional centre-crop and a full-frame
148  * squeeze, and compares the served image to each:
149  *   - correct crop  → served matches the centre-crop  (mseCorrect << mseSquish)
150  *   - squeezed bug  → served matches the stretch       (mseCorrect >> mseSquish)
151  */
152 async function measureAvatarSqueeze(
153     page: Page,
154     servedSrc: string,
155     originalPath: string
156 ): Promise<{ mseCorrect: number; mseSquish: number }> {
157     const origMime = originalPath.endsWith('.png') ? 'image/png' : 'image/jpeg'
158     const origDataUrl = `data:${origMime};base64,${fs
159         .readFileSync(originalPath)
160         .toString('base64')}`
161     const servedBody = await (await page.request.get(servedSrc)).body()
162     const servedDataUrl = `data:image/jpeg;base64,${servedBody.toString('base64')}`
163 
164     return page.evaluate(
165         async ({ origDataUrl, servedDataUrl }) => {
166             const load = (src: string): Promise<HTMLImageElement> =>
167                 new Promise((res, rej) => {
168                     const im = new Image()
169                     im.onload = () => res(im)
170                     im.onerror = () => rej(new Error('img load failed'))
171                     im.src = src
172                 })
173             const N = 180
174             const draw = (fn: (ctx: CanvasRenderingContext2D) => void) => {
175                 const c = document.createElement('canvas')
176                 c.width = N
177                 c.height = N
178                 const ctx = c.getContext('2d')!
179                 fn(ctx)
180                 return ctx.getImageData(0, 0, N, N).data
181             }
182             const orig = await load(origDataUrl)
183             const served = await load(servedDataUrl)
184             const w = orig.naturalWidth
185             const h = orig.naturalHeight
186             const side = Math.min(w, h)
187             const sx = (w - side) / 2
188             const sy = (h - side) / 2
189             const correct = draw((ctx) =>
190                 ctx.drawImage(orig, sx, sy, side, side, 0, 0, N, N)
191             )
192             const squish = draw((ctx) =>
193                 ctx.drawImage(orig, 0, 0, w, h, 0, 0, N, N)
194             )
195             const srv = draw((ctx) =>
196                 ctx.drawImage(
197                     served,
198                     0,
199                     0,
200                     served.naturalWidth,
201                     served.naturalHeight,
202                     0,
203                     0,
204                     N,
205                     N
206                 )
207             )
208             const mse = (a: Uint8ClampedArray, b: Uint8ClampedArray) => {
209                 let s = 0
210                 for (let i = 0; i < a.length; i += 4) {
211                     const dr = a[i] - b[i]
212                     const dg = a[i + 1] - b[i + 1]
213                     const db = a[i + 2] - b[i + 2]
214                     s += dr * dr + dg * dg + db * db
215                 }
216                 return s / ((a.length / 4) * 3)
217             }
218             return {
219                 mseCorrect: mse(srv, correct),
220                 mseSquish: mse(srv, squish),
221             }
222         },
223         { origDataUrl, servedDataUrl }
224     )
225 }

227 /**
228  * Asserts the served avatar preserved the original aspect (proportional crop),
229  * i.e. it is NOT a full-frame squeeze. Fails when the backend stretched the
230  * image into the square (the OQG3-2242 bug). Use with a NON-square original.
231  */
232 async function expectAvatarNotSqueezed(
233     page: Page,
234     servedSrc: string,
235     originalPath: string
236 ): Promise<void> {
237     const { mseCorrect, mseSquish } = await measureAvatarSqueeze(
238         page,
239         servedSrc,
240         originalPath
241     )
242     expect(
243         mseCorrect,
244         `avatar is SQUEEZED (aspect not preserved): the served image matches a ` +
245             `full-frame stretch (MSE ${mseSquish.toFixed(1)}) far better than a ` +
246             `proportional centre-crop (MSE ${mseCorrect.toFixed(1)}). The backend ` +
247             `must crop to square, not stretch. See docs/vCard_Avatar_OQG3-2242_Bugs.md.`
248     ).toBeLessThan(mseSquish)
249 }
