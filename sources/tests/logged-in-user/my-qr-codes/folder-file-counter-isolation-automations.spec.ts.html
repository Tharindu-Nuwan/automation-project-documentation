19 const SUITE_NAME = 'OQG3-1424 folder file counter isolation'

23 
24     test.beforeEach(async ({ page, pageManager }) => {

32         }

35     const multiFolderDeleteTestTitle =
36         'OQG3-1424-TC43: keeps each counter correct when deleting across multiple folders'
37 
38     test(
39         multiFolderDeleteTestTitle,
40         { tag: '@Medium' },
41         async ({ pageManager }) => {
42             qase.id(0)
43             qase.title(multiFolderDeleteTestTitle)
44             qase.suite(SUITE_NAME)
45 
46             const folderA = `multi-a-${pageManager.myQrCodesPage.fakeString(6)}`
47             const folderB = `multi-b-${pageManager.myQrCodesPage.fakeString(6)}`
48             await createFolderWithWebsiteQrs(pageManager, folderA, 1)
49             await createFolderWithWebsiteQrs(pageManager, folderB, 1)
50 
51             for (const folderName of [folderA, folderB]) {
52                 await pageManager.myQrCodesPage.clickFolder(folderName)
53                 await pageManager.myQrCodesPage.deleteFirstVisibleQrCode()
54                 await pageManager.myQrCodesPage.returnToFolderList()
55             }
56 
57             await pageManager.myQrCodesPage.expectFolderFileCount(folderA, 0)
58             await pageManager.myQrCodesPage.expectFolderFileCount(folderB, 0)
59         }
60     )
