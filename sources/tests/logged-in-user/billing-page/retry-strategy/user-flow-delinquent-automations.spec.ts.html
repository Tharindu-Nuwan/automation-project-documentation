131             await test.step('Update payment method with a valid card via UI', async () => {
132                 await pageManager.paymentPage.navigateToSubscriptionsWithFallback()

134                 await pageManager.paymentPage.makePayment(
135                     stripeVisaValidCard.cardNumber,

142             await test.step('Stripe: charge is FULL price and subscription returns active', async () => {
143                 await apiManager.stripeApi.waitForSubscriptionStatus(
144                     ctx!.subscriptionId,
145                     ['active']
146                 )
147                 const charges = await apiManager.stripeApi.getCustomerCharges(
148                     ctx!.customerId,
149                     10
150                 )
151                 const successful = (
152                     charges.data as Array<{
153                         amount: number
154                         paid: boolean
155                         status: string
156                     }>
157                 ).filter((c) => c.paid && c.status === 'succeeded')
158                 expect(
159                     successful.some((c) => c.amount === ctx!.fullPriceCents),
160                     `Expected a successful FULL-price (${ctx!.fullPriceCents}) charge; got [${successful
161                         .map((c) => c.amount)
162                         .join(', ')}]`
163                 ).toBe(true)
164             })
165         }
