4  * Feature: Stripe custom retry strategy — dunning / cancellation emails

11 import { expect } from '@playwright/test'
12 import { qase } from 'playwright-qase-reporter'
13 import { test } from '../../../../fixtures/baseFixture'
14 import { MailpitApi } from '../../../../api/Mailpit'
15 import {
16     DUNNING_STEP_DAYS,
17     POST_RECOVERY_ADVANCE_DAYS,
18     RETRY_SUITE,
19     USD_FULL_PRICE_PLANS,
20     USD_MONTHLY_LADDER_CENTS,
21     retryScenarioTitle,
22 } from '../../../data/stripeRetryStrategy.data'
23 import {
24     advanceDays,
25     advanceFullRetryLadder,
26     attachValidCardForRecovery,
27     cleanupRetryClock,
28     RetryClockContext,
29     setupRetrySubscription,
30     triggerTestClockIntoDunning,
31     waitForCancelAfterFinalRetry,
32 } from '../../../helpers/stripeRetryStrategy'
33 
34 const S11_TITLE = retryScenarioTitle('S11', 'Dunning and Recovery Email Delivery')
35 const S11B_TITLE = retryScenarioTitle('S11b', 'Cancellation Email After Final Retry')
36 
37 // Test cases covered by this spec: OQG3-1721-S11, S11b
38 test.describe(`${RETRY_SUITE} — dunning emails`, { tag: ['@High'] }, () => {
39     let userData: { email: string; password: string }
40     let ctx: RetryClockContext | undefined
41     let mailpit: MailpitApi
42 
43     test.beforeEach(async ({ page, pageManager }) => {
44         const creds = pageManager.registerPage.createUserData(true)
45         userData = creds
46         await page.goto(`${process.env.BASE_URL}register`)
47         await pageManager.registerPage.signUp(creds.email, creds.password)
48         await expect(pageManager.step1Page.locator.qrCodeTypes).toBeVisible({
49             timeout: 25_000,
50         })
51         ctx = undefined
52         mailpit = new MailpitApi()
53     })
54 
55     test.afterEach(async ({ apiManager }) => {
56         await cleanupRetryClock(apiManager.stripeApi, ctx)
57         if (userData?.email)
58             await apiManager.userApi.deleteUser([userData.email])
59     })
60 
61     test(S11_TITLE, { tag: ['@High'] }, async ({ page, pageManager, apiManager }) => {
62             qase.id(9390)
63             qase.title(S11_TITLE)
64             qase.suite(RETRY_SUITE)
65             test.setTimeout(600_000)
66 
67             const sentBeforeDunning = Date.now() - 1_000
68 
69             ctx = await setupRetrySubscription(
70                 page,
71                 pageManager,
72                 apiManager.stripeApi,
73                 userData.email,
74                 USD_FULL_PRICE_PLANS[0]
75             )
76 
77             await test.step('Dunning reminder email after entering past_due', async () => {
78                 await triggerTestClockIntoDunning(apiManager.stripeApi, ctx!)
79                 await advanceDays(apiManager.stripeApi, ctx!, DUNNING_STEP_DAYS)
80 
81                 const message = await mailpit.waitForEmail(
82                     userData.email,
83                     sentBeforeDunning,
84                     120_000
85                 )
86                 const html = await mailpit.getMessageHtml(message.id)
87                 expect(html.toLowerCase()).toMatch(
88                     /payment|retry|past due|update.*(card|payment)|failed/
89                 )
90             })
91 
92             await test.step('Recovery receipt after discounted retry succeeds', async () => {
93                 const amounts = await advanceFullRetryLadder(
94                     apiManager.stripeApi,
95                     ctx!,
96                     6
97                 )
98                 expect(amounts[5]).toBe(USD_MONTHLY_LADDER_CENTS[5])
99 
100                 const sentBeforeRecovery = Date.now() - 1_000
101                 await attachValidCardForRecovery(apiManager.stripeApi, ctx!)
102                 await advanceDays(apiManager.stripeApi, ctx!, DUNNING_STEP_DAYS)
103                 await apiManager.stripeApi.waitForSubscriptionStatus(
104                     ctx!.subscriptionId,
105                     ['active']
106                 )
107 
108                 const recovery = await mailpit.waitForEmail(
109                     userData.email,
110                     sentBeforeRecovery,
111                     120_000
112                 )
113                 const html = await mailpit.getMessageHtml(recovery.id)
114                 expect(html.toLowerCase()).toMatch(
115                     /receipt|payment|success|thank|invoice|24[.,]?97|50%/
116                 )
117 
118                 await advanceDays(
119                     apiManager.stripeApi,
120                     ctx!,
121                     POST_RECOVERY_ADVANCE_DAYS
122                 )
123             })
124         }
125     )
126 
127     test(S11B_TITLE, { tag: ['@High'] }, async ({ page, pageManager, apiManager }) => {
128             qase.id(9391)
129             qase.title(S11B_TITLE)
130             qase.suite(RETRY_SUITE)
131             test.setTimeout(600_000)
132 
133             const sentBefore = Date.now() - 1_000
134             ctx = await setupRetrySubscription(
135                 page,
136                 pageManager,
137                 apiManager.stripeApi,
138                 userData.email,
139                 USD_FULL_PRICE_PLANS[0]
140             )
141             await advanceFullRetryLadder(apiManager.stripeApi, ctx)
142             await waitForCancelAfterFinalRetry(apiManager.stripeApi, ctx)
143 
144             const message = await mailpit.waitForEmail(
145                 userData.email,
146                 sentBefore,
147                 180_000
148             )
149             const html = await mailpit.getMessageHtml(message.id)
150             expect(html.toLowerCase()).toMatch(
151                 /cancel|expired|subscription.*(end|stopp)|no longer/
152             )
153         }
154     )
155 })
