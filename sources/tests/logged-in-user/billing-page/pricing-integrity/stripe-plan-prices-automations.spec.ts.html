1 import { qase } from 'playwright-qase-reporter'
2 import { test } from '@/fixtures/baseFixture'
3 import {
4     attachPricingMismatchReport,

6 } from '@/tests/helpers/pricing-report.helper'
7 import {
8     COUPON_IDS,
9     expectedCouponAmount,
10     hasOfficialFixedPricing,
11     PLAN_NAMES,
12     PLAN_PRICES,
13     PromoDiscount,
14     STRIPE_PLAN_INTERVAL_MONTHS,
15     STRIPE_RECURRING_PLAN_IDS,
16     StripePlanId,

18 } from '@/tests/data/pricing.data'

20 const SUITE = 'Billing pricing integrity — Stripe dashboard'
21 const PROMO_PLAN_IDS: Array<Exclude<StripePlanId, 4>> = [1, 2, 3]
22 const PROMO_DISCOUNTS: PromoDiscount[] = [70, 90]

24 test.describe(SUITE, { tag: ['@High'] }, () => {
25     for (const planId of STRIPE_RECURRING_PLAN_IDS) {
26         const title = `Plan ${planId} (${PLAN_NAMES[planId]}) has the correct Stripe price in every official fixed-price currency`
27 
28         test(title, { tag: ['@High'] }, async ({ apiManager }, testInfo) => {
29             qase.id(0)
30             qase.title(title)
31             qase.suite(SUITE)
32 
33             const expectedRows = PLAN_PRICES.filter(
34                 (price) =>
35                     price.plan_id === planId &&
36                     hasOfficialFixedPricing(price.code)
37             )
38             const mismatches: PricingMismatch[] = []
39             const currentPrices =
40                 await apiManager.stripeApi.getCurrentPlanPrices(
41                     STRIPE_RECURRING_PLAN_IDS
42                 )
43             const actual = currentPrices.get(planId)
44             if (!actual) {
45                 mismatches.push({
46                     planId,
47                     currency: 'all',
48                     field: 'active price',
49                     expected: 'present',
50                     actual: 'missing',
51                 })
52             } else {
53                 const configurationFields = [
54                     ['active', true, actual.active],
55                     ['interval', 'month', actual.interval],
56                     [
57                         'interval count',
58                         STRIPE_PLAN_INTERVAL_MONTHS[planId],
59                         actual.intervalCount,
60                     ],
61                 ] as const
62                 for (const [
63                     field,
64                     expectedValue,
65                     actualValue,
66                 ] of configurationFields) {
67                     if (Object.is(expectedValue, actualValue)) continue
68                     mismatches.push({
69                         planId,
70                         currency: 'all',
71                         field,
72                         expected: expectedValue,
73                         actual: actualValue,
74                     })
75                 }
76 
77                 const expectedCurrencies = new Set(
78                     expectedRows.map((row) => row.code)
79                 )
80                 for (const expected of expectedRows) {
81                     const expectedAmount = toStripeAmount(
82                         expected.code,
83                         expected.value * STRIPE_PLAN_INTERVAL_MONTHS[planId]
84                     )
85                     const actualAmount = actual.amounts[expected.code]
86                     if (Object.is(expectedAmount, actualAmount)) continue
87                     mismatches.push({
88                         planId,
89                         currency: expected.code,
90                         field: 'Stripe amount',
91                         expected: expectedAmount,
92                         actual: actualAmount,
93                     })
94                 }
95                 for (const currency of Object.keys(actual.amounts).filter(
96                     hasOfficialFixedPricing
97                 )) {
98                     if (expectedCurrencies.has(currency)) continue
99                     mismatches.push({
100                         planId,
101                         currency,
102                         field: 'currency',
103                         expected: 'unsupported or absent',
104                         actual: 'present',
105                     })
106                 }
107             }
108 
109             const report = await attachPricingMismatchReport(
110                 testInfo,
111                 `Stripe price summary — plan ${planId} (${PLAN_NAMES[planId]})`,
112                 mismatches,
113                 expectedRows.length
114             )
115             if (mismatches.length > 0) throw new Error(report)
116         })
117     }
118

131 
132                     const coupon =
133                         await apiManager.stripeApi.getCoupon(couponId)
134                     const expectedRows = PLAN_PRICES.filter(
135                         (price) =>
136                             price.plan_id === planId &&
137                             hasOfficialFixedPricing(price.code)
138                     )
139                     const mismatches: PricingMismatch[] = []
140                     const configurationFields = [
141                         ['valid', true, coupon.valid],
142                         ['duration', 'forever', coupon.duration],
143                         ['percent off', null, coupon.percentOff],
144                     ] as const
145                     for (const [
146                         field,
147                         expectedValue,
148                         actualValue,
149                     ] of configurationFields) {
150                         if (Object.is(expectedValue, actualValue)) continue
151                         mismatches.push({
152                             planId,
153                             currency: 'all',
154                             field,
155                             expected: expectedValue,
156                             actual: actualValue,
157                         })
158                     }
159 
160                     for (const expected of expectedRows) {
161                         const expectedAmount = expectedCouponAmount(
162                             expected,
163                             discount
164                         )
165                         const actualAmount = coupon.amounts[expected.code]
166                         if (Object.is(expectedAmount, actualAmount)) continue
167                         mismatches.push({
168                             planId,
169                             currency: expected.code,
170                             field: `${discount}% Promo`,
171                             expected: expectedAmount,
172                             actual: actualAmount,
173                         })
174                     }
175 
176                     const report = await attachPricingMismatchReport(
177                         testInfo,
178                         `Stripe coupon summary — ${couponId}, plan ${planId} (${PLAN_NAMES[planId]})`,
179                         mismatches,
180                         expectedRows.length
181                     )
182                     if (mismatches.length > 0) throw new Error(report)
183                 }
