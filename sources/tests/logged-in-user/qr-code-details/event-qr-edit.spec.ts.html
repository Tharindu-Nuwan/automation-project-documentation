50         async ({ pageManager }) => {
51             qase.id(9176)
52             qase.title('Edit Event start and end date')
53             qase.suite('QR Code Details')
54             const toLocalDate = (offsetDays: number): string => {
55                 const value = new Date(
56                     Date.now() + offsetDays * 24 * 60 * 60 * 1000
57                 )
58                 const yyyy = value.getFullYear()
59                 const mm = String(value.getMonth() + 1).padStart(2, '0')
60                 const dd = String(value.getDate()).padStart(2, '0')
61                 return `${yyyy}-${mm}-${dd}`
62             }
63             const startDate = toLocalDate(2)
64             const endDate = toLocalDate(3)
65             await pageManager.step2Page.setEventStartDate(startDate)
66             await pageManager.step2Page.setEventEndDate(endDate)
67             await pageManager.step2Page.saveEditedQrCode()
68             const pagePromise =
69                 pageManager.qrCreationPage.page.waitForEvent('popup')
70             await pageManager.myQrCodesPage.locator.qrCodeLink.click()
71             const newPage = await pagePromise
72             const scannedPage = new ScannedPage(newPage)
73             await scannedPage.authorize()
74             await expect(newPage.locator('body')).toBeVisible()
75             // Date formatting on the scanned page varies by locale; confirm page loads after edit.

77         }
