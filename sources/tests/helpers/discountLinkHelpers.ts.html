118 /** Rewrite a discount URL: remove some params, set others (for tampering). */
119 export function craftLink(
120     base: string,
121     remove: string[],
122     set: Record<string, string>
123 ): string {
124     const url = new URL(base)
125     for (const key of remove) url.searchParams.delete(key)
126     for (const [k, v] of Object.entries(set)) url.searchParams.set(k, v)
127     return url.toString()
128 }

130 /**
131  * Binding token in a discount URL.
132  * OQG uses `referral_key`; QCI redirects to `/plan?id=…`.
133  */
134 export function referralKey(url: string): string {
135     const params = new URL(url).searchParams
136     return params.get('referral_key') || params.get('id') || ''
137 }

139 /** Query name that binds the link to a user on this URL. */
140 export function bindKeyParamName(url: string): 'referral_key' | 'id' {
141     const params = new URL(url).searchParams
142     if (params.has('id') && !params.has('referral_key')) return 'id'
143     return 'referral_key'
144 }

146 /** Query name for the 70% promo on this URL. */
147 export function promoParamName(url: string): 'promo_code' | 'promo' {
148     const params = new URL(url).searchParams
149     if (params.has('promo') && !params.has('promo_code')) return 'promo'
150     return 'promo_code'
151 }

153 /** Turn a granted discount URL into a 70% heist (OQG `promo_code` / QCI `promo`). */
154 export function craft70OffHeist(base: string): string {
155     const promoKey = promoParamName(base)
156     return craftLink(base, ['plan_type', 'promo_code', 'promo'], {
157         [promoKey]: '70OFF',
158     })
159 }

174 /**
175  * Rewrite a discount URL onto a locale prefix (`/es/plans-and-prices?…`).
176  * Dev often omits `/en/`; this still inserts the requested locale.
177  */
178 export function withLocalePath(url: string, locale: string): string {
179     const parsed = new URL(url)
180     if (/^\/[a-z]{2}(\/|$)/i.test(parsed.pathname)) {
181         parsed.pathname = parsed.pathname.replace(/^\/[a-z]{2}/i, `/${locale}`)
182     } else {
183         parsed.pathname = `/${locale}${parsed.pathname}`
184     }
185     return parsed.toString()
186 }
