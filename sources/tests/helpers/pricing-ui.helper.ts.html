42 /**
43  * Mirrors PHP `custom_currency_format` for expected UI substrings.
44  * Prefer {@link parseUiAmount} + numeric compare for assertions.
45  */
46 export function formatUiAmount(amount: number): string {
47     if (!Number.isFinite(amount)) {
48         throw new Error(`Cannot format non-finite amount: ${amount}`)
49     }
50     if (amount % 1 !== 0) {
51         return amount.toFixed(2).replace(/\B(?=(\d{3})+(?!\d))/g, ',')
52     }
53     if (amount > 999) {
54         return Math.round(amount)
55             .toString()
56             .replace(/\B(?=(\d{3})+(?!\d))/g, ',')
57     }
58     return String(amount)
59 }

61 /** Extracts the first numeric amount from UI text (`$ 49.95 /mo`, `71.88`, …). */
62 export function parseUiAmount(text: string): number | null {
63     const match = text.replaceAll(',', '').match(/-?\d+(?:\.\d+)?/)
64     if (!match) return null
65     const amount = Number(match[0])
66     return Number.isFinite(amount) ? amount : null
67 }

69 export function amountsMatch(
70     expected: number,
71     actual: number,
72     epsilon = 0.005
73 ): boolean {
74     return Math.abs(expected - actual) <= epsilon
75 }

77 /**
78  * Round money to 2 dp after × / ÷ so expected UI amounts never carry IEEE noise
79  * (e.g. `47.85 / 3` → `15.950000000000001`).
80  */
81 function roundMoney(amount: number): number {
82     return Math.round(amount * 100) / 100
83 }
84 
85 /** Period total ÷ interval months, rounded like PHP `promoMonthlyPrice`. */
86 function promoMonthlyPrice(
87     periodTotal: number,
88     planId: StripePlanId
89 ): number {
90     return roundMoney(periodTotal / STRIPE_PLAN_INTERVAL_MONTHS[planId])
91 }

99 /** Default (non-promo) card monthly rates and checkout period totals. */
100 export function expectedCardAndTotals(
101     currency: string
102 ): ExpectedCardAndTotals {
103     const code = currency.toLowerCase()
104     const cards = {} as Record<BillingPlanName, number>
105     const totals = {} as Record<BillingPlanName, number>
106 
107     for (const plan of BILLING_PLANS) {
108         const planId = BILLING_PLAN_IDS[plan]
109         const row = getPlanPrice(planId, code)
110         cards[plan] = row.value
111         totals[plan] = roundMoney(
112             row.value * STRIPE_PLAN_INTERVAL_MONTHS[planId]
113         )
114     }
115 
116     return { currency: code, cards, totals }
117 }

127 /**
128  * Admin Generate URL expected amounts.
129  * 70% quarterly/annual cards show period-total ÷ 3 / ÷ 12 (PHP `promoMonthlyPrice`).
130  */
131 export function expectedDiscountAmounts(
132     currency: string
133 ): ExpectedDiscountAmounts {
134     const code = currency.toLowerCase()
135     const defaultCards = expectedCardAndTotals(code).cards
136     const promo70Cards = {} as Record<BillingPlanName, number>
137 
138     for (const plan of BILLING_PLANS) {
139         const planId = BILLING_PLAN_IDS[plan]
140         const row = getPlanPrice(planId, code)
141         if (row.price_after_70_discount == null) {
142             throw new Error(
143                 `Official sheet has no 70% price for plan ${planId} (${code})`
144             )
145         }
146         const periodTotal = Number(row.price_after_70_discount)
147         promo70Cards[plan] = promoMonthlyPrice(periodTotal, planId)
148     }
149 
150     return {
151         currency: code,
152         defaultCards,
153         promo70Cards,
154         discounted: getPlanPrice(4, code).value,
155         onetime: getPlanPrice(5, code).value,
156     }
157 }

159 /**
160  * Cancel-retention 90% offer shows a monthly rate (PHP `promoMonthlyPrice` with 90).
161  * Sheet `price_after_90_discount` is the **period total** (e.g. Annually 23.88),
162  * not the /mo figure. Divide by interval months, then round to 2 dp.
163  */
164 export function expectedCancelPromo90MonthlyRate(
165     currency: string,
166     plan: BillingPlanName
167 ): number {
168     const code = currency.toLowerCase()
169     const planId = BILLING_PLAN_IDS[plan]
170     const row = getPlanPrice(planId, code)
171     if (row.price_after_90_discount == null) {
172         throw new Error(
173             `Official sheet has no 90% price for plan ${planId} (${code})`
174         )
175     }
176     return promoMonthlyPrice(
177         Number(row.price_after_90_discount),
178         planId
179     )
180 }

195 /**
196  * Records a mismatch when expected ≠ parsed UI amount.
197  * @returns true when a mismatch was recorded
198  */
199 export function pushAmountMismatch(
200     mismatches: PricingMismatch[],
201     options: AmountMismatchOptions
202 ): boolean {
203     const actual = parseUiAmount(options.actualText)
204     if (actual != null && amountsMatch(options.expected, actual)) return false
205     mismatches.push({
206         planId: options.planId,
207         currency: options.currency,
208         field: options.field,
209         expected: options.expected,
210         actual: actual ?? (options.actualText || 'missing'),
211     })
212     return true
213 }
214 
215 /** Records a mismatch and attaches a full-page PNG when one is found. */
216 export async function recordAmountMismatch(
217     mismatches: PricingMismatch[],
218     options: AmountMismatchOptions,
219     visual?: MismatchVisualCapture
220 ): Promise<boolean> {
221     const mismatched = pushAmountMismatch(mismatches, options)
222     if (!mismatched || !visual) return mismatched
223     const last = mismatches[mismatches.length - 1]
224     await attachPricingMismatchScreenshot(
225         visual.testInfo,
226         visual.page,
227         last
228     )
229     return true
230 }
