11 function display(value: unknown): string {
12     if (value === null) return 'null'
13     if (value === undefined) return 'missing'
14     return String(value).replaceAll('|', '\\|')
15 }

17 const BASE_PLAN_NAMES: Record<number, string> = {
18     1: 'Monthly',
19     2: 'Annually',
20     3: 'Quarterly',
21     4: 'Discounted',
22     5: 'One Time',
23     6: '14-Day Full',
24     7: '14-Day Limited',
25 }

27 function reportPlanName(mismatch: PricingMismatch): string {
28     const baseName =
29         BASE_PLAN_NAMES[mismatch.planId] ?? `Plan ${mismatch.planId}`
30     const field = mismatch.field.toLowerCase()
31 
32     if (field === 'first charge') {
33         return `${baseName} - Charge 01`
34     }
35     if (field === 'second charge') {
36         return `${baseName} - Charge 02`
37     }
38     if (field.includes('90%')) {
39         return `${baseName} - 90% Forever`
40     }
41     if (field.includes('70%')) {
42         return `${baseName} - 70% Forever`
43     }
44     // UI /plan: card monthly rate vs summary Total after selecting the plan
45     if (field.startsWith('card')) {
46         return `${baseName} - Card price`
47     }
48     if (field.includes('checkout total')) {
49         return `${baseName} - Checkout total`
50     }
51     if (field.startsWith('default')) {
52         return `${baseName} - Default card`
53     }
54     if (field.includes('14-day')) {
55         return baseName
56     }
57     return baseName
58 }

60 function reportFieldLabel(mismatch: PricingMismatch): string {
61     const field = mismatch.field.toLowerCase()
62     if (field.startsWith('card')) return 'Card price (/mo)'
63     if (field.includes('checkout total')) return 'Checkout total'
64     if (field.startsWith('default')) return 'Default card'
65     if (field.includes('14-day limited')) return '14-day Limited'
66     if (field.includes('14-day full')) return '14-day Full'
67     if (field.includes('discounted')) return 'Discounted plan'
68     if (field.includes('one-time')) return 'One-time plan'
69     return mismatch.field
70 }

72 function alignedTable(headers: string[], rows: string[][]): string[] {
73     const widths = headers.map((header, index) =>
74         Math.max(header.length, ...rows.map((row) => row[index]?.length ?? 0))
75     )
76     const formatRow = (row: string[]): string =>
77         `| ${row
78             .map((cell, index) => cell.padEnd(widths[index]))
79             .join(' | ')} |`
80 
81     return [
82         formatRow(headers),
83         `| ${widths.map((width) => '-'.repeat(width)).join(' | ')} |`,
84         ...rows.map(formatRow),
85     ]
86 }

88 function screenshotAttachmentName(mismatch: PricingMismatch): string {
89     const slug = [
90         mismatch.currency,
91         mismatch.field,
92         `exp-${mismatch.expected}`,
93         `act-${mismatch.actual}`,
94     ]
95         .join('-')
96         .toLowerCase()
97         .replace(/[^a-z0-9._-]+/g, '-')
98         .replace(/-+/g, '-')
99         .replace(/(^-|-$)/g, '')
100         .slice(0, 140)
101     return `mismatch-${slug}`
102 }

105  * Attaches a full-page PNG of the failing UI so the HTML report shows the

108 export async function attachPricingMismatchScreenshot(
109     testInfo: TestInfo,
110     page: Page,
111     mismatch: PricingMismatch
112 ): Promise<void> {
113     const name = screenshotAttachmentName(mismatch)
114     try {
115         const body = await page.screenshot({ type: 'png', fullPage: true })
116         await testInfo.attach(name, {
117             body,
118             contentType: 'image/png',
119         })
120     } catch (error) {
121         await testInfo.attach(`${name}-capture-failed`, {
122             body: Buffer.from(
123                 `Could not capture mismatch screenshot: ${String(error)}`
124             ),
125             contentType: 'text/plain',
126         })
127     }
128 }

130 /** Builds a compact Markdown audit grouped by plan name and currency. */
131 export function formatPricingMismatchReport(
132     title: string,
133     mismatches: PricingMismatch[],
134     checkedCurrencyCount?: number
135 ): string {
136     if (mismatches.length === 0) {
137         return [
138             `# ${title}`,
139             '',
140             '## ✅ PASSED',
141             '',
142             checkedCurrencyCount == null
143                 ? 'All supported currencies match.'
144                 : `All ${checkedCurrencyCount} supported currencies match.`,
145         ].join('\n')
146     }
147 
148     const currencyCount = new Set(
149         mismatches.map((mismatch) => mismatch.currency)
150     ).size
151     const planCounts = new Map<string, number>()
152     for (const mismatch of mismatches) {
153         const planName = reportPlanName(mismatch)
154         planCounts.set(planName, (planCounts.get(planName) ?? 0) + 1)
155     }
156     const planSummary = [...planCounts.entries()]
157         .map(([planName, count]) => `${planName}: ${count}`)
158         .join(', ')
159     const rows = [...mismatches]
160         .sort(
161             (left, right) =>
162                 reportPlanName(left).localeCompare(reportPlanName(right)) ||
163                 left.currency.localeCompare(right.currency) ||
164                 reportFieldLabel(left).localeCompare(reportFieldLabel(right))
165         )
166         .map((mismatch) => [
167             String(mismatch.planId),
168             reportPlanName(mismatch),
169             mismatch.currency.toUpperCase(),
170             reportFieldLabel(mismatch),
171             display(mismatch.expected),
172             display(mismatch.actual),
173             screenshotAttachmentName(mismatch),
174         ])
175     const table = alignedTable(
176         [
177             'Plan ID',
178             'Plan Name',
179             'Currency',
180             'Field',
181             'Expected',
182             'Actual',
183             'Screenshot',
184         ],
185         rows
186     )
187 
188     return [
189         `# ${title}`,
190         '',
191         '## ❌ FAILED',
192         '',
193         checkedCurrencyCount == null
194             ? `Checked supported currencies: not recorded.`
195             : `Checked supported currencies: ${checkedCurrencyCount}.`,
196         '',
197         `${mismatches.length} mismatches across ${currencyCount} currencies.`,
198         '',
199         `**By plan:** ${planSummary}`,
200         '',
201         'Each row has a matching Playwright attachment (PNG) named in the Screenshot column.',
202         '',
203         ...table,
204     ].join('\n')
205 }

207 /** Adds the full Markdown audit to the Playwright HTML report. */
208 export async function attachPricingMismatchReport(
209     testInfo: TestInfo,
210     title: string,
211     mismatches: PricingMismatch[],
212     checkedCurrencyCount?: number
213 ): Promise<string> {
214     const report = formatPricingMismatchReport(
215         title,
216         mismatches,
217         checkedCurrencyCount
218     )
219     const attachmentName = title
220         .toLowerCase()
221         .replace(/[^a-z0-9]+/g, '-')
222         .replace(/(^-|-$)/g, '')
223     await testInfo.attach(`pricing-summary-${attachmentName}`, {
224         body: Buffer.from(report),
225         contentType: 'text/markdown',
226     })
227     return report
228 }
