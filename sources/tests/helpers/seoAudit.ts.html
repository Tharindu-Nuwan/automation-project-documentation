8 /**
9  * Technical-SEO / HTML-structure / accessibility auditor.
10  *
11  * Extracts the DOM facts once in the browser, then evaluates the rule set in
12  * Node and returns a structured, CI-friendly report. Broken-link checks use
13  * Playwright's request context (per CLAUDE.md §17 — never axios/fetch).
14  *
15  * A critical failure = a real SEO/a11y defect. Per CLAUDE.md §11 these are
16  * product bugs to document and quarantine, not assertions to weaken.
17  */

64 async function collectDomFacts(page: Page): Promise<DomFacts> {
65     return page.evaluate(() => {
66         const q = (sel: string) => Array.from(document.querySelectorAll(sel))
67         const attr = (sel: string, name: string) =>
68             document.querySelector(sel)?.getAttribute(name) ?? null
69         return {
70             title: document.querySelector('title')?.textContent?.trim() ?? null,
71             metaDescription: attr('meta[name="description"]', 'content'),
72             htmlLang: document.documentElement.getAttribute('lang'),
73             viewport: attr('meta[name="viewport"]', 'content'),
74             canonical: attr('link[rel="canonical"]', 'href'),
75             headings: q('h1,h2,h3,h4,h5,h6').map((h) => {
76                 const text = (h.textContent || '').trim()
77                 return {
78                     level: Number(h.tagName.substring(1)),
79                     text: text.slice(0, 130),
80                     length: text.length,
81                 }
82             }),
83             mainCount: q('main').length,
84             divRoleMainCount: q('div[role="main"]').length,
85             headerCount: q('header').length,
86             footerCount: q('footer').length,
87             navCount: q('nav').length,
88             divCount: q('div').length,
89             semanticCount: q('main,nav,header,footer,article,section,aside')
90                 .length,
91             images: q('img').map((im) => ({
92                 altMissing: !im.hasAttribute('alt'),
93                 hasWidth: im.hasAttribute('width'),
94                 hasHeight: im.hasAttribute('height'),
95                 src: (im.getAttribute('src') || '').slice(0, 60),
96             })),
97             anchors: q('a').map((a) => ({
98                 href: a.getAttribute('href') || '',
99                 text: (a.textContent || '').trim().toLowerCase(),
100             })),
101             jsonLd: q('script[type="application/ld+json"]').map(
102                 (s) => s.textContent || ''
103             ),
104             inlineStyleBytes: q('style').map(
105                 (s) => (s.textContent || '').length
106             ),
107             origin: location.origin,
108         }
109     })
110 }

112 /** Audits the page currently loaded in `page`. Assumes navigation + networkidle done. */
113 export async function auditCurrentPage(
114     page: Page,
115     overrides: Partial<SeoAuditConfig> = {}
116 ): Promise<SeoReport> {
117     const cfg: SeoAuditConfig = { ...DEFAULT_SEO_CONFIG, ...overrides }
118     const facts = await collectDomFacts(page)
119     const url = page.url()
120 
121     const criticalFailures: SeoFinding[] = []
122     const warnings: SeoFinding[] = []
123     const failedRules = new Set<string>()
124     const checkedRules = new Set<string>()
125     const ignored = (r: string) => cfg.ignoreRules.includes(r)
126     const check = (r: string) => {
127         if (!ignored(r)) checkedRules.add(r)
128     }
129     const fail = (rule: string, selector: string, message: string) => {
130         if (ignored(rule)) return
131         criticalFailures.push({ rule, selector, message })
132         failedRules.add(rule)
133     }
134     const warn = (rule: string, selector: string, message: string) => {
135         if (ignored(rule)) return
136         warnings.push({ rule, selector, message })
137     }
138 
139     // ---- 1. Heading hierarchy ----
140     check(SEO_RULES.H1_COUNT)
141     const h1s = facts.headings.filter((h) => h.level === 1)
142     if (h1s.length !== 1) {
143         fail(
144             SEO_RULES.H1_COUNT,
145             'h1',
146             `Expected exactly 1 <h1>, found ${h1s.length}` +
147                 (h1s.length > 1
148                     ? `: ${h1s.map((h) => `"${h.text.slice(0, 25)}"`).join(', ')}`
149                     : '')
150         )
151     }
152 
153     check(SEO_RULES.HEADING_ORDER)
154     const seenLevels = new Set<number>()
155     for (const h of facts.headings) {
156         if (h.level > 1 && !seenLevels.has(h.level - 1)) {
157             fail(
158                 SEO_RULES.HEADING_ORDER,
159                 `h${h.level}`,
160                 `Skipped level: <h${h.level}> ("${h.text.slice(0, 25)}") appears before any <h${h.level - 1}>`
161             )
162         }
163         seenLevels.add(h.level)
164     }
165 
166     check(SEO_RULES.HEADING_LENGTH)
167     check(SEO_RULES.HEADING_EMPTY)
168     for (const h of facts.headings) {
169         if (h.length > cfg.headingMaxLength) {
170             fail(
171                 SEO_RULES.HEADING_LENGTH,
172                 `h${h.level}`,
173                 `Heading exceeds ${cfg.headingMaxLength} chars (${h.length}): "${h.text.slice(0, 30)}…"`
174             )
175         }
176         if (h.text.replace(/[\s\u200B-\u200D\uFEFF]+/g, '').length === 0) {
177             fail(SEO_RULES.HEADING_EMPTY, `h${h.level}`, 'Empty heading tag')
178         }
179     }
180 
181     // ---- 2. Meta / document ----
182     check(SEO_RULES.TITLE)
183     const title = facts.title || ''
184     if (!title) {
185         fail(SEO_RULES.TITLE, 'title', 'Missing <title>')
186     } else {
187         if (title.length < cfg.title.min || title.length > cfg.title.max) {
188             fail(
189                 SEO_RULES.TITLE,
190                 'title',
191                 `Title length ${title.length} is outside ${cfg.title.min}-${cfg.title.max}`
192             )
193         }
194         for (const brand of cfg.brandTerms) {
195             const count = title.split(brand).length - 1
196             if (count > 1) {
197                 fail(
198                     SEO_RULES.TITLE,
199                     'title',
200                     `Brand term "${brand}" repeated ${count}x in title (keyword stuffing)`
201                 )
202             }
203         }
204     }
205 
206     check(SEO_RULES.META_DESCRIPTION)
207     const desc = facts.metaDescription
208     if (!desc) {
209         fail(
210             SEO_RULES.META_DESCRIPTION,
211             'meta[name=description]',
212             'Missing meta description'
213         )
214     } else {
215         if (
216             desc.length < cfg.metaDescription.min ||
217             desc.length > cfg.metaDescription.max
218         ) {
219             fail(
220                 SEO_RULES.META_DESCRIPTION,
221                 'meta[name=description]',
222                 `Meta description length ${desc.length} is outside ${cfg.metaDescription.min}-${cfg.metaDescription.max}`
223             )
224         }
225         if (desc.trim() === title.trim()) {
226             fail(
227                 SEO_RULES.META_DESCRIPTION,
228                 'meta[name=description]',
229                 'Meta description is identical to the title'
230             )
231         }
232     }
233 
234     check(SEO_RULES.LANG)
235     const lang = facts.htmlLang
236     if (!lang) {
237         fail(SEO_RULES.LANG, 'html', 'Missing <html lang> attribute')
238     } else if (
239         !cfg.validLangPrefixes.includes(lang.split('-')[0].toLowerCase())
240     ) {
241         fail(SEO_RULES.LANG, 'html', `Unrecognised lang="${lang}"`)
242     }
243 
244     check(SEO_RULES.VIEWPORT)
245     const vp = facts.viewport || ''
246     if (!vp) {
247         fail(SEO_RULES.VIEWPORT, 'meta[name=viewport]', 'Missing viewport meta')
248     } else if (
249         !/width=device-width/.test(vp) ||
250         !/initial-scale=1(\.0)?/.test(vp)
251     ) {
252         fail(
253             SEO_RULES.VIEWPORT,
254             'meta[name=viewport]',
255             `Viewport must contain width=device-width and initial-scale=1.0 (got "${vp}")`
256         )
257     }
258 
259     check(SEO_RULES.CANONICAL)
260     const canonical = facts.canonical
261     if (!canonical) {
262         fail(
263             SEO_RULES.CANONICAL,
264             'link[rel=canonical]',
265             'Missing canonical link'
266         )
267     } else if (!/^https?:\/\/.+/.test(canonical)) {
268         fail(
269             SEO_RULES.CANONICAL,
270             'link[rel=canonical]',
271             `Canonical must be an absolute URL (got "${canonical}")`
272         )
273     }
274 
275     // ---- 3. Semantic HTML ----
276     check(SEO_RULES.MAIN_LANDMARK)
277     if (facts.mainCount !== 1) {
278         fail(
279             SEO_RULES.MAIN_LANDMARK,
280             'main',
281             `Expected exactly 1 <main>, found ${facts.mainCount}`
282         )
283     }
284 
285     check(SEO_RULES.SEMANTIC)
286     if (facts.divRoleMainCount > 0 && facts.mainCount === 0) {
287         fail(
288             SEO_RULES.SEMANTIC,
289             'div[role=main]',
290             'Uses <div role="main"> instead of a <main> element'
291         )
292     }
293     if (facts.semanticCount > 0 && facts.divCount / facts.semanticCount > 30) {
294         warn(
295             SEO_RULES.SEMANTIC,
296             'div',
297             `${facts.divCount} <div>s vs ${facts.semanticCount} semantic elements — consider article/section`
298         )
299     }
300 
301     check(SEO_RULES.HEADER_FOOTER)
302     if (facts.headerCount === 0 && facts.footerCount === 0) {
303         warn(
304             SEO_RULES.HEADER_FOOTER,
305             'header,footer',
306             'No <header> or <footer> landmark found'
307         )
308     }
309 
310     check(SEO_RULES.NAV)
311     if (facts.navCount === 0 && facts.anchors.length > 5) {
312         warn(SEO_RULES.NAV, 'nav', 'Page has links but no <nav> landmark')
313     }
314 
315     // ---- 4. Images ----
316     check(SEO_RULES.IMG_ALT)
317     check(SEO_RULES.IMG_DIMENSIONS)
318     facts.images.forEach((im, i) => {
319         if (im.altMissing) {
320             fail(
321                 SEO_RULES.IMG_ALT,
322                 `img[src="${im.src}"]`,
323                 `Image #${i} missing alt attribute`
324             )
325         }
326         if (!im.hasWidth || !im.hasHeight) {
327             warn(
328                 SEO_RULES.IMG_DIMENSIONS,
329                 `img[src="${im.src}"]`,
330                 `Image #${i} missing width/height (CLS risk)`
331             )
332         }
333     })
334 
335     // ---- 5. Links ----
336     check(SEO_RULES.EMPTY_HREF)
337     check(SEO_RULES.GENERIC_ANCHOR)
338     for (const a of facts.anchors) {
339         const href = a.href.trim()
340         if (href === '#' || /^javascript:\s*void/i.test(href)) {
341             fail(
342                 SEO_RULES.EMPTY_HREF,
343                 `a[href="${href}"]`,
344                 `Non-navigable href "${href}"`
345             )
346         }
347         if (a.text && cfg.genericAnchorTexts.includes(a.text)) {
348             warn(
349                 SEO_RULES.GENERIC_ANCHOR,
350                 'a',
351                 `Non-descriptive anchor text "${a.text}"`
352             )
353         }
354     }
355 
356     check(SEO_RULES.BROKEN_LINKS)
357     const internal = [
358         ...new Set(
359             facts.anchors
360                 .map((a) => a.href.trim())
361                 .filter((h) => h && !/^(#|mailto:|tel:|javascript:)/i.test(h))
362                 .map((h) => {
363                     try {
364                         return new URL(h, facts.origin).href
365                     } catch {
366                         return ''
367                     }
368                 })
369                 .filter((h) => h.startsWith(facts.origin))
370         ),
371     ].slice(0, 25) // cap to keep runtime bounded
372     for (const link of internal) {
373         try {
374             const res = await page.request.head(link, {
375                 timeout: cfg.linkCheckTimeoutMs,
376                 maxRedirects: 5,
377             })
378             const st = res.status()
379             // Only 404/410/5xx are truly broken. 401/403/405/429 mean the URL
380             // exists but rejects an unauthenticated HEAD — reachable, not broken.
381             if (st === 404 || st === 410 || st >= 500) {
382                 fail(
383                     SEO_RULES.BROKEN_LINKS,
384                     `a[href="${link}"]`,
385                     `Internal link returned ${st}`
386                 )
387             } else if (st >= 400) {
388                 warn(
389                     SEO_RULES.BROKEN_LINKS,
390                     `a[href="${link}"]`,
391                     `Internal link returned ${st} (method/auth restricted, not broken)`
392                 )
393             }
394         } catch {
395             // Timeout / network error — surface as a warning, not a hard fail,
396             // since a HEAD may be blocked while the page itself is fine.
397             warn(
398                 SEO_RULES.BROKEN_LINKS,
399                 `a[href="${link}"]`,
400                 'HEAD request failed/timed out'
401             )
402         }
403     }
404 
405     // ---- 6. JSON-LD ----
406     check(SEO_RULES.JSON_LD)
407     if (facts.jsonLd.length === 0) {
408         warn(
409             SEO_RULES.JSON_LD,
410             'script[type=ld+json]',
411             'No JSON-LD structured data present'
412         )
413     } else {
414         facts.jsonLd.forEach((raw, i) => {
415             try {
416                 JSON.parse(raw)
417             } catch {
418                 fail(
419                     SEO_RULES.JSON_LD,
420                     `script[type=ld+json]:nth(${i})`,
421                     'Malformed JSON-LD (not parseable)'
422                 )
423             }
424         })
425     }
426 
427     // ---- 7. Inline style bloat ----
428     check(SEO_RULES.INLINE_STYLE)
429     facts.inlineStyleBytes.forEach((bytes, i) => {
430         if (bytes > cfg.inlineStyleWarnBytes) {
431             warn(
432                 SEO_RULES.INLINE_STYLE,
433                 `style:nth(${i})`,
434                 `Inline <style> is ${(bytes / 1024).toFixed(1)}KB (> ${cfg.inlineStyleWarnBytes / 1024}KB)`
435             )
436         }
437     })
438 
439     const rulesChecked = checkedRules.size
440     return {
441         url,
442         passed: criticalFailures.length === 0,
443         criticalFailures,
444         warnings,
445         summary: {
446             rulesChecked,
447             passed: rulesChecked - failedRules.size,
448             failed: failedRules.size,
449             warnings: warnings.length,
450         },
451     }
452 }

454 /** Navigates to `url`, waits for network idle, then audits it. */
455 export async function auditUrl(
456     page: Page,
457     url: string,
458     overrides: Partial<SeoAuditConfig> = {}
459 ): Promise<SeoReport> {
460     await page.goto(url, { waitUntil: 'domcontentloaded' })
461     await page.waitForLoadState('networkidle').catch(() => {})
462     return auditCurrentPage(page, overrides)
463 }

465 /** Human-readable failure list for assertion messages / CI logs. */
466 export function describeReport(report: SeoReport): string {
467     const s = report.summary
468     return [
469         `SEO audit: ${report.url}`,
470         `rules ${s.passed}/${s.rulesChecked} passed, ${s.failed} failed, ${s.warnings} warnings`,
471         ...report.criticalFailures.map(
472             (f) => `  ✗ [${f.rule}] ${f.selector} — ${f.message}`
473         ),
474     ].join('\n')
475 }
