20 /**
21  * Verifies rendered text remains inside its immediate layout container and does
22  * not create its own horizontal overflow. The assertion is retried while the
23  * preview iframe finishes reflowing.
24  */
25 export async function expectTextContained(text: Locator): Promise<void> {
26     await expect(text).toBeVisible()
27 
28     await expect
29         .poll(
30             async (): Promise<boolean> =>
31                 text.evaluate((element) => {
32                     const container = element.parentElement
33                     if (!container) {
34                         throw new Error('Preview text has no parent container')
35                     }
36 
37                     const elementRect = element.getBoundingClientRect()
38                     const containerRect = container.getBoundingClientRect()
39                     const htmlElement = element as HTMLElement
40 
41                     return (
42                         elementRect.left >= containerRect.left - 1 &&
43                         elementRect.right <= containerRect.right + 1 &&
44                         htmlElement.scrollWidth <= htmlElement.clientWidth + 1
45                     )
46                 }),
47             {
48                 message:
49                     'preview text should finish reflowing inside its container',
50             }
51         )
52         .toBe(true)
53 
54     const metrics = await text.evaluate((element): ContainmentMetrics => {
55         const container = element.parentElement
56         if (!container) throw new Error('Preview text has no parent container')
57         const elementRect = element.getBoundingClientRect()
58         const containerRect = container.getBoundingClientRect()
59         const htmlElement = element as HTMLElement
60         return {
61             elementLeft: elementRect.left,
62             elementRight: elementRect.right,
63             containerLeft: containerRect.left,
64             containerRight: containerRect.right,
65             scrollWidth: htmlElement.scrollWidth,
66             clientWidth: htmlElement.clientWidth,
67         }
68     })
69 
70     expect(metrics.elementLeft).toBeGreaterThanOrEqual(
71         metrics.containerLeft - 1
72     )
73     expect(metrics.elementRight).toBeLessThanOrEqual(metrics.containerRight + 1)
74     expect(metrics.scrollWidth).toBeLessThanOrEqual(metrics.clientWidth + 1)
75 }

77 /** Fills a field, uses the accepted value (including maxlength truncation), and checks its preview. */
78 export async function fillAndAssertContained(
79     input: Locator,
80     preview: Locator,
81     requestedValue: string
82 ): Promise<string> {
83     await input.fill(requestedValue)
84     const acceptedValue = await input.inputValue()
85     expect(acceptedValue.length).toBeGreaterThan(0)
86     await expect(preview).toContainText(acceptedValue)
87     await expectTextContained(preview)
88     return acceptedValue
89 }

100 /** Opens the MP3 content step, retrying the existing selection once if needed. */
101 export async function openMp3Step2(pageManager: PageManager): Promise<void> {
102     await pageManager.step1Page.selectMp3QrType()
103     const titleInput = pageManager.step2Page.locator.mp3FileTitle
104     let reachedStep2 = false
105     try {
106         await titleInput.waitFor({ state: 'visible', timeout: 15_000 })
107         reachedStep2 = true
108     } catch {
109         // Retry the QR-type selection once when the Step 1 transition stalls.
110     }
111     if (!reachedStep2) {
112         await pageManager.step1Page.locator.mp3QrType.click()
113         await titleInput.waitFor({ state: 'visible', timeout: 30_000 })
114     }
115 }

158 /** Verifies that the document has no horizontal viewport overflow. */
159 export async function expectNoPageOverflow(page: Page): Promise<void> {
160     await expect
161         .poll(() =>
162             page.evaluate(
163                 () =>

171 /** Verifies the published title and description match the accepted preview text. */
172 export async function expectPublishedText(
173     page: Page,
174     accepted: AcceptedText
175 ): Promise<{ title: Locator; description: Locator }> {
176     const title = page.locator('h1#mp3_title')
177     const description = page.locator("//h1[@id='mp3_title']/../p").last()
178     await expect(title).toContainText(accepted.title)
179     await expect(description).toContainText(accepted.description)
180     await expectTextContained(title)
181     await expectTextContained(description)
182     await expectNoPageOverflow(page)
183     return { title, description }
184 }
