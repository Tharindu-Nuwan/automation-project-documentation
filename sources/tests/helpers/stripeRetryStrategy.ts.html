1 /**
2  * Shared setup / advance helpers for OQG3-1721 Stripe custom retry strategy.
3  *
4  * Required sequence before every related test:
5  *  1. Register the user (the app creates its Stripe customer)
6  *  2. Attach a Test Clock to that customer and wait until the clock is `ready`
7  *  3. Pay for Monthly | Quarterly | Annually via the billing UI (valid card)
8  *  4. Update the Stripe default payment method to a decline card (test token)
9  *  5. Advance the Test Clock into dunning
10  *
11  * Stripe forbids adding a Test Clock except at clock-create time; since May 2026
12  * `POST /v1/test_helpers/test_clocks` accepts `customer=` to attach an existing
13  * (UI-created) customer. `frozen_time` must not be in the past.
14  */
15

104 /**
105  * Step 2 — Attach a Test Clock immediately after registration, before creating
106  * the subscription. `createTestClock()` waits for `ready`, because Stripe blocks
107  * all customer/subscription modifications while attachment is `advancing`.
108  */
109 export async function attachTestClockToCustomer(
110     stripeApi: StripeAPI,
111     email: string,
112     plan: RetryPlanFixture
113 ): Promise<AttachedClockContext> {
114     const customerId = await stripeApi.waitForCustomerByEmail(email)
115     // Must not be in the past when attaching an existing customer. A small
116     // future offset prevents request latency from making it backdated.
117     const frozenTime = Math.floor(Date.now() / 1000) + 5
118     const clock = await stripeApi.createTestClock(
119         frozenTime,
120         `oqg3-1721-${plan.key}-${Date.now()}`,
121         customerId
122     )
123 
124     return {
125         clockId: clock.id,
126         customerId,
127         frozenTime: clock.frozen_time,
128         priceId: plan.priceId,
129         email,
130     }
131 }

133 /**
134  * Full OQG3-1721 setup used by every related test. Registration happens in the
135  * spec's `beforeEach`, then this runs:
136  *  1. Attach clock + await `ready`
137  *  2. UI plan payment
138  *  3. Update to decline card
139  */
140 export async function setupRetrySubscription(
141     page: Page,
142     pageManager: PageManager,
143     stripeApi: StripeAPI,
144     email: string,
145     plan: RetryPlanFixture
146 ): Promise<RetryClockContext> {
147     const attachedClock = await attachTestClockToCustomer(
148         stripeApi,
149         email,
150         plan
151     )
152 
153     try {
154         await payForPlanSubscription(page, pageManager, plan)
155 
156         const subscriptionId = await stripeApi.waitForSubscriptionId(
157             attachedClock.customerId
158         )
159         const subscription =
160             await stripeApi.retrieveASubscription(subscriptionId)
161         const currentPeriodEnd = Number(subscription.current_period_end)
162 
163         await pageManager.paymentPage.navigateToSubscriptionsWithFallback()
164         await pageManager.billingPage.updatePaymentMethod(
165             stripeDeclinedCard.cardNumber,
166             stripeDeclinedCard.expDate,
167             stripeDeclinedCard.cvc
168         )
169 
170         await page.reload()
171 
172         const multiplier = getPlanMultiplier(plan.key)
173         const fullPriceCents = toMinorUnits(
174             plan.listPrice * multiplier,
175             plan.zeroDecimal
176         )
177         const expectedLadderCents =
178             plan.currency === 'usd' && plan.key === 'monthly'
179                 ? [...USD_MONTHLY_LADDER_CENTS]
180                 : retryLadderAmountsCents(
181                       plan.listPrice,
182                       plan.zeroDecimal,
183                       plan.key
184                   )
185 
186         return {
187             ...attachedClock,
188             subscriptionId,
189             currentPeriodEnd,
190             fullPriceCents,
191             expectedLadderCents,
192         }
193     } catch (error) {
194         // Avoid leaking the attached customer/clock when UI payment setup fails.
195         await stripeApi
196             .deleteTestClock(attachedClock.clockId)
197             .catch(() => undefined)
198         throw error
199     }
200 }

213 /** Advance the Test Clock to an absolute unix timestamp (must be ahead). */
214 export async function advanceClockTo(
215     stripeApi: StripeAPI,
216     ctx: RetryClockContext,
217     targetUnixSeconds: number
218 ): Promise<number> {
219     const target = Math.max(ctx.frozenTime + 60, targetUnixSeconds)
220     const clock = await stripeApi.advanceTestClock(ctx.clockId, target)
221     ctx.frozenTime = clock.frozen_time
222     return ctx.frozenTime
223 }

225 /**
226  * Step 5 — Advance the Test Clock just past the subscription period end so the renewal
227  * invoice fails on the decline card (`past_due`).
228  */
229 export async function triggerTestClockIntoDunning(
230     stripeApi: StripeAPI,
231     ctx: RetryClockContext
232 ): Promise<void> {
233     const target = Math.max(
234         ctx.frozenTime + 3600 * 24,
235         ctx.currentPeriodEnd + 3600 * 24
236     )
237     await advanceClockTo(stripeApi, ctx, target)
238     await stripeApi.waitForSubscriptionStatus(ctx.subscriptionId, [
239         'past_due',
240         'unpaid',
241         'canceled',
242         'cancelled',
243     ])
244 }

246 /**
247  * Reads the open/failed renewal invoice and returns Stripe's retry counters.
248  * Prefer this over guessing schedule offsets — `attempt_count` and
249  * `next_payment_attempt` are the source of truth for the dunning ladder.
250  */
251 export async function getFailedInvoiceRetryState(
252     stripeApi: StripeAPI,
253     customerId: string
254 ): Promise<FailedInvoiceRetryState> {
255     const invoice = await stripeApi.getFailedRenewalInvoice(customerId)
256     if (!invoice) {
257         throw new Error(
258             `No failed/open renewal invoice found for customer ${customerId}`
259         )
260     }
261 
262     return {
263         invoiceId: invoice.id,
264         attemptCount: invoice.attempt_count,
265         nextPaymentAttempt: invoice.next_payment_attempt,
266         amountDue: invoice.amount_due,
267         status: invoice.status,
268     }
269 }

271 /**
272  * Polls until the failed renewal invoice has a scheduled `next_payment_attempt`.
273  * Stripe sets this asynchronously after each failed retry on a Test Clock.
274  */
275 export async function waitForNextPaymentAttempt(
276     stripeApi: StripeAPI,
277     customerId: string,
278     timeoutMs = 90_000
279 ): Promise<FailedInvoiceRetryState> {
280     const deadline = Date.now() + timeoutMs
281     let last: FailedInvoiceRetryState | null = null
282 
283     while (Date.now() < deadline) {
284         try {
285             last = await getFailedInvoiceRetryState(stripeApi, customerId)
286             if (last.nextPaymentAttempt != null) return last
287         } catch {
288             // Invoice may not exist yet right after the first clock advance.
289         }
290         await new Promise((resolve) => setTimeout(resolve, 2_000))
291     }
292 
293     throw new Error(
294         `No next_payment_attempt scheduled within ${timeoutMs}ms` +
295             (last
296                 ? ` (attempt_count=${last.attemptCount}, status=${last.status})`
297                 : '')
298     )
299 }

301 /**
302  * Advance the Test Clock to just after the failed invoice's
303  * `next_payment_attempt`. Returns the invoice retry state after the advance.
304  */
305 export async function advanceToNextPaymentAttempt(
306     stripeApi: StripeAPI,
307     ctx: RetryClockContext
308 ): Promise<FailedInvoiceRetryState> {
309     const state = await waitForNextPaymentAttempt(stripeApi, ctx.customerId)
310     await advanceClockTo(
311         stripeApi,
312         ctx,
313         state.nextPaymentAttempt! + ONE_DAY_SECONDS
314     )
315     return getFailedInvoiceRetryState(stripeApi, ctx.customerId)
316 }

318 /**
319  * Advance the Test Clock just past the subscription's `current_period_end` so
320  * the next billing cycle invoice is generated. Use after recovery when the
321  * dunning invoice is closed and `next_payment_attempt` is no longer available.
322  */
323 export async function advanceToSubscriptionRenewal(
324     stripeApi: StripeAPI,
325     ctx: RetryClockContext
326 ): Promise<number> {
327     const subscription = await stripeApi.retrieveASubscription(
328         ctx.subscriptionId
329     )
330     const periodEnd = Number(subscription.current_period_end)
331     return advanceClockTo(stripeApi, ctx, periodEnd + ONE_DAY_SECONDS)

334 /** Poll until a paid `subscription_cycle` invoice matches `expectedAmountPaid`. */
335 export async function waitForPaidSubscriptionCycleInvoice(
336     stripeApi: StripeAPI,
337     customerId: string,
338     expectedAmountPaid: number,
339     timeoutMs = 90_000
340 ): Promise<{ amount_paid: number; billing_reason: string | null }> {
341     const deadline = Date.now() + timeoutMs
342 
343     while (Date.now() < deadline) {
344         const invoices = await stripeApi.listInvoices(customerId, 10)
345         const paid = invoices.data.find(
346             (invoice) =>
347                 invoice.status === 'paid' &&
348                 invoice.amount_paid === expectedAmountPaid &&
349                 invoice.billing_reason === 'subscription_cycle'
350         )
351         if (paid) return paid
352         await new Promise((resolve) => setTimeout(resolve, 2_000))
353     }
354 
355     throw new Error(
356         `Expected paid subscription_cycle invoice for ${expectedAmountPaid} within ${timeoutMs}ms`
357     )
358 }

360 /**
361  * After the 8-attempt ladder, advance using the failed invoice's
362  * `next_payment_attempt` (when still set) and wait until the subscription
363  * is canceled / unpaid.
364  */
365 export async function waitForCancelAfterFinalRetry(
366     stripeApi: StripeAPI,
367     ctx: RetryClockContext
368 ): Promise<string> {
369     const before = await getFailedInvoiceRetryState(stripeApi, ctx.customerId)
370     expect(
371         before.attemptCount,
372         `Expected at least ${RETRY_ATTEMPT_COUNT} failed attempts before cancel`
373     ).toBeGreaterThanOrEqual(RETRY_ATTEMPT_COUNT)
374 
375     if (before.nextPaymentAttempt != null) {
376         await advanceClockTo(
377             stripeApi,
378             ctx,
379             before.nextPaymentAttempt + 3600 * 24
380         )
381     }
382 
383     // Stripe cancels in the same pass that marks the invoice uncollectible, but
384     // the subscription read can still lag a Test Clock advance by a few minutes.
385     return stripeApi.waitForSubscriptionStatus(
386         ctx.subscriptionId,
387         ['canceled', 'unpaid'],
388         180_000
389     )
390 }

392 /**
393  * Polls the failed invoice until `attempt_count` is at least `minAttempt`,
394  * or until timeout (webhook / clock processing lag).
395  */
396 export async function waitForInvoiceAttemptCount(
397     stripeApi: StripeAPI,
398     customerId: string,
399     minAttempt: number,
400     timeoutMs = 90_000
401 ): Promise<FailedInvoiceRetryState> {
402     const deadline = Date.now() + timeoutMs
403     let last: FailedInvoiceRetryState | null = null
404 
405     while (Date.now() < deadline) {
406         try {
407             last = await getFailedInvoiceRetryState(stripeApi, customerId)
408             if (last.attemptCount >= minAttempt) return last
409         } catch {
410             // Invoice may not exist yet right after the first clock advance.
411         }
412         await new Promise((resolve) => setTimeout(resolve, 2_000))
413     }
414 
415     throw new Error(
416         `Expected invoice attempt_count >= ${minAttempt} within ${timeoutMs}ms` +
417             (last
418                 ? ` (last attempt_count=${last.attemptCount}, next_payment_attempt=${last.nextPaymentAttempt})`
419                 : '')

423 /**
424  * Polls the dunning invoice's charges until at least `expectedCount` attempts
425  * have been recorded, and returns their amounts oldest-first.
426  */
427 export async function waitForInvoiceChargeAmounts(
428     stripeApi: StripeAPI,
429     customerId: string,
430     invoiceId: string,
431     expectedCount: number,
432     timeoutMs = 60_000
433 ): Promise<number[]> {
434     const deadline = Date.now() + timeoutMs
435     let amounts: number[] = []
436 
437     while (Date.now() < deadline) {
438         amounts = await stripeApi.getInvoiceChargeAmounts(customerId, invoiceId)
439         if (amounts.length >= expectedCount) return amounts
440         await new Promise((resolve) => setTimeout(resolve, 2_000))
441     }
442 
443     throw new Error(
444         `Expected ${expectedCount} charge attempts on invoice ${invoiceId} within ` +
445             `${timeoutMs}ms, got ${amounts.length}: [${amounts.join(', ')}]`
446     )
447 }

449 /**
450  * Walks the dunning ladder by reading the failed invoice's `attempt_count` and
451  * `next_payment_attempt`, advancing the Test Clock to each scheduled retry.
452  *
453  * Returns the amount **charged** on each attempt (minor units), oldest first,
454  * taken from the invoice's charges. `amount_due` cannot be used here: the next
455  * tier's discount is applied the moment an attempt fails, so the value read
456  * after attempt #N is what attempt #N+1 will charge.
457  */
458 export async function advanceFullRetryLadder(
459     stripeApi: StripeAPI,
460     ctx: RetryClockContext,
461     attempts: number = RETRY_ATTEMPT_COUNT
462 ): Promise<number[]> {
463     await triggerTestClockIntoDunning(stripeApi, ctx)
464 
465     let invoiceId = ''
466     let lastAttemptCount = 0
467 
468     for (let step = 1; step <= attempts; step++) {
469         const state = await waitForInvoiceAttemptCount(
470             stripeApi,
471             ctx.customerId,
472             Math.max(step, lastAttemptCount + 1)
473         )
474         invoiceId = state.invoiceId
475         lastAttemptCount = state.attemptCount
476 
477         if (lastAttemptCount >= attempts) break
478         if (state.nextPaymentAttempt == null) {
479             // No further retry scheduled — ladder ended early or cancelled.
480             break
481         }
482 
483         await advanceToNextPaymentAttempt(stripeApi, ctx)
484     }
485 
486     const charged = await waitForInvoiceChargeAmounts(
487         stripeApi,
488         ctx.customerId,
489         invoiceId,
490         Math.min(attempts, lastAttemptCount)
491     )
492     return charged.slice(0, attempts)
493 }

495 /** Assert the collected retry amounts match the expected ladder. */
496 export function expectRetryLadderAmounts(
497     actual: number[],
498     expected: number[]
499 ): void {
500     expect(
501         actual.length,
502         `Expected ${expected.length} retry attempt amounts, got ${actual.length}: [${actual.join(', ')}]`
503     ).toBe(expected.length)
504     for (let i = 0; i < expected.length; i++) {
505         expect(actual[i], `Retry #${i + 1} amount mismatch`).toBe(expected[i])
506     }
507 }

509 /**
510  * Swap customer + subscription default PMs to a valid Visa so the next retry
511  * succeeds. UI dunning setup sets the subscription PM to a decline card; both
512  * defaults must be updated for Smart Retry to charge the recovery card.
513  */
514 export async function attachValidCardForRecovery(
515     stripeApi: StripeAPI,
516     ctx: Pick<RetryClockContext, 'customerId' | 'subscriptionId'>
517 ): Promise<string> {
518     const paymentMethodId = await createAndAttachCard(

520         ctx.customerId,
521         stripeValidCardToken
522     )
523     await stripeApi.setSubscriptionDefaultPaymentMethod(
524         ctx.subscriptionId,
525         paymentMethodId
526     )
527     return paymentMethodId
528 }

530 /**
531  * Attach a valid card and advance the Test Clock to the invoice's scheduled
532  * `next_payment_attempt + 1 day`, then wait until the subscription recovers.
533  */
534 export async function recoverOnNextPaymentAttempt(
535     stripeApi: StripeAPI,
536     ctx: RetryClockContext,
537     timeoutMs = 180_000
538 ): Promise<void> {
539     const state = await waitForNextPaymentAttempt(stripeApi, ctx.customerId)
540     await attachValidCardForRecovery(stripeApi, ctx)
541     await advanceClockTo(
542         stripeApi,
543         ctx,
544         state.nextPaymentAttempt! + ONE_DAY_SECONDS
545     )
546     await stripeApi.waitForSubscriptionStatus(
547         ctx.subscriptionId,
548         ['active'],
549         timeoutMs
550     )
551 }
