7  * Local Event stack note:
8  * - Event is available on guest /create?step=1
9  * - /register redirects home on this local setup

13 export async function dismissStep2HelpPopup(page: Page): Promise<void> {
14     const helpPopup = page.getByTestId('help-popup-step-2')
15     for (let i = 0; i < 8; i++) {
16         const isOpen = await helpPopup.isVisible().catch(() => false)
17         if (!isOpen) return
18 
19         const candidates = [
20             page.getByTestId('help-popup-step-2-close-icon-button'),
21             page.getByTestId('help-popup-step-2-close-button'),
22             page.getByTestId('help-popup-step-2-continue-button'),
23             page.getByTestId('help-popup-step-2-close'),
24             page.getByTestId('help-popup-step-2-continue'),
25         ]
26 
27         let clicked = false
28         for (const candidate of candidates) {
29             if (await candidate.isVisible().catch(() => false)) {
30                 await candidate.click({ force: true }).catch(() => {})
31                 clicked = true
32                 break
33             }
34         }
35         if (!clicked) {
36             await page.keyboard.press('Escape')
37         }
38         await page.waitForTimeout(300)
39     }
40 
41     await helpPopup.waitFor({ state: 'hidden', timeout: 5_000 }).catch(() => {})
42 }

44 export async function waitForStep2NextEnabled(page: Page): Promise<void> {
45     const nextButton = page.getByTestId('footer-step2-next-button')
46     await nextButton.waitFor({ state: 'visible', timeout: 30000 })
47     await page.waitForFunction(
48         () => {
49             const btn = document.querySelector(
50                 '[data-qa-id="footer-step2-next-button"]'
51             ) as HTMLButtonElement | null
52             return Boolean(btn && !btn.disabled)
53         },
54         { timeout: 60000 }
55     )

105 /**
106  * Logs in with the local persistent account (LOCAL_USER_LOGIN / _PASSWORD).
107  *
108  * Unlike the guest `/create` funnel, a logged-in session persists the QR to the
109  * dashboard, which is what the edit / scan-page / dashboard suites need.
110  * `authorizeUser()` cannot be used on the local stack (`/register` is disabled),
111  * so these suites log in instead.
112  */
113 export async function loginLocalUser(
114     page: Page,
115     pageManager: PageManager
116 ): Promise<void> {
117     const login = (process.env.LOCAL_USER_LOGIN || '').trim()
118     const password = (process.env.LOCAL_USER_PASSWORD || '').trim()
119     if (!login || !password) {
120         throw new Error(
121             'LOCAL_USER_LOGIN / LOCAL_USER_PASSWORD must be set in .env to run the logged-in Event suites.'
122         )
123     }
124     await page.goto('/login')
125     await pageManager.loginPage.logIn(login, password)
126     await page.waitForURL(/qr-codes/, { timeout: 30000 })
127 }

158 /**
159  * Registers a fresh user, then creates and publishes an Event QR through the
160  * logged-in route.
161  *
162  * Uses `authorizeUser()` rather than the `LOCAL_USER_*` account: that account
163  * belongs to the local Event stack and its login never advances past the email
164  * step on devii. Registration works there, and the logged-in Event flow saves
165  * correctly (the DEV-EVENT-STEP3-422 rejection is guest-funnel only).
166  *
167  * Returns the event title *and* the QR name — they are different fields, and
168  * the dashboard card is keyed on the QR name.
169  */
170 export async function loginCreateAndPublishEventQr(
171     page: Page,
172     pageManager: PageManager,
173     options: {
174         setQrName?: boolean
175         user?: { email: string; password: string }
176     } = {}
177 ): Promise<PublishedEventQr> {
178     // Reuse an existing session when the caller passes one (publishing a
179     // second QR for the same user), otherwise register a fresh user.
180     const user =
181         options.user ?? (await pageManager.registerPage.authorizeUser())
182     // Mirror the entry sequence proven green by event-qr-step3.spec.ts.
183     await page.goto('/qr-codes')
184     await pageManager.menuPage.goToCreateQrCode()
185     await pageManager.step1Page.locator.websiteQrType.waitFor({
186         state: 'visible',
187         timeout: 30_000,
188     })
189     await selectEventCard(page, pageManager)
190     // The help popup only appears the first time a user reaches Step 2, so the
191     // strict close (which waits for it) is tried first and falls back to the
192     // tolerant sweep when publishing a second QR on the same account.
193     await pageManager.step2Page
194         .ensureHelpClosedBeforeUpload()
195         .catch(() => dismissStep2HelpPopup(page))
196     await waitForStep2NextEnabled(page)
197 
198     // Title + start AND end date, All day on — the same combination the
199     // Step 3 suite uses. Omitting the end date leaves Next unable to advance.
200     const stamp = Date.now()
201     const title = `E2E Event ${stamp}`
202     const qrName = `E2E Event QR ${stamp}`
203     await pageManager.step2Page.fillEventRequiredFields(title)
204 
205     // Name the QR explicitly so dashboard lookups can key on it. Pass
206     // `setQrName: false` to leave it blank and exercise the auto-generated name.
207     if (options.setQrName !== false) {
208         await pageManager.step2Page.locator.qrCodeNameTab.click()
209         await pageManager.step2Page.locator.qrCodeNameInput.fill(qrName)
210     }
211 
212     await pageManager.qrCreationPage.clickStep2NextButton()
213     await page.waitForURL('**step=3**', { timeout: 60_000 })
214     await pageManager.qrCreationPage.clickStep3CreateButton()
215     await pageManager.qrCreationPage.closeDownloadModalIfPresent()
216     return { title, qrName: options.setQrName === false ? '' : qrName, user }
217 }

235 /**
236  * Re-enters the Event date/time that edit mode drops.
237  *
238  * OQG3-2533: opening a saved Event QR in edit mode restores the title
239  * but leaves start date, start time and end time empty. Save is then blocked
240  * *silently* — no validation message is shown and no request is sent. Re-typing
241  * the date makes the save fire (`PUT /api/steps/qr/{uid}/content`).
242  *
243  * This is a workaround so edit-mode assertions about *other* fields can run.
244  * The defect itself is asserted by its own `test.fail()` case — do not treat
245  * this helper as evidence the bug is fixed.
246  */
247 export async function restoreEventDatesInEdit(
248     pageManager: PageManager
249 ): Promise<void> {
250     await pageManager.step2Page.setEventStartDate()
251     await pageManager.step2Page.setEventEndDate()
252     await pageManager.step2Page.toggleEventAllDay()
253 }

255 /**
256  * Opens a published QR's public scan page and returns the ScannedPage plus the
257  * URL used.
258  *
259  * The dashboard card's href is `{scan-origin}/p/{uid}`. The uid form is built
260  * here as well because that is what the transactional-email suite uses for other
261  * QR types; the server redirects it to the `/p/` form.
262  *
263  * The password gate is only submitted when it is actually present:
264  * `ScannedPage.authorize()` fills `input#password` unconditionally and would
265  * hang on an unprotected QR.
266  */
267 export async function openEventScanPage(
268     page: Page,
269     pageManager: PageManager,
270     options: { password?: string } = {}
271 ): Promise<{ scanned: ScannedPage; url: string }> {
272     const card = pageManager.myQrCodesPage.locator
273     const href = await card.qrCodeLink
274         .locator('xpath=ancestor::a[1]')
275         .getAttribute('href')
276         .catch(() => null)
277     const uid = await card.qrCodeUidInput.inputValue()
278     const origin = href
279         ? new URL(href, page.url()).origin
280         : new URL(page.url()).origin
281     const url = `${origin.replace(/\/+$/, '')}/${uid}`
282 
283     const scanPage = await page.context().newPage()
284     await scanPage.goto(url, { waitUntil: 'domcontentloaded' })
285     const scanned = new ScannedPage(scanPage)
286 
287     // 3s probe: only authorize when the gate is actually up.
288     const gated = await scanned.locator.passwordInput
289         .waitFor({ state: 'visible', timeout: 3_000 })
290         .then(() => true)
291         .catch(() => false)
292     if (gated) {
293         await scanned.authorize(options.password ?? 'Admin123')
294     }
295     return { scanned, url }
296 }
