5 /** Waits for the onboarding backdrop to finish fading (blocks clicks if left open). */
6 export async function waitForQrBackdropHidden(page: Page): Promise<void> {
7     await page
8         .locator('.qr-backdrop')
9         .first()
10         .waitFor({ state: 'hidden', timeout: 5_000 })
11         .catch(() => {})
12 }

14 /**
15  * Logged-in Email QR Step 2 entry — mirrors Event QR `openEventStep2`:

17  */
18 export async function openEmailQrStep2(
19     page: Page,
20     pageManager: PageManager

22     await page.goto('/qr?step=1')
