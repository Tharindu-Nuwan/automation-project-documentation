22 function viewportDisplayName(
23     id: string,
24     width: number,
25     height: number
26 ): string {
27     const category = id.startsWith('desktop_')
28         ? 'Desktop'
29         : id.startsWith('tablet_')
30           ? 'Tablet'
31           : 'Mobile'
32     return `${category} ${width}×${height}`
33 }

35 export const CREATE_STEP1_PATH = '/create?step=1'
36 export type VisualQaScreen = 'create-step-1' | 'generate'
37 
38 export function loadAllLanguages(): VisualQaLanguage[] {
39     return CREATE_PAGE_VISUAL_LANGUAGES.map((lang) => ({
40         id: lang.code,
41         name: lang.name,
42         enabled: true,
43         defaultInUi: lang.code === 'en',
44     }))
45 }
46 
47 export function loadAllViewports(): VisualQaViewport[] {
48     return SCREEN_VIEWPORTS.map((vp) => ({
49         id: vp.label,
50         name: viewportDisplayName(vp.label, vp.width, vp.height),
51         width: vp.width,
52         height: vp.height,
53         enabled: true,
54         defaultInUi: false,
55     }))
56 }
57 
58 function parseCsvEnv(name: string): string[] | null {
59     const raw = process.env[name]?.trim()
60     if (!raw) return null
61     return raw
62         .split(',')
63         .map((entry) => entry.trim())
64         .filter(Boolean)
65 }
66 
67 export function loadLanguagesForRun(): VisualQaLanguage[] {
68     const filter = parseCsvEnv('VISUAL_LANG')
69     return loadAllLanguages().filter((lang) => {
70         if (!lang.enabled) return false
71         if (filter && !filter.includes(lang.id)) return false
72         return true
73     })
74 }
75 
76 export function loadViewportsForRun(): VisualQaViewport[] {
77     const filter = parseCsvEnv('VISUAL_VIEWPORT')
78     return loadAllViewports().filter((vp) => {
79         if (!vp.enabled) return false
80         if (filter && !filter.includes(vp.id)) return false
81         return true
82     })
83 }
