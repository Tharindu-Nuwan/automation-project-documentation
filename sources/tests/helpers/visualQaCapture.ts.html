14 export async function expectNoHorizontalOverflowForViewport(
15     page: Page,
16     viewport: VisualQaViewport
17 ): Promise<void> {
18     if (!viewport.id.startsWith('desktop_')) return
19     const overflow = await page.evaluate(
20         () =>
21             document.documentElement.scrollWidth >
22             document.documentElement.clientWidth
23     )
24     expect(
25         overflow,
26         'Horizontal overflow detected — layout breaks at this viewport'
27     ).toBe(false)
28 }

30 function readPngDimensions(
31     filePath: string
32 ): { width: number; height: number } | null {
33     const buf = fs.readFileSync(filePath)
34     if (buf.length < 24 || buf.toString('ascii', 1, 4) !== 'PNG') {
35         return null
36     }
37     return { width: buf.readUInt32BE(16), height: buf.readUInt32BE(20) }
38 }

40 function baselineMatchesClip(filePath: string, clip: CaptureClip): boolean {
41     const size = readPngDimensions(filePath)
42     if (!size) return false
43     return size.width === clip.width && size.height === clip.height
44 }

46 function isUpdatingSnapshots(testInfo: TestInfo): boolean {
47     const mode = testInfo.config.updateSnapshots
48     return mode === 'all' || mode === 'changed'
49 }

51 export async function saveOrCompareLandingPageVisual(
52     page: Page,
53     pageManager: PageManager,
54     screen: VisualQaScreen,
55     lang: VisualQaLanguage,
56     viewport: VisualQaViewport,
57     testInfo: TestInfo
58 ): Promise<void> {
59     const clip = await pageManager.visualLandingPage.resolveCaptureClip(
60         screen,
61         viewport
62     )
63     // Firefox needs fullPage only when the capture extends below the viewport.
64     const fullPage = clip.height > viewport.height
65     const snapshotName = [
66         screen,
67         testInfo.project.name,
68         visualQaSnapshotName(lang, viewport),
69     ]
70     const filePath = testInfo.snapshotPath(...snapshotName)
71     fs.mkdirSync(path.dirname(filePath), { recursive: true })
72     const screenshotOptions = { clip, fullPage }
73 
74     if (isUpdatingSnapshots(testInfo)) {
75         await expect(page).toHaveScreenshot(snapshotName, screenshotOptions)
76         return
77     }
78 
79     const hasMatchingBaseline =
80         fs.existsSync(filePath) && baselineMatchesClip(filePath, clip)
81 
82     if (!hasMatchingBaseline) {
83         await page.screenshot({
84             path: filePath,
85             type: 'png',
86             ...screenshotOptions,
87         })
88         return
89     }
90 
91     await expect(page).toHaveScreenshot(snapshotName, screenshotOptions)
92 }
