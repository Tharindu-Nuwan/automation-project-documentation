52 export function isSupportedPricingCurrency(currency: string): boolean {
53     return !UNSUPPORTED_PRICING_CURRENCIES.has(currency.toLowerCase())
54 }

56 /** Currencies with one canonical numeric price in the official client sheet. */
57 export function hasOfficialFixedPricing(currency: string): boolean {
58     return (

60         !VARIABLE_PRICE_CURRENCIES.includes(currency.toLowerCase())

82 const ZERO_DECIMAL_CURRENCIES = new Set(['clp', 'jpy', 'krw', 'vnd'])
83 
84 /** Converts a JSON/DB major-unit amount to Stripe's integer minor units. */
85 export function toStripeAmount(currency: string, amount: number): number {
86     const multiplier = ZERO_DECIMAL_CURRENCIES.has(currency.toLowerCase())
87         ? 1
88         : 100
89     return Math.round(amount * multiplier)
90 }

92 /** Finds the expected row and fails clearly if the pricing source is incomplete. */
93 export function getPlanPrice(planId: PlanId, currency: string): PlanPrice {
94     const price = PLAN_PRICES.find(
95         (entry) => entry.plan_id === planId && entry.code === currency
96     )
97     if (!price) {
98         throw new Error(
99             `Official Stripe price sheet has no plan ${planId} price for ${currency}`
100         )
101     }
102     return price
103 }
104 
105 /** Expected Stripe Promo: full charge minus chargeable promo price. */
106 export function expectedCouponAmount(
107     price: PlanPrice,
108     discount: PromoDiscount
109 ): number {
110     if (price.plan_id === 4 || price.plan_id > 4) {
111         throw new Error(
112             `Plan ${price.plan_id} does not use Stripe promo coupons`
113         )
114     }
115 
116     const discountedValue =
117         discount === 70
118             ? price.price_after_70_discount
119             : price.price_after_90_discount
120     if (discountedValue == null) {
121         throw new Error(
122             `Plan ${price.plan_id} has no ${discount}% chargeable price for ${price.code}`
123         )
124     }
125 
126     const intervalMonths =
127         STRIPE_PLAN_INTERVAL_MONTHS[price.plan_id as StripePlanId]
128     const fullAmount = toStripeAmount(price.code, price.value * intervalMonths)
129     const chargeableAmount = toStripeAmount(price.code, Number(discountedValue))
130     return fullAmount - chargeableAmount
131 }
