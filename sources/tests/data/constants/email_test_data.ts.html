94 /**
95  * Origin (protocol + host, no path) of a product `baseURL`.
96  * Must not throw: an invalid URL would prevent Playwright from discovering
97  * email specs (they disappear from --ui / --list).
98  */
99 function originFromBaseURL(baseURL: string): string {
100     const raw = (baseURL || DEFAULT_EMAIL_LINK_ORIGIN).trim()
101     const withProtocol = /^https?:\/\//i.test(raw) ? raw : `http://${raw}`
102     try {
103         return new URL(withProtocol).origin
104     } catch {
105         return DEFAULT_EMAIL_LINK_ORIGIN
106     }
107 }

109 function productBaseURL(product: Product): string {
110     if (product === 'qci') return process.env.QCI_BASE_URL || ''
111     if (product === 'qga') return process.env.QGA_BASE_URL || ''
112     return process.env.OQG_BASE_URL || ''
113 }

119 function resolveEmailLinkOrigin(product: Product = 'oqg'): string {
120     return originFromBaseURL(

427 /** Returns the names of secret patterns that match `body`. */
428 export function findSensitiveDataLeaks(body: string): string[] {
429     return emailSensitiveDataPatterns
430         .filter(({ pattern }) =>
431             new RegExp(pattern.source, pattern.flags).test(body)
432         )
433         .map(({ name }) => name)
434 }
