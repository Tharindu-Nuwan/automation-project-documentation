23 function parseAmount(value: string | undefined): number | null {
24     const normalized = value?.trim().replaceAll(',', '')
25     if (!normalized) return null
26     const amount = Number(normalized)
27     return Number.isFinite(amount) ? amount : null
28 }

30 function requiredAmount(
31     row: string[],
32     index: number,
33     currency: string,
34     label: string
35 ): number {
36     const amount = parseAmount(row[index])
37     if (amount == null) {
38         throw new Error(
39             `Official Stripe price sheet has no numeric ${label} for ${currency}`
40         )
41     }
42     return amount
43 }

45 function asPrice(value: number | null): string | null {

49 /** 2 dp after ÷ so sheet fallbacks never store float noise as plan value. */
50 function roundMoney(amount: number): number {

139 /**
140  * Parses the second, ISO-code-based table from the client-maintained sheet.
141  * Duplicate country rows sharing USD/EUR are accepted only when prices agree.
142  */
143 export function parseOfficialStripePriceSheet(
144     csv = readFileSync(CSV_PATH, 'utf8')
145 ): OfficialStripePriceSheet {
146     const rows = parse(csv, {
147         bom: true,
148         relax_column_count: true,
149         skip_empty_lines: false,
150     }) as string[][]
151     const headerIndex = rows.findIndex(
152         (row) =>
153             row[0]?.trim() === 'Country' &&
154             row[1]?.trim() === 'Country abreviation'
155     )
156     if (headerIndex < 0) {
157         throw new Error(
158             'Official Stripe price sheet second table header was not found'
159         )
160     }
161 
162     const pricesByCurrency = new Map<string, OfficialPlanPrice[]>()
163     const variablePriceCurrencies = new Set<string>()
164     for (const row of rows.slice(headerIndex + 1)) {
165         const currency = row[3]?.trim().toLowerCase()
166         if (!/^[a-z]{3}$/.test(currency)) continue
167         if (parseAmount(row[4]) == null) {
168             variablePriceCurrencies.add(currency)
169             continue
170         }
171 
172         const prices = mapCurrencyRow(row)
173         const existing = pricesByCurrency.get(currency)
174         if (
175             existing &&
176             comparablePriceValues(existing) !== comparablePriceValues(prices)
177         ) {
178             throw new Error(
179                 `Official Stripe price sheet has conflicting duplicate rows for ${currency.toUpperCase()}`
180             )
181         }
182         pricesByCurrency.set(currency, prices)
183     }
184 
185     if (pricesByCurrency.size === 0) {
186         throw new Error(
187             'Official Stripe price sheet contains no numeric prices'
188         )
189     }
190 
191     return {
192         planPrices: [...pricesByCurrency.values()].flat(),
193         variablePriceCurrencies: [...variablePriceCurrencies].sort(),
194     }
195 }
