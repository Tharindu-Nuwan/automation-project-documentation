92 export type MaliciousUploadKind =
93     | 'svg-as-png'
94     | 'html-as-image'

99 /**
100  * In-memory malicious file for `locator.setInputFiles(...)` — no fixture file
101  * needed. Covers MIME/extension mismatch, empty file, path-traversal names,
102  * and an oversized payload.
103  */
104 export function maliciousUpload(kind: MaliciousUploadKind): {
105     name: string
106     mimeType: string
107     buffer: Buffer
108 } {
109     switch (kind) {
110         case 'svg-as-png':
111             return {
112                 name: 'logo.png',
113                 mimeType: 'image/png',
114                 buffer: Buffer.from(
115                     '<svg xmlns="http://www.w3.org/2000/svg" onload="alert(1)"><script>alert(1)</script></svg>'
116                 ),
117             }
118         case 'html-as-image':
119             return {
120                 name: 'photo.jpg',
121                 mimeType: 'image/jpeg',
122                 buffer: Buffer.from(
123                     '<html><body><script>alert(document.domain)</script></body></html>'
124                 ),
125             }
126         case 'zero-byte':
127             return {
128                 name: 'empty.png',
129                 mimeType: 'image/png',
130                 buffer: Buffer.alloc(0),
131             }
132         case 'path-traversal':
133             return {
134                 name: '../../../etc/passwd.png',
135                 mimeType: 'image/png',
136                 buffer: Buffer.from('not-a-real-image'),
137             }
138         case 'oversized-buffer':
139             // ~12 MB of zeros — above typical image size limits.
140             return {
141                 name: 'huge.png',
142                 mimeType: 'image/png',
143                 buffer: Buffer.alloc(12 * 1024 * 1024),
144             }
145     }
