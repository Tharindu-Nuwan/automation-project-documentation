Bundled source snapshot. Credential-like literals are omitted. Line numbers correspond to the audited repository file; this page is for reading and is not executable test code.

1 /**
2  * Tunable configuration for the technical-SEO / HTML / a11y auditor
3  * (tests/helpers/seoAudit.ts). Kept as data so thresholds and brand terms can
4  * be adjusted in one place without touching the engine.
5  */
6 
7 export interface SeoAuditConfig {
8     /** Brand terms that must not appear in <title> more than once. */
9     brandTerms: string[]
10     /** Accepted BCP-47 language prefixes for the <html lang> attribute. */
11     validLangPrefixes: string[]
12     /** Anchor inner-texts considered non-descriptive. */
13     genericAnchorTexts: string[]
14     title: { min: number; max: number }
15     metaDescription: { min: number; max: number }
16     /** Max characters allowed in any single heading (anti-stuffing). */
17     headingMaxLength: number
18     /** Warn when a single inline <style> block exceeds this many bytes. */
19     inlineStyleWarnBytes: number
20     /** HEAD-request timeout for internal broken-link checks (ms). */
21     linkCheckTimeoutMs: number
22     /** Rule ids to skip entirely, e.g. ['H1', 'CANONICAL']. */
23     ignoreRules: string[]
24 }
