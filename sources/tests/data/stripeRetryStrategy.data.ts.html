1 /**
2  * OQG3-1721 — Stripe custom retry (dunning) ladder test data.
3  *
4  * Ladder (full-price plans only) — amount **charged** per failed attempt:
5  *   Attempts #1–#5 → full plan price
6  *   Attempt #6 → 50% off
7  *   Attempt #7 → 70% off
8  *   Attempt #8 → 90% off
9  *   After #8 fails → invoice uncollectible, subscription cancelled
10  *
11  * The backend applies the next tier's discount as soon as an attempt fails, so
12  * `invoice.amount_due` read *after* attempt #N is what attempt #N+1 will charge
13  * — see {@link amountDueAfterAttempt}. Per-attempt charged amounts must come
14  * from the invoice's charges, not from `amount_due`.
15  *
16  * Other plans/currencies scale from the published list prices in
17  * `default-pricing-test-data.ts` via {@link retryLadderAmountsCents}.
18  */

119 /**
120  * Converts a major-unit list price to Stripe minor units.
121  * Zero-decimal currencies (JPY) stay as whole numbers.
122  */
123 export function toMinorUnits(listPrice: number, zeroDecimal = false): number {
124     if (zeroDecimal) return Math.round(listPrice)
125     return Math.round(listPrice * 100)
126 }
127 
128 /**
129  * Discounted charge for retry attempts #6/#7/#8.
130  *
131  * Stripe rounds the discount up, so the remaining charge is floored. Verified on
132  * dev for USD monthly ($49.95): 2497 / 1498 / 499 — i.e. $24.97, not $24.98.
133  */
134 export function discountedChargeCents(
135     fullCents: number,
136     percentOff: 50 | 70 | 90
137 ): number {
138     return Math.floor((fullCents * (100 - percentOff)) / 100)
139 }
140 
141 export function getPlanMultiplier(key?: FullPricePlanKey): number {
142     if (key === 'quarterly') return 3
143     if (key === 'annually') return 12
144     return 1
145 }
146 
147 /** Expected charge amounts (minor units) for attempts #1..#8. */
148 export function retryLadderAmountsCents(
149     listPrice: number,
150     zeroDecimal = false,
151     key?: FullPricePlanKey
152 ): number[] {
153     const multiplier = getPlanMultiplier(key)
154     const full = toMinorUnits(listPrice * multiplier, zeroDecimal)
155     return [
156         full,
157         full,
158         full,
159         full,
160         full,
161         discountedChargeCents(full, 50),
162         discountedChargeCents(full, 70),
163         discountedChargeCents(full, 90),
164     ]
165 }

175 /**
176  * `invoice.amount_due` as observed once attempt `attempt` has failed. The next
177  * tier's discount is applied immediately, so this is the amount attempt
178  * `attempt + 1` will charge; it stays at the #8 amount once the ladder ends.
179  */
180 export function amountDueAfterAttempt(
181     ladder: number[],
182     attempt: number
183 ): number {
184     return ladder[Math.min(attempt, ladder.length - 1)]
185 }
