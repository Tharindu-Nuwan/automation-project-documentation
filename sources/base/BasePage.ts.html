4 type NavigationOptions = Parameters<Page['goto']>[1]

6 /**
7  * Browser-level connection failures that mean "the machine lost the network",
8  * not "the app is broken". Only these are retried — HTTP errors, timeouts on a
9  * reachable host and assertion failures still fail the test on first attempt.
10  */
11 const TRANSIENT_NETWORK_ERRORS = [
12     'ERR_INTERNET_DISCONNECTED',
13     'ERR_NETWORK_CHANGED',
14     'ERR_NAME_NOT_RESOLVED',
15     'ERR_CONNECTION_RESET',
16     'ERR_CONNECTION_CLOSED',
17     'ERR_CONNECTION_ABORTED',
18     'ERR_CONNECTION_FAILED',
19     'ERR_EMPTY_RESPONSE',
20     'ERR_SOCKET_NOT_CONNECTED',
21     'ERR_ADDRESS_UNREACHABLE',
22     'ERR_PROXY_CONNECTION_FAILED',
23 ] as const

25 const NAVIGATION_ATTEMPTS = 3
26 const NAVIGATION_RETRY_DELAY_MS = 2_000

28 export class BasePage {

33     constructor(page: Page) {

37     }

39     async openPage(url: string, options?: NavigationOptions): Promise<void> {
40         await this.retryOnNetworkDrop(() => this.page.goto(url, options))
41     }
42 
43     /**
44      * Runs a navigation step, retrying only when the browser reports a lost
45      * connection (e.g. `ERR_INTERNET_DISCONNECTED`). Backs off linearly so a
46      * brief Wi-Fi/VPN drop does not fail an otherwise healthy test.
47      */
48     protected async retryOnNetworkDrop<T>(
49         navigate: () => Promise<T>
50     ): Promise<T> {
51         for (let attempt = 1; ; attempt++) {
52             try {
53                 return await navigate()
54             } catch (error) {
55                 const isLastAttempt = attempt >= NAVIGATION_ATTEMPTS
56                 if (isLastAttempt || !this.isNetworkDrop(error)) throw error
57                 await this.wait(NAVIGATION_RETRY_DELAY_MS * attempt)
58             }
59         }
60     }
61 
62     private isNetworkDrop(error: unknown): boolean {
63         const message = error instanceof Error ? error.message : String(error)
64         return TRANSIENT_NETWORK_ERRORS.some((code) => message.includes(code))
65     }
66 
67     async isInvisible(locator: Locator): Promise<boolean> {
68         await this.expect(locator).not.toBeVisible()
69         return true
70     }

72     /**
73      * Scrolls so the element sits near the middle of the viewport — clearer in
74      * Playwright UI mode and trace filmstrips than `scrollIntoViewIfNeeded()`,
75      * which only nudges when the node is off-screen.
76      */
77     protected async scrollIntoViewCenter(locator: Locator): Promise<void> {
78         await locator.evaluate((element) =>
79             element.scrollIntoView({
80                 block: 'center',
81                 inline: 'nearest',
82                 behavior: 'instant',
83             })
84         )
85     }

87     /**
88      * Aligns the element just below the sticky app header — best for long Step 2
89      * forms where `center` still leaves fields under the fold.
90      */
91     protected async scrollIntoViewBelowHeader(
92         locator: Locator,
93         headerOffsetPx = 96
94     ): Promise<void> {
95         await locator.evaluate((element, offset) => {

99                 behavior: 'instant',
100             })
101             window.scrollBy(0, -offset)
102         }, headerOffsetPx)
103     }
