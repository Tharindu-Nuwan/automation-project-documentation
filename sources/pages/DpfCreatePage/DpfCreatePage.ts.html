17 abstract class SharedDpfCreatePage extends BasePage {
18     protected abstract readonly locators: DpfCreatePageLocators
19     protected readonly scrollToFxLocations: boolean = true
20 
21     abstract getExpectedFxLocations(): Promise<Locator[]>

23     async open(options: OpenOptions = {}) {
24         const { fx, rawFx, locale, params = {} } = options
25         const searchParams = new URLSearchParams({
26             step: '1',
27             qr_onboarding: 'active_dpf',
28         })
29 
30         if (fx !== undefined) {
31             searchParams.set('fx', fx)
32         }
33 
34         for (const [name, value] of Object.entries(params)) {
35             searchParams.set(name, value)
36         }
37 
38         let path = '/create'
39         if (locale) {
40             path = `/${locale}/create`
41         }
42 
43         let url = `${path}?${searchParams.toString()}`
44         if (rawFx !== undefined) {
45             url += `&fx=${rawFx}`
46         }
47 
48         await this.page.goto(url)
49     }
50 
51     async getFxLocationTexts() {
52         const locations = await this.getExpectedFxLocations()
53 
54         return Promise.all(
55             locations.map(async (location) =>
56                 (await location.innerText()).replace(/\s+/g, ' ').trim()
57             )
58         )
59     }
60 
61     async expectFxValueInAllLocations(fxValue: string) {
62         const locations = await this.getExpectedFxLocations()
63 
64         await this.expectFxValueAtLocations(locations, fxValue)
65     }
66 
67     private async expectFxValueAtLocations(
68         locations: Locator[],
69         fxValue: string
70     ) {
71         for (const location of locations) {
72             if (this.scrollToFxLocations) {
73                 await location.scrollIntoViewIfNeeded()
74                 await this.expect(location).toBeVisible()
75             }
76 
77             await this.expect(location).toContainText(fxValue, {
78                 ignoreCase: true,
79             })
80         }
81     }
82 
83     async expectFxValueRenderedAsPlainText(fxValue: string) {
84         await this.expectFxValueInAllLocations(fxValue)
85         await this.expect(this.locators.fxHtmlInjectionMarker).toHaveCount(0)
86     }
87 
88     async expectFxScriptPayloadIsInert(fxValue: string) {
89         await this.expectFxValueInAllLocations(fxValue)
90         await this.expect(this.locators.fxScriptInjectionMarker).toHaveCount(0)
91     }

161     private async activate(
162         control: Locator,
163         stateAttribute: 'aria-selected' | 'aria-expanded'
164     ) {
165         await control.scrollIntoViewIfNeeded()
166 
167         if ((await control.getAttribute(stateAttribute)) !== 'true') {
