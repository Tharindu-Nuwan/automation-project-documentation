81     /**
82      * Asserts rendered text glyphs stay inside the phone preview frame.
83      * Uses Range client rects (not the element box) so CSS transform:scale
84      * on preview titles does not produce false positives.
85      */
86     async expectTextWithinPhoneFrame(
87         textLocator: Locator,
88         type: FitYourNeedsQrType = this.selectedType
89     ): Promise<void> {
90         await textLocator.waitFor({ state: 'visible' })
91         const frame = this.locator.phoneFrame(type)
92         await frame.waitFor({ state: 'visible' })
93 
94         const result = await textLocator.evaluate((el, frameQaPrefix) => {
95             const preview = el.closest(
96                 `[data-qa-id^="${frameQaPrefix}"][data-qa-id$="-preview"]`
97             )
98             if (!preview) {
99                 return { ok: false, reason: 'preview ancestor missing' }
100             }
101             const wrappers = Array.from(
102                 preview.querySelectorAll('.content-wrapper')
103             ) as HTMLElement[]
104             const frameEl = wrappers.find((w) => {
105                 const s = window.getComputedStyle(w)
106                 return (
107                     s.display !== 'none' &&
108                     s.visibility !== 'hidden' &&
109                     s.opacity !== '0' &&
110                     w.getBoundingClientRect().width > 0
111                 )
112             })
113             if (!frameEl) {
114                 return { ok: false, reason: 'visible phone frame missing' }
115             }
116 
117             const frameRect = frameEl.getBoundingClientRect()
118             const range = document.createRange()
119             range.selectNodeContents(el)
120             const rects = Array.from(range.getClientRects()).filter(
121                 (r) => r.width > 0 && r.height > 0
122             )
123             if (rects.length === 0) {
124                 return { ok: false, reason: 'no glyph rects' }
125             }
126 
127             // Allow small sub-pixel / anti-alias bleed past the rounded frame.
128             const tol = 6
129             const offenders: string[] = []
130             for (const r of rects) {
131                 if (
132                     r.left < frameRect.left - tol ||
133                     r.right > frameRect.right + tol ||
134                     r.top < frameRect.top - tol ||
135                     r.bottom > frameRect.bottom + tol
136                 ) {
137                     offenders.push(
138                         `glyph(${Math.round(r.left)},${Math.round(r.top)},${Math.round(r.right)},${Math.round(r.bottom)}) vs frame(${Math.round(frameRect.left)},${Math.round(frameRect.top)},${Math.round(frameRect.right)},${Math.round(frameRect.bottom)})`
139                     )
140                 }
141             }
142             return {
143                 ok: offenders.length === 0,
144                 reason:
145                     offenders.length === 0
146                         ? 'ok'
147                         : offenders.slice(0, 3).join(' | '),
148             }
149         }, 'fit-your-needs-')
150 
151         this.expect(
152             result.ok,
153             `text outside phone frame for type=${type}: ${result.reason}`
154         ).toBeTruthy()
155     }

157     /**
158      * Asserts text is not ellipsized / horizontally self-clipped.
159      * Ancestor overflow:hidden is ignored — preview frames intentionally clip
160      * scaled content; use expectTextWithinPhoneFrame for boundary checks.
161      */
162     async expectTextFullyVisible(textLocator: Locator): Promise<void> {
163         await textLocator.waitFor({ state: 'visible' })
164 
165         const result = await textLocator.evaluate((el) => {
166             const style = window.getComputedStyle(el)
167             const clippedBySelf =
168                 (style.overflowX === 'hidden' ||
169                     style.textOverflow === 'ellipsis') &&
170                 el.scrollWidth > el.clientWidth + 1
171             if (clippedBySelf) {
172                 return {
173                     ok: false,
174                     reason: `self-clipped scrollWidth=${el.scrollWidth} clientWidth=${el.clientWidth}`,
175                 }
176             }
177 
178             // Visible glyph area must be non-empty.
179             const range = document.createRange()
180             range.selectNodeContents(el)
181             const rects = Array.from(range.getClientRects()).filter(
182                 (r) => r.width > 0 && r.height > 0
183             )
184             if (rects.length === 0) {
185                 return { ok: false, reason: 'no visible glyph rects' }
186             }
187             return { ok: true, reason: 'ok' }
188         })
189 
190         this.expect(result.ok, result.reason).toBeTruthy()
191     }

193     /**
194      * Asserts the target text renders on a single visual line.
195      * Counts unique glyph Y positions — not raw getClientRects length —
196      * because inline children (e.g. `.text-nowrap` spans) create multiple
197      * fragment rects on the same line.
198      */
199     async expectSingleLine(textLocator: Locator): Promise<void> {
200         await textLocator.waitFor({ state: 'visible' })
201 
202         const lineCount = await textLocator.evaluate((el) => {
203             const range = document.createRange()
204             range.selectNodeContents(el)
205             const tops = new Set<number>()
206             for (const rect of Array.from(range.getClientRects())) {
207                 if (rect.width <= 0 || rect.height <= 0) continue
208                 // Round to absorb sub-pixel / anti-alias jitter on the same line.
209                 tops.add(Math.round(rect.top))
210             }
211             return tops.size
212         })
213 
214         this.expect(
215             lineCount,
216             `expected single line, got ${lineCount}`
217         ).toBeLessThanOrEqual(2)
218     }

235     /**
236      * Asserts opening quotation marks share a client-rect line with following text.
237      */
238     async expectOpeningQuotesOnSameLine(textLocator: Locator): Promise<void> {
239         await textLocator.waitFor({ state: 'visible' })
240 
241         const result = await textLocator.evaluate(
242             (el, openQuotes: string[]) => {
243                 const text = el.textContent ?? ''
244                 const openIdx = [...text].findIndex((ch) =>
245                     openQuotes.includes(ch)
246                 )
247                 if (openIdx < 0) {
248                     return { ok: true, reason: 'no opening quote' }
249                 }
250 
251                 const walker = document.createTreeWalker(
252                     el,
253                     NodeFilter.SHOW_TEXT
254                 )
255                 let remaining = openIdx
256                 let quoteNode: Text | null = null
257                 let quoteOffset = 0
258                 while (walker.nextNode()) {
259                     const node = walker.currentNode as Text
260                     const len = node.data.length
261                     if (remaining < len) {
262                         quoteNode = node
263                         quoteOffset = remaining
264                         break
265                     }
266                     remaining -= len
267                 }
268                 if (!quoteNode) {
269                     return { ok: false, reason: 'quote node not found' }
270                 }
271 
272                 const quoteRange = document.createRange()
273                 quoteRange.setStart(quoteNode, quoteOffset)
274                 quoteRange.setEnd(quoteNode, quoteOffset + 1)
275                 const quoteRect = quoteRange.getBoundingClientRect()
276 
277                 let nextOffset = quoteOffset + 1
278                 let nextNode: Text | null = quoteNode
279                 while (nextNode) {
280                     while (
281                         nextOffset < nextNode.data.length &&
282                         /\s/.test(nextNode.data[nextOffset])
283                     ) {
284                         nextOffset++
285                     }
286                     if (nextOffset < nextNode.data.length) break
287                     const next = walker.nextNode() as Text | null
288                     nextNode = next
289                     nextOffset = 0
290                     if (!nextNode) break
291                 }
292                 if (!nextNode || nextOffset >= nextNode.data.length) {
293                     return { ok: false, reason: 'no text after opening quote' }
294                 }
295 
296                 const textRange = document.createRange()
297                 textRange.setStart(nextNode, nextOffset)
298                 textRange.setEnd(nextNode, nextOffset + 1)
299                 const textRect = textRange.getBoundingClientRect()
300 
301                 const sameLine = Math.abs(quoteRect.top - textRect.top) < 2
302                 return {
303                     ok: sameLine,
304                     reason: sameLine
305                         ? 'ok'
306                         : `quote y=${quoteRect.top} text y=${textRect.top}`,
307                 }
308             },
309             OPEN_QUOTES
310         )
311 
312         this.expect(result.ok, result.reason).toBeTruthy()
313     }

315     /**
316      * Asserts closing quotation marks are present when the text opens a quote.
317      */
318     async expectClosingQuotesPresent(textLocator: Locator): Promise<void> {
319         await textLocator.waitFor({ state: 'visible' })
320         const text = ((await textLocator.textContent()) ?? '').trim()
321 
322         const hasOpen = OPEN_QUOTES.some((q) => text.includes(q))
323         if (!hasOpen) {
324             const asciiCount = (text.match(/"/g) ?? []).length
325             this.expect(
326                 asciiCount % 2,
327                 `unbalanced ASCII quotes in: ${text}`
328             ).toBe(0)
329             return
330         }
331 
332         const hasClose = CLOSE_QUOTES.some((q) => text.includes(q))
333         const asciiCount = (text.match(/"/g) ?? []).length
334         const balanced =
335             hasClose ||
336             asciiCount >= 2 ||
337             (text.includes('«') && text.includes('»'))
338 
339         this.expect(balanced, `missing closing quote in: ${text}`).toBeTruthy()
340     }

453     /**
454      * Asserts vCard name text has sufficient contrast against the dark backdrop.
455      */
456     async expectVcardTextContrastSufficient(): Promise<void> {
457         await this.selectQrType('vcard')
458         const name = this.locator.vcardName()
459         await name.waitFor({ state: 'visible' })
460 
461         const contrast = await name.evaluate((el) => {
462             const parse = (c: string): [number, number, number] | null => {
463                 const m = c.match(/rgba?\((\d+),\s*(\d+),\s*(\d+)/)
464                 if (!m) return null
465                 return [Number(m[1]), Number(m[2]), Number(m[3])]
466             }
467             const luminance = ([r, g, b]: [number, number, number]) => {
468                 const toLin = (v: number) => {
469                     const s = v / 255
470                     return s <= 0.03928
471                         ? s / 12.92
472                         : Math.pow((s + 0.055) / 1.055, 2.4)
473                 }
474                 const R = toLin(r)
475                 const G = toLin(g)
476                 const B = toLin(b)
477                 return 0.2126 * R + 0.7152 * G + 0.0722 * B
478             }
479 
480             const color = parse(window.getComputedStyle(el).color)
481             const bg = parse('rgb(9, 58, 50)')
482             if (!color || !bg) return 0
483             const L1 = luminance(color)
484             const L2 = luminance(bg)
485             const lighter = Math.max(L1, L2)
486             const darker = Math.min(L1, L2)
487             return (lighter + 0.05) / (darker + 0.05)
488         })
489 
490         this.expect(contrast).toBeGreaterThanOrEqual(3)
491     }
