214     /**
215      * Returns accessible names of currently visible icon options.
216      */
217     async getVisibleIconOptionNames(): Promise<string[]> {
218         const items = this.locator.iconMenuItems()
219         await this.expect(items.first()).toBeVisible()
220         const count = await items.count()
221         const names: string[] = []
222         for (let i = 0; i < count; i++) {
223             const item = items.nth(i)
224             const qaId = (await item.getAttribute('data-qa-id')) ?? ''
225             if (qaId.endsWith('-icon-button')) continue
226             const label =
227                 (await item.getAttribute('aria-label')) ||
228                 (await item.getAttribute('title')) ||
229                 ((await item.innerText()) || '').trim()
230             if (label && label !== 'Icon') names.push(label)
231         }
232         return names
233     }

238     async selectIconByName(name: string, cardIndex: number = 0): Promise<void> {
239         const trigger = this.locator.iconButton(cardIndex)

242         await this.openIconDropdown(cardIndex)

244         await this.expect(option).toBeAttached()
245         // The icon list is taller than the editor pane; Playwright's click

247         await option.evaluate((element) => (element as HTMLElement).click())

253     /**
254      * Stable icon identity (viewBox + first path). SVG mask ids change on
255      * remount and cannot be used for persistence comparisons.

257     async selectedIconSignature(index: number = 0): Promise<string> {
258         const svg = this.locator.iconButton(index).locator('svg').first()
259         const viewBox = (await svg.getAttribute('viewBox')) ?? ''
260         const path = (await svg.locator('path').first().getAttribute('d')) ?? ''

264     /**
265      * Drags the body textarea resize handle. Returns heights and CSS resize;
266      * callers assert a size change or record a Playwright grip limitation.
267      */
268     async resizeBodyTextarea(index: number = 0): Promise<{
269         before: number
270         after: number
271         cssResize: string
272     }> {
273         const textarea = this.locator.bodyTextarea(index)
274         await textarea.scrollIntoViewIfNeeded()
275         const cssResize = await textarea.evaluate(
276             (element) => getComputedStyle(element).resize
277         )
278         const beforeBox = await textarea.boundingBox()
279         this.expect(beforeBox).not.toBeNull()
280         const startX = beforeBox!.x + beforeBox!.width - 1
281         const startY = beforeBox!.y + beforeBox!.height - 1
282         await this.page.mouse.move(startX, startY)
283         await this.page.mouse.down()
284         await this.page.mouse.move(startX + 30, startY + 80, { steps: 20 })
285         await this.page.mouse.up()
286         const afterBox = await textarea.boundingBox()
287         this.expect(afterBox).not.toBeNull()
288         return {
289             before: beforeBox!.height,
290             after: afterBox!.height,
291             cssResize,
292         }
293     }

317     /**
318      * Asserts a visible focus indicator (outline or box-shadow) on a control.
319      */
320     async expectVisibleFocusIndicator(target: Locator): Promise<void> {
321         await this.expect(target).toBeFocused()
322         const hasIndicator = await target.evaluate((element) => {
323             const style = getComputedStyle(element)
324             const outlineWidth = Number.parseFloat(style.outlineWidth || '0')
325             const outlineStyle = style.outlineStyle
326             const boxShadow = style.boxShadow
327             const ring =
328                 outlineWidth > 0 &&
329                 outlineStyle !== 'none' &&
330                 outlineStyle !== ''
331             const shadow = Boolean(boxShadow && boxShadow !== 'none')
332             return ring || shadow
333         })
334         this.expect(
335             hasIndicator,
336             'Focused control has no visible outline or box-shadow'
337         ).toBe(true)
338     }
