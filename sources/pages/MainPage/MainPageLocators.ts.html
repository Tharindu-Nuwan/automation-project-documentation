22 const CARD_TEST_ID = (typeId: string, product: Product): string => {
23     const slug =
24         product === 'qci' && typeId === 'website'
25             ? 'url'
26             : (product === 'qci' || product === 'qga') && typeId === 'apps'
27               ? 'app'
28               : typeId
29 
30     return {
31         oqg: `home-fit-your-needs-${slug}-card`,
32         qci: `home-qr-type-${slug}-card`,
33         qga: `home-qr-type-${slug}-card-button`,
34         qgc: `qgc-qr-type-${slug === 'event' ? 'events' : slug}-tab`,
35     }[product]
36 }

152     /**
153      * Active preview for the selected homepage QR type.
154      * OQG uses dedicated preview test-ids; QCI/QGA use the phone preview markers.
155      */
156     fitYourNeedsPreview(typeId: string): Locator {
157         if (this.product === 'oqg') {
158             return this.pageRef.getByTestId(`fit-your-needs-${typeId}-preview`)
159         }
160 
161         const key = typeId === 'apps' ? 'app' : typeId
162         const marker =
163             this.product === 'qga' && typeId === 'whatsapp'
164                 ? '.whatsapp-messages'
165                 : getStep1QrTypeByKey(key)!.previewMarker
166 
167         // QCI/QGA render mobile + desktop preview trees; prefer the visible node.
168         return this.pageRef
169             .locator(
170                 '.phone-preview-container, .mobile-preview-container, .phone-mockup, [data-qa-id="phone-mockup-container"]'
171             )
172             .locator(marker)
173             .filter({ visible: true })
174             .first()
175     }
