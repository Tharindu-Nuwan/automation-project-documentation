23     /**
24      * Types into a Stripe Payment Element field via focused keyboard input.
25      * fill()/pressSequentially frequently no-op on #payment-numberInput on oqg.dev.
26      *
27      * force:true — BillingAddressForm's country label (aria-hidden) can sit over
28      * card fields and report "subtree intercepts pointer events".
29      *
30      * Card number only — expiry/CVC must use pressSequentially on the field
31      * locator (see makePayment). force + page.keyboard.type can land in the
32      * previously focused field, leaving expiry/CVC empty with red borders.
33      */
34     private async typeIntoStripeField(
35         field: Locator,
36         value: string
37     ): Promise<void> {
38         await field.click({ force: true })
39         await this.page.keyboard.press('ControlOrMeta+A')
40         await this.page.keyboard.press('Backspace')
41         await this.page.keyboard.type(value, { delay: 25 })
42     }

82     /**
83      * Waits until Stripe's Payment Element iframe is interactable.
84      *
85      * On `/pay` the iframe stays 0×0 / hidden until billing info is submitted.
86      * Waiting on `.filter({ visible: true })` then reports "element(s) not found"
87      * instead of a billing-step timeout — finish that step first.
88      */
89     private async waitForPaymentIframe(timeout = 30_000): Promise<void> {
90         if (await this.isOnPaymentThankYou()) {
91             return
92         }
93         if (await this.locator.iframePayment.isVisible().catch(() => false)) {
94             return
95         }
96         if (await this.isBillingInfoStepVisible()) {
97             await this.fillBillingDetailsPaymentFrame()
98             if (await this.isOnPaymentThankYou()) {
99                 return
100             }
101             if (
102                 await this.locator.iframePayment.isVisible().catch(() => false)
103             ) {
104                 return
105             }
106         }
107 
108         const mountedIframe = this.locator.paymentElement
109             .locator('iframe')
110             .first()
111         try {
112             await this.expect(mountedIframe).toBeAttached({ timeout })
113             await this.expect(this.locator.iframePayment).toBeVisible({
114                 timeout,
115             })
116         } catch (error) {
117             const message =
118                 error instanceof Error ? error.message : String(error)
119             throw new Error(
120                 `Stripe payment iframe was not visible at ${this.page.url()}. ` +
121                     `Checkout may still be on the plan picker or billing step, ` +
122                     `or Stripe failed to mount. ${message}`
123             )
124         }
125     }

127     async makePayment(
128         cardNumber: string = stripeVisaValidCard.cardNumber,
129         expDate: string = stripeVisaValidCard.expDate,
130         cvc: string = stripeVisaValidCard.cvc
131     ): Promise<void> {
132         const timeout = 30_000
133         const digits = cardNumber.replace(/\s/g, '')
134         const expiryDigits = expDate.replace(/\D/g, '')
135 
136         // DPF can confirm + navigate as soon as the Payment Element is complete;
137         // a second makePayment() (from clickOnSubmitPaymentButton) must not hang
138         // waiting for CVC on a torn-down iframe.
139         if (await this.isOnPaymentThankYou()) {
140             return
141         }
142 
143         await this.waitForPaymentIframe(timeout)
144 
145         // US/USD Payment Element often defaults to Cash App Pay / Link.
146         // Card number/expiry/CVC only mount after Card is selected.
147         const cardTab = this.locator.cardPaymentMethodTab
148         if (await cardTab.isVisible().catch(() => false)) {
149             await cardTab.click()
150         }
151 
152         await this.selectCardPaymentMethodIfPresent()
153 
154         // DPF / Payment Element BillingAddressForm can overlay card fields —
155         // fill country/zip first so expiry/cvc clicks are not intercepted.
156         if (
157             await this.locator.paymentCountryDropdown
158                 .isVisible()
159                 .catch(() => false)
160         ) {
161             await this.selectCountryAndZipInPaymentFrame()
162         }
163 
164         if (await this.isOnPaymentThankYou()) {
165             return
166         }
167 
168         const card = this.locator.cardNumber
169         if (!(await card.isVisible().catch(() => false))) {
170             if (await this.isOnPaymentThankYou()) {
171                 return
172             }
173             await this.expect(card).toBeVisible({ timeout })
174         }
175 
176         // First focus after plan select often fails to accept the card number —
177         // retry until Stripe shows the typed digits.
178         for (let attempt = 0; attempt < 4; attempt++) {
179             if (await this.isOnPaymentThankYou()) {
180                 return
181             }
182             const already = await card.inputValue().catch(() => '')
183             if (already.replace(/\s/g, '').includes(digits.slice(0, 4))) {
184                 break
185             }
186             await this.typeIntoStripeField(card, digits)
187             const value = await card.inputValue().catch(() => '')
188             if (value.replace(/\s/g, '').includes(digits.slice(0, 4))) {
189                 break
190             }
191             await this.page.waitForTimeout(400)
192         }
193 
194         // Prefer placeholder locators — expiry/CVC sit side-by-side. Country label
195         // often intercepts a normal click; force+pressSequentially keeps focus correct
196         // (unlike force+keyboard.type which can type into the previously focused field).
197         // Stripe localises the expiry placeholder (EN: "MM / YY", ES: "MM / AA", …).
198         // Fall back to id/name selectors when placeholders are absent (filled state).
199         const expiry = this.locator.framePayment
200             .getByPlaceholder(/MM\s*\/\s*(YY|AA|JJ|MM)/i)
201             .or(this.locator.expiryDateInput)
202             .first()
203         if (!(await expiry.isVisible().catch(() => false))) {
204             if (await this.isOnPaymentThankYou()) {
205                 return
206             }
207             await this.expect(expiry).toBeVisible({ timeout })
208         }
209         for (let attempt = 0; attempt < 4; attempt++) {
210             if (await this.isOnPaymentThankYou()) {
211                 return
212             }
213             const current = await expiry.inputValue().catch(() => '')
214             if (current.replace(/\D/g, '').length >= 4) {
215                 break
216             }
217             await expiry.click({ force: true })
218             await expiry.press('ControlOrMeta+A')
219             await expiry.press('Backspace')
220             await expiry.pressSequentially(expiryDigits, { delay: 30 })
221             const value = await expiry.inputValue().catch(() => '')
222             if (value.replace(/\D/g, '').length >= 4) {
223                 break
224             }
225             await this.page.waitForTimeout(400)
226         }
227 
228         const cvcInput = this.locator.framePayment
229             .getByPlaceholder(/CVC|CVV|CVC\/CVV/i)
230             .or(this.locator.cvcCodeInput)
231             .first()
232         if (!(await cvcInput.isVisible().catch(() => false))) {
233             if (await this.isOnPaymentThankYou()) {
234                 return
235             }
236             await this.expect(cvcInput).toBeVisible({ timeout })
237         }
238         for (let attempt = 0; attempt < 4; attempt++) {
239             if (await this.isOnPaymentThankYou()) {
240                 return
241             }
242             const current = await cvcInput.inputValue().catch(() => '')
243             if (current.length >= cvc.length) {
244                 break
245             }
246             await cvcInput.click({ force: true })
247             await cvcInput.press('ControlOrMeta+A')
248             await cvcInput.press('Backspace')
249             await cvcInput.pressSequentially(cvc, { delay: 30 })
250             const value = await cvcInput.inputValue().catch(() => '')
251             if (value.length >= cvc.length) {
252                 break
253             }
254             await this.page.waitForTimeout(400)
255         }
256 
257         if (await this.isOnPaymentThankYou()) {
258             return
259         }
260 
261         if (await this.locator.paymentZipInput.isVisible().catch(() => false)) {
262             await this.locator.paymentZipInput.click({ force: true })
263             await this.locator.paymentZipInput.fill(paymentData.zip)
264         }
265 
266         await this.acceptSubscriptionTermsIfPresent()
267     }

378     /**
379      * Refills the default Visa test card and submits payment.
380      *
381      * Race (seen on DPF /pay → thank-you / DownloadModal): payment can succeed and
382      * navigate while Playwright is still refilling fields or waiting for Submit.
383      * Submit also disables as soon as Stripe accepts the click — force-click and
384      * treat an already-visible success UI as done.
385      */
386     async clickOnSubmitPaymentButton(): Promise<void> {
387         if (await this.isOnPaymentThankYou()) {
388             return
389         }
390 
391         // Refill before submit — Stripe Element can drop card digits after first pass.
392         await this.makePayment()
393         if (await this.isOnPaymentThankYou()) {
394             return
395         }
396 
397         await this.page.waitForTimeout(300)
398         await this.acceptCheckoutConsentIfPresent()
399 
400         if (await this.isOnPaymentThankYou()) {
401             return
402         }
403 
404         const submit = this.locator.submitPaymentButton
405         try {
406             // DPF may auto-confirm and land on DownloadModal while #submit stays disabled.
407             await expect
408                 .poll(
409                     async () => {
410                         if (await this.isOnPaymentThankYou()) {
411                             return true
412                         }
413                         return submit.isEnabled().catch(() => false)
414                     },
415                     { timeout: 15_000 }
416                 )
417                 .toBe(true)
418 
419             if (await this.isOnPaymentThankYou()) {
420                 return
421             }
422 
423             // force: skip re-checking enabled after the click starts (button disables
424             // immediately while payment processes / navigates).
425             await submit.click({ force: true })
426 
427             await expect
428                 .poll(async () => this.isOnPaymentThankYou(), {
429                     timeout: 60_000,
430                 })
431                 .toBe(true)
432         } catch (error) {
433             if (await this.isOnPaymentThankYou()) {
434                 return
435             }
436             const message =
437                 error instanceof Error ? error.message : String(error)
438             if (
439                 /detached|Target closed|Execution context was destroyed/i.test(
440                     message
441                 )
442             ) {
443                 try {
444                     await expect
445                         .poll(async () => this.isOnPaymentThankYou(), {
446                             timeout: 30_000,
447                         })
448                         .toBe(true)
449                     return
450                 } catch {
451                     if (await this.isOnPaymentThankYou()) {
452                         return
453                     }
454                 }
455             }
456             throw error
457         }
458     }

503     async fillBillingDetailsPaymentFrame(): Promise<void> {
504         // Local /pay flow: Stripe Address Element + "Continue to payment" first;
505         // the payment iframe stays 0×0/hidden until that step completes.
506         // oqg.dev often has no separate Address Element (country lives in the
507         // payment form) — skip only when the billing-info step never appears.
508         const continueBtn = this.locator.billingInfoContinueButton
509         const pageContinue = this.locator.continueToPaymentButton
510         const addressFrame = this.locator.iframeBilling
511         const pageFullName = this.locator.pageBillingFullNameInput
512         const paymentFrame = this.locator.iframePayment
513 
514         // After Buy Now the next surface may be billing, a ready Payment Element,
515         // or (handled by selectPlanIfPickerPresent) the /plan picker.
516         await continueBtn
517             .or(addressFrame)
518             .or(pageContinue)
519             .or(pageFullName)
520             .or(paymentFrame)
521             .first()
522             .waitFor({ state: 'visible', timeout: 30_000 })
523             .catch(() => undefined)
524 
525         const billingStepAppeared =
526             (await continueBtn.isVisible().catch(() => false)) ||
527             (await addressFrame.isVisible().catch(() => false)) ||
528             (await pageContinue.isVisible().catch(() => false)) ||
529             (await pageFullName.isVisible().catch(() => false))
530 
531         if (!billingStepAppeared) {
532             return
533         }
534 
535         if (!(await addressFrame.isVisible().catch(() => false))) {
536             if (await pageFullName.isVisible().catch(() => false)) {
537                 await this.fillPageBillingInformation()
538                 return
539             }
540             await addressFrame
541                 .or(pageFullName)
542                 .first()
543                 .waitFor({ state: 'visible', timeout: 15_000 })
544             if (await pageFullName.isVisible().catch(() => false)) {
545                 await this.fillPageBillingInformation()
546                 return
547             }
548         }
549 
550         await this.expect(addressFrame).toBeVisible({ timeout: 15_000 })
551         await this.expect(this.locator.billingFullNameInput).toBeVisible({
552             timeout: 10_000,
553         })
554         await this.locator.billingFullNameInput.fill(paymentData.fullName)
555         await this.locator.billingCountryInput.selectOption({
556             value: paymentData.country,
557         })
558         await this.locator.billingAddressLine1Input.fill(paymentData.address)
559         if (
560             await this.locator.billingCityInput
561                 .waitFor({ state: 'visible', timeout: 10_000 })
562                 .then(() => true)
563                 .catch(() => false)
564         ) {
565             await this.locator.billingCityInput.fill(paymentData.city)
566         }
567         if (
568             await this.locator.billingStateInput.isVisible().catch(() => false)
569         ) {
570             await this.locator.billingStateInput
571                 .selectOption({ value: paymentData.state })
572                 .catch(() => undefined)
573         }
574         if (
575             await this.locator.billingPostalCodeInput
576                 .isVisible()
577                 .catch(() => false)
578         ) {
579             await this.locator.billingPostalCodeInput.fill(paymentData.zip)
580         }
581         await this.locator.exitFromPaymentFrame.click()
582         const infoContinue = continueBtn.or(pageContinue).first()
583         await infoContinue.click()
584         // Payment Element only becomes visible after leaving the info step.
585         await this.expect(infoContinue).toBeHidden({ timeout: 15_000 })
586     }

618     /**
619      * Sets country on page-level `/pay` billing (native `<select>` or combobox).
620      */
621     private async setPageBillingCountry(
622         countryName: string,
623         countryCode: string
624     ): Promise<void> {
625         const country = this.locator.pageBillingCountryInput
626         await this.expect(country).toBeVisible({ timeout: 15_000 })
627         const tagName = await country.evaluate((el) => el.tagName)
628         if (tagName === 'SELECT') {
629             await country
630                 .selectOption({ value: countryCode })
631                 .catch(() => country.selectOption({ label: countryName }))
632             return
633         }
634         await country.click()
635         const option = this.page.getByRole('option', {
636             name: new RegExp(`^${countryName}$`, 'i'),
637         })
638         if (
639             await option
640                 .first()
641                 .isVisible()
642                 .catch(() => false)
643         ) {
644             await option.first().click()
645             return
646         }
647         await country.fill(countryName)
648         await this.page
649             .getByRole('option', { name: new RegExp(countryName, 'i') })
650             .first()
651             .click()
652     }

1027     /**
1028      * Selects a plan on the /plan subscribe page (or Plans & Pricing / Change plan).
1029      *
1030      * New /plan UI: plan cards render inside `.new-plan-plan-cards-wrap`, each a
1031      * `div.plan-card-wrap.{monthly|annually|quarterly}-wrap` wrapping a
1032      * `input.custom-radio-btn[name=plan][value=…]`. The cards start blurred
1033      * (`blur-cards`) with the native radio `disabled`, and only become selectable
1034      * once Stripe's payment element mounts into `#payment-element`. If the payment
1035      * element never loads — e.g. the backend returns an invalid Stripe
1036      * `clientSecret` (`IntegrationError: Invalid value for elements()`) — the cards
1037      * stay blurred and no plan can be selected. We detect that and fail loudly so
1038      * the failure names the real cause instead of a cryptic radio timeout.
1039      *
1040      * Billing / pricing still uses a visible `<a data-plan-name>` Buy Now plus a
1041      * hidden `input.url-picker[data-plan-name]` radio. That hidden radio must not
1042      * be used for a `visible` wait — `.or().first()` matches it and hangs.
1043      * Older/staging layouts also keep `input.custom-radio-btn` (often opacity:0).
1044      */
1045     private async selectPlanByValue(
1046         plan: 'Monthly' | 'Annually' | 'Quarterly'
1047     ): Promise<void> {
1048         const radio = this.page.locator(
1049             `input.custom-radio-btn[name='plan'][value='${plan}']`
1050         )
1051         const buyLink = this.page.locator(`a[data-plan-name='${plan}']`)
1052         const visiblePlanControl = this.page
1053             .locator(
1054                 `a[data-plan-name='${plan}'], input[data-plan-name='${plan}']`
1055             )
1056             .filter({ visible: true })
1057         const cardsWrap = this.page.locator('.new-plan-plan-cards-wrap')
1058         const planCard = this.page.locator(
1059             `div.plan-card-wrap.${plan.toLowerCase()}-wrap`
1060         )
1061 
1062         await radio
1063             .or(buyLink)
1064             .or(cardsWrap)
1065             .first()
1066             .waitFor({ state: 'attached', timeout: 30000 })
1067 
1068         // New /plan card UI
1069         if ((await cardsWrap.count()) > 0) {
1070             try {
1071                 await this.expect(cardsWrap).not.toHaveClass(/blur-cards/, {
1072                     timeout: 30000,
1073                 })
1074             } catch {
1075                 throw new Error(
1076                     `Cannot select plan "${plan}": the /plan cards stayed blurred because ` +
1077                         `the Stripe payment element never rendered. The page throws ` +
1078                         `"IntegrationError: Invalid value for elements(): clientSecret …" — ` +
1079                         `the backend is not providing a valid Stripe clientSecret. This is a ` +
1080                         `dev payment-integration bug, not a test/locator problem.`
1081                 )
1082             }
1083             // Once un-blurred the native radio is re-enabled. Selecting the card
1084             // works in most cases; fall back to force-checking the radio if the
1085             // card click did not register the selection.
1086             await this.expect(radio)
1087                 .toBeEnabled({ timeout: 15000 })
1088                 .catch(() => {})
1089             for (let attempt = 0; attempt < 3; attempt++) {
1090                 if (await radio.isChecked().catch(() => false)) break
1091                 await planCard.click({ force: true }).catch(() => {})
1092                 if (!(await radio.isChecked().catch(() => false))) {
1093                     await radio.check({ force: true }).catch(() => {})
1094                 }
1095             }
1096             await this.expect(radio).toBeChecked({ timeout: 10000 })
1097             return
1098         }
1099 
1100         // Legacy /plan radios (often opacity:0 — not Playwright-visible)
1101         if ((await radio.count()) > 0) {
1102             const target = radio.first()
1103             await this.expect(target).toBeEnabled({ timeout: 20000 })
1104             await target.check({ force: true })
1105             await this.expect(target).toBeChecked({ timeout: 10000 })
1106             return
1107         }
1108 
1109         // Billing / pricing Buy Now — click only the painted control so the
1110         // hidden `.url-picker` radio is not the match.
1111         await this.expect(visiblePlanControl.first()).toBeVisible({
1112             timeout: 15000,
1113         })
1114         await visiblePlanControl.first().click({ force: true })
1115     }

1332     async selectPlanAndReadTotal(
1333         plan: 'Monthly' | 'Annually' | 'Quarterly'
1334     ): Promise<string> {
1335         const total = this.locator.checkoutSummaryTotal
1336         await this.expect(total).toBeVisible({ timeout: 30_000 })
1337         const before = ((await total.textContent()) ?? '').trim()
1338 
1339         const changePlanResponse = this.page
1340             .waitForResponse(
1341                 (response) => {
1342                     if (
1343                         response.request().method() !== 'POST' ||
1344                         !response.url().includes('/api/ajax')
1345                     ) {
1346                         return false
1347                     }
1348                     const body = response.request().postData() ?? ''
1349                     return body.includes('change_plan')
1350                 },
1351                 { timeout: 30_000 }
1352             )
1353             .catch(() => null)
1354 
1355         await this.selectPlanByValue(plan)
1356         await changePlanResponse
1357 
1358         const cardsWrap = this.page.locator('.new-plan-plan-cards-wrap')
1359         await this.expect(cardsWrap).not.toHaveClass(/blur-cards/, {
1360             timeout: 30_000,
1361         })
1362         await this.expect(total).toBeVisible({ timeout: 30_000 })
1363 
1364         await expect
1365             .poll(
1366                 async () => {
1367                     const text = ((await total.textContent()) ?? '').trim()

1370                     return this.planCardRadio(plan).isChecked()

1372                 { timeout: 30_000 }
