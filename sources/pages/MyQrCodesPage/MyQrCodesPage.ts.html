231     /**
232      * Opens a QR card's kebab menu and returns the open menu.
233      *
234      * The listing closes any open dropdown when the page scrolls, so a click
235      * that has to scroll the kebab into view opens the menu and loses it a
236      * frame later — the state the all-codes view lands in, where the card sits
237      * at the bottom edge of the viewport. Centring the button first keeps the
238      * click itself scroll-free; Escape resets the toggle before a retry.
239      */
240     private async openQrCardActionMenu(
241         qrCard: Locator,
242         page: Page = this.page
243     ): Promise<Locator> {
244         const kebabButton = qrCard.locator('button.kebab-button')
245         const actionMenu = qrCard.locator('div.dropdown-menu')
246 
247         for (let attempt = 1; attempt <= 3; attempt++) {
248             await kebabButton.evaluate((element) =>
249                 element.scrollIntoView({
250                     block: 'center',
251                     behavior: 'instant',
252                 })
253             )
254             await kebabButton.click()
255 
256             try {
257                 await actionMenu.waitFor({ state: 'visible', timeout: 5_000 })
258                 return actionMenu
259             } catch {
260                 await page.keyboard.press('Escape')
261             }
262         }
263 
264         throw new Error(
265             'The QR code action menu did not stay open after 3 attempts'
266         )
267     }

378     async confirmQrCodeDeletionOn(page: Page): Promise<void> {
379         const deleteModal = page.locator('#DeleteModal.show')
380         // Await the delete round-trip itself: the caller usually navigates
381         // straight after, which would abort a still-pending request.
382         const deleteResponse = page.waitForResponse(
383             (response) =>
384                 response.request().method() === 'POST' &&
385                 (response.request().postData() ?? '').includes(
386                     'action=make_qrcode_deleted'
387                 )
388         )
389         await deleteModal
390             .getByRole('button', { name: 'Delete', exact: true })
391             .click()
392         await deleteResponse
393         // Best effort: when the code was already removed in another tab the app
394         // keeps the dialog open, which is the state TC26 deliberately exercises.
395         await deleteModal
396             .waitFor({ state: 'hidden', timeout: 10_000 })
397             .catch(() => undefined)
398     }

497     async dismissDownloadModal(): Promise<void> {

499         if ((await modal.count()) === 0) return
500 
501         const closeBtn = this.page.locator(
502             '#DownloadModal button[aria-label="Close"]'
503         )
504         if (
505             await modal

507                 .catch(() => false)
508         ) {
509             await closeBtn.click({ force: true })
510         }
511         await expect
512             .poll(
513                 async () =>
514                     !(await modal
515                         .evaluate((el) => el.classList.contains('show'))
516                         .catch(() => false)),
517                 { timeout: 15_000 }
518             )
519             .toBe(true)
520     }
