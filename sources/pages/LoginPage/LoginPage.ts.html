pages/LoginPage/LoginPage.ts

Bundled source snapshot. Credential-like literals are omitted. Line numbers correspond to the audited repository file; this page is for reading and is not executable test code.

1 import type { Page } from '@playwright/test'
2 import { BasePage } from '../../base/BasePage'
3 import { PageFactory } from '../../base/PageFactory'
4 import {
5     LoginPageLocators,
6     OqgLoginPageLocators,
7     QciLoginPageLocators,

10 abstract class SharedLoginPage extends BasePage {
11     abstract readonly locator: LoginPageLocators
12 
13     /**
14      * Two-step login: email → `/api/guest_data` → password field is shown.
15      * The password input stays in a Bootstrap `d-none` wrapper until that
16      * response returns `result=password`; do not treat it as visible after fill.
17      */
18     async logIn(temporaryMail: string, signupPassword: string): Promise<void> {
19         await this.locator.emailInput.waitFor({ state: 'visible' })
20         await this.locator.emailInput.fill(temporaryMail)
21         await this.locator.logInConfirmButton.click()
22         await this.locator.passwordInput.waitFor({
23             state: 'visible',
24             timeout: 15_000,
25         })
26         await this.locator.passwordInput.fill(signupPassword)
27         await this.locator.logInConfirmButton.click()
28     }
29 
30     /** Fills the login email field. */
31     async enterEmail(email: string): Promise<void> {
32         await this.locator.emailInput.fill(email)
33     }
34 
35     /** Submits the current login step. */
36     async clickLoginButton(): Promise<void> {
37         await this.locator.logInConfirmButton.click()
38     }
39 
40     /** Fills the login password field. */
41     async fillInPassword(password: string): Promise<void> {
42         await this.locator.passwordInput.fill(password)
43     }
44 
45     /** Opens the account-registration flow. */
46     async clickCreateAccount(): Promise<void> {
47         await this.locator.haveAccountLink.click()
48     }
49 
50     /** Opens the FAQ from the login page. */
51     async clickFaqIcon(): Promise<void> {
52         await this.locator.faqIcon.click()
53     }
54 
55     /** Opens the Lost Password form from the login password step. */
56     async clickForgotPassword(): Promise<void> {
57         await this.locator.forgotPasswordLink.click()
58     }
59 
60     /**
61      * Fills the Lost Password email field (`input#email`).
62      * @param email - Registered user email to send the recovery link to
63      */
64     async enterEmailForPasswordReset(email: string): Promise<void> {
65         await this.locator.resetPassFormNewPassInput.fill(email)
66     }
67 
68     /** Submits the Lost Password form and waits for the success alert. */
69     async clickResetPasswordButton(): Promise<void> {
70         await this.locator.submitButton.click()
71         await this.expect(this.locator.successAlert).toBeVisible()
72     }
73 
74     /** Returns from the Lost Password form to login. */
75     async clickBackToLoginButton(): Promise<void> {
76         await this.locator.backToLoginButton.click()
77     }
78 
79     /**
80      * Fills the new password field on the Set a new password page.
81      * @param newPassword - Password to set
82      */
83     async enterNewPassword(newPassword: string): Promise<void> {
84         await this.locator.newPasswordInput.fill(newPassword)
85     }
86 
87     /**
88      * Fills the repeat-password field on the Set a new password page.
89      * @param repeatNewPassword - Confirmation password
90      */
91     async enterRepeatNewPassword(repeatNewPassword: string): Promise<void> {
92         await this.locator.repeatNewPasswordInput.fill(repeatNewPassword)
93     }
94 
95     /** Submits the Set a new password form. */
96     async submitNewPassword(): Promise<void> {
97         await this.locator.submitButton.click()
98     }
99 
100     /**
101      * Completes the Lost Password form for the given email and waits for success.
102      * @param email - Registered user email
103      */
104     async requestPasswordReset(email: string): Promise<void> {
105         await this.clickForgotPassword()
106         await this.enterEmailForPasswordReset(email)
107         await this.clickResetPasswordButton()
108     }
109 
110     /**
111      * Sets a new password on the reset-password page (both fields).
112      * @param newPassword - Password to set
113      */
114     async completePasswordReset(newPassword: string): Promise<void> {
115         await this.locator.newPasswordInput.waitFor({ state: 'visible' })
116         await this.enterNewPassword(newPassword)
117         await this.enterRepeatNewPassword(newPassword)
118         await this.submitNewPassword()
119     }
120 
121     /**
122      * Completes the admin-generated Create New Password form (`/create-password`).
123      * Uses `#password` / `#confirm-password` (not the forgot-reset field ids).
124      * @param newPassword - Password to set
125      */
126     async completeCreatePassword(newPassword: string): Promise<void> {
127         await this.locator.createPasswordInput.waitFor({ state: 'visible' })
128         await this.locator.createPasswordInput.fill(newPassword)
129         await this.locator.confirmCreatePasswordInput.fill(newPassword)
130         await this.locator.createPasswordSubmitButton.click()
131     }
132 
133     /**
134      * Fills the Set a new password form without submitting.
135      * Used for cancel / leave-without-submit negative cases.
136      * @param newPassword - Password to enter in both fields
137      */
138     async fillPasswordResetWithoutSubmit(newPassword: string): Promise<void> {
139         await this.locator.newPasswordInput.waitFor({ state: 'visible' })
140         await this.enterNewPassword(newPassword)
141         await this.enterRepeatNewPassword(newPassword)
142     }
143 
144     /**
145      * Submits email-only login and waits for the OTP entry UI (passwordless users).
146      * @param email - Account email that triggers a one-time password
147      */
148     async submitEmailForOtpLogin(email: string): Promise<void> {
149         await this.enterEmail(email)
150         await this.clickLoginButton()
151         await this.locator.oneTimePassLeftInput.waitFor({ state: 'visible' })
152     }
153 
154     /**
155      * Completes the OTP verification step after email-only login.
156      * @param otpCode - One-time password from the verification email
157      */
158     async completeOtpVerification(otpCode: string): Promise<void> {
159         await this.locator.oneTimePassLeftInput.waitFor({ state: 'visible' })
160         await this.locator.oneTimePassLeftInput.pressSequentially(otpCode, {
161             delay: 100,
162         })
163         await this.locator.oneTimePassVerifiBtn.click()
164     }
165 
166     /**
167      * Asserts successful login landed on the authenticated app area.
168      */
169     async expectLoggedIn(): Promise<void> {
170         await this.expect(this.page).toHaveURL(/qr-codes|create|qr\?/)
171     }
172 
173     /**
174      * Asserts an expired / invalid / already-used reset-password link was rejected.
175      * Product behavior (ResetPassword.php): unknown, superseded, or consumed
176      * `lost_password_code` redirects away with no set-password form.
177      * There is no dedicated "link expired" toast in the current UI.
178      */
179     async expectResetLinkRejected(): Promise<void> {
180         await this.expect(this.page).not.toHaveURL(/reset-password/)
181         await this.expect(this.locator.newPasswordInput).toBeHidden()
182     }
183 }
