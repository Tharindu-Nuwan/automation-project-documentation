1 import { Page, Locator } from '@playwright/test'
2 import { BasePage } from '../../base/BasePage'
3 import { AdminPageLocators } from './AdminPageLocators'
4 
5 export class AdminPage extends BasePage {
6     locator: AdminPageLocators
7 
8     constructor(page: Page) {
9         super(page)
10         this.locator = new AdminPageLocators(page)
11     }
12 
13     getUserMenuDotsButtonUsersTab(userEmail: string): Locator {
14         return this.page.locator(
15             `//tr[td//a[contains(text(), '${userEmail}')]]//div[contains(@class, 'dropdown')]//button`
16         )
17     }
18 
19     getUserMenuDotsButtonPaymentsTab(emailDpf: string): Locator {
20         return this.page.locator(
21             `//tr[.//span[contains(text(),'${emailDpf}')]]//div[contains(@class,'dropdown actions-dropdown')]`
22         )
23     }
24 
25     getUserMenuRefundButtonPaymentsTab(emailDpf: string): Locator {
26         return this.page.locator(
27             `//tr[.//span[contains(text(),'${emailDpf}')]]//a[text()=' Refund']`
28         )
29     }
30 
31     async findUserByEmail(email: string): Promise<void> {
32         await this.locator.globalSearchInput.fill(email)
33         await this.locator.searchButton.click()
34         await this.page.waitForLoadState('networkidle')
35         await this.locator.foundUser.click()
36     }
37 
38     /**
39      * Opens the "Generate URL" discount modal (`#user_plan_generate_modal`) for
40      * the user whose page is currently open (call {@link findUserByEmail} first).
41      */
42     async openGenerateDiscountModal(): Promise<void> {
43         await this.locator.kebabMenu.click()
44         await this.locator.generateDiscountUrlBtn.click()
45     }
46 
47     /**
48      * Inside the already-open discount modal, picks a plan option, generates the
49      * link and returns its URL. Multiple types can be generated back-to-back
50      * without re-opening (matches the admin UI). Locators live in
51      * AdminPageLocators.
52      *
53      * `70off`      → `?promo_code=70OFF&referral_key=…`
54      * `discounted` → `?plan_type=discounted&referral_key=…` ($8.99 monthly)
55      * `onetime`    → `?plan_type=onetime&referral_key=…`    ($50 one-time)
56      */
57     async pickDiscountLink(
58         type: 'default' | '70off' | 'discounted' | 'onetime'
59     ): Promise<string> {
60         const option = {
61             default: this.locator.defaultPricingButton,
62             '70off': this.locator.discount70PromoButton,
63             discounted: this.locator.discount899MonthlyButton,
64             onetime: this.locator.discount50OneTimeButton,
65         }[type]
66         await option.click()
67         await this.locator.generateLinkButton.click()
68         return ((await this.locator.generatedLink.textContent()) ?? '').trim()
69     }
70 
71     /**
72      * From the user detail view: Generate URL → Create New Password → Generate Link.
73      * @returns Absolute create-password URL (`.../create-password?referral_key=...`)
74      */
75     async generateCreatePasswordLink(): Promise<string> {
76         await this.locator.kebabMenu.click()
77         await this.locator.generateDiscountUrlBtn.click()
78         await this.locator.createNewPasswordButton.waitFor({ state: 'visible' })
79         await this.locator.createNewPasswordButton.click()
80         await this.locator.generateLinkButton.click()
81         await this.locator.generatedLink.waitFor({ state: 'visible' })
82         const link = (await this.locator.generatedLink.textContent())?.trim()
83         if (!link) {
84             throw new Error('Admin Create New Password link was empty')
85         }
86         return link
87     }
88 
89     async logout(): Promise<void> {
90         await this.locator.adminUserMenuButton.click()
91         await this.locator.adminLogoutButton.click()
92     }
93 }
