236 /**
237  * QCI: flash may animate away or redirect to `/qr-codes`.
238  * Only asserts a visible banner when one is present.
239  */
240 class QciAccountPage extends SharedAccountPage {
241     async expectSuccessAlertVisible(): Promise<void> {
242         const visibleFlash = this.page
243             .locator('div.alert-success')
244             .filter({ visible: true })
245         if ((await visibleFlash.count()) > 0) {
246             await this.expect(visibleFlash.last()).toBeVisible()
247         }
248     }
249 
250     protected async waitForPasswordChangeApplied(): Promise<void> {
251         try {
252             await Promise.race([
253                 this.expect(this.locator.passwordInput)
254                     .toHaveValue('')
255                     .then(async () => {
256                         await this.expect(
257                             this.locator.passwordConfirmInput
258                         ).toHaveValue('')
259                     }),
260                 this.page.waitForURL(/\/qr-codes(?:\/|$|\?)/, {
261                     timeout: 25_000,
262                 }),
263             ])
264         } catch {
265             throw new Error(
266                 'Password change did not apply: password fields were not cleared and no redirect to /qr-codes'
267             )
268         }
269     }
270 }
