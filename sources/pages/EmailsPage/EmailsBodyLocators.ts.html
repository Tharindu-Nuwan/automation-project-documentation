5 /**
6  * Shared contract for email HTML locators.
7  * Product-specific selectors belong in the concrete locator classes.
8  */
9 export abstract class EmailBodyLocators extends BaseLocators {
10     abstract readonly logoLink: Locator
11     readonly faqLink: Locator
12     readonly loginLink: Locator
13     readonly contactLink: Locator
14     readonly downloadButton: Locator
15     readonly myAccount: Locator
16     readonly createQrCodes: Locator
17     readonly loginLinkTId8: Locator
18     /** Primary CTA in template #17 (forgot password / recovery) emails. */
19     readonly resetPasswordLink: Locator
20 
21     protected constructor(page: Page) {
22         super(page)
23 
24         this.faqLink = page.locator('//a[contains(@href, "/faq")]')
25         this.loginLink = page.locator('//tbody/tr/td[2]/a')
26         this.contactLink = page.locator('//a[contains(@href, "/contact")]')
27         this.downloadButton = page.locator('(//tr/td/a)[2]')
28         this.myAccount = page.locator('(//tr/td/a)[2]')
29         this.createQrCodes = page.locator('(//tr/td/a)[2]')
30         this.loginLinkTId8 = page.locator('//tbody/tr/td/div/div/p/a')
31         this.resetPasswordLink = page.locator('a[href*="/reset-password/"]')
32     }
33 }

61 /**
62  * Creates product-specific locators for legacy specs that render email HTML
63  * outside PageManager.
64  *
65  * @deprecated Use `pageManager.emailBodyPage.locator` in migrated specs.
66  */
67 export function createEmailBodyLocators(
68     page: Page,
69     product: Product
70 ): EmailBodyLocators {
71     const locators: Partial<Record<Product, () => EmailBodyLocators>> = {
72         oqg: () => new OqgEmailBodyLocators(page),
73         qci: () => new QciEmailBodyLocators(page),
74     }
75 
76     const createLocators = locators[product]
77     if (!createLocators) {
78         throw new Error(
79             `Email body locators are not implemented for ${product.toUpperCase()}`
80         )
81     }
82 
83     return createLocators()
84 }
