380     async addSocialMediaLink(
381         url: string = 'example.com',
382         urlText: string = 'Test Link Text'
383     ): Promise<void> {
384         const frameSocialMediaLogo = `//div[@class='App']//img[@alt='LinkedIn']`
385         // The Event preview renders the link label with a different class than
386         // the vCard/Links previews, so match either markup rather than
387         // duplicating this helper per QR type.
388         const frameUrlTextLegacy = `//div[@class='App']//img[@alt='LinkedIn']//../div/p[contains(@class, 'word-break-fix')]`
389         const frameUrlTextEvent = `//div[@class='App']//img[@alt='LinkedIn']/following-sibling::div/p[contains(@class, 'text-event-500')]`
390         await this.page.waitForTimeout(500)
391         if (
392             await this.locator.eventSocialNetworksAccordionTrigger
393                 .isVisible()
394                 .catch(() => false)
395         ) {
396             await this.expandEventAccordion(
397                 this.locator.eventSocialNetworksAccordionTrigger
398             )
399         }
400         await this.locator.linkedInSocialNetworkItem.click()
401         await this.expect(
402             this.locator.frameView.locator(frameSocialMediaLogo)
403         ).toBeVisible()
404         await this.locator.linkedInURLInput.pressSequentially(url)
405         await this.locator.linkedInTextInput.pressSequentially(urlText)
406         await this.expect(
407             this.locator.frameView
408                 .locator(frameUrlTextLegacy)
409                 .or(this.locator.frameView.locator(frameUrlTextEvent))
410         ).toContainText(urlText)
411     }

1138      */
1139     /**
1140      * Navigates the open Event date picker until the exact target day
1141      * button exists. Uses mid-month samples so adjacent-month overflow
1142      * cells (e.g. early Feb visible on a January grid) do not short-circuit.
1143      */
1144     private async navigateEventDatePickerToDay(
1145         prefix: 'start' | 'end',
1146         dateValue: string
1147     ): Promise<void> {
1148         const [year, month] = dateValue.split('-').map(Number)
1149         const targetDay = this.page.getByTestId(
1150             `step2-event-date-${prefix}-date-day-${dateValue}-button`
1151         )
1152         const midMonthSample = this.page.getByTestId(
1153             `step2-event-date-${prefix}-date-day-${year}-${String(month).padStart(2, '0')}-15-button`
1154         )
1155         const nextMonth = this.page.getByRole('button', { name: /next month/i })
1156         const prevMonth = this.page.getByRole('button', {
1157             name: /previous month/i,
1158         })
1159 
1160         for (let i = 0; i < 48; i++) {
1161             if ((await targetDay.count()) > 0) {
1162                 return
1163             }
1164             if ((await midMonthSample.count()) > 0) {
1165                 return
1166             }
1167             const dayButtons = this.page.locator(
1168                 `[data-qa-id*="step2-event-date-${prefix}-date-day-"]`
1169             )
1170             const qaIds = await dayButtons.evaluateAll((els) =>
1171                 els.map((el) => el.getAttribute('data-qa-id') ?? '')
1172             )
1173             // Prefer a day near the middle of the currently shown month
1174             const sample =
1175                 qaIds.find((id) => /-15-button$/.test(id)) ??
1176                 qaIds.find((id) =>
1177                     id.includes(`step2-event-date-${prefix}-date-day-`)
1178                 )
1179             if (!sample) {
1180                 throw new Error(
1181                     `Event ${prefix} date picker has no day cells to navigate from`
1182                 )
1183             }
1184             const match = sample.match(
1185                 new RegExp(
1186                     `step2-event-date-${prefix}-date-day-(\\d{4})-(\\d{2})-`
1187                 )
1188             )
1189             if (!match) {
1190                 throw new Error(
1191                     `Unable to parse month from date picker day qa-id: ${sample}`
1192                 )
1193             }
1194             const visibleIndex = Number(match[1]) * 12 + Number(match[2])
1195             const targetIndex = year * 12 + month
1196             if (visibleIndex < targetIndex) {
1197                 await nextMonth.click()
1198             } else if (visibleIndex > targetIndex) {
1199                 await prevMonth.click()
1200             } else if ((await targetDay.count()) === 0) {
1201                 // Same month header but target day missing — advance once
1202                 await nextMonth.click()
1203             } else {
1204                 return
1205             }
1206         }
1207         throw new Error(
1208             `Unable to navigate Event ${prefix} date picker to ${dateValue}`
1209         )
1210     }

1234             const currentYear = Number((await yearButton.textContent())?.trim())
