26     constructor(page: Page) {
27         super(page)
28 
29         this.variablePlanButtonUniversalLocator = page.locator(
30             "//div[@class='buy-btn-area']//a[@data-plan-name='Monthly'] | //div[@class='buy-btn-area']//a[@data-plan-name='Annually'] | //div[@class='buy-btn-area']//a[@data-plan-name='Quarterly'] | //div[@class='buy-btn-area']//a[@data-plan-name='One Time'] | //div[@class='buy-btn-area']//a[@data-plan-name='Discounted'] | //input[@class='custom-radio-btn' and @name='plan']"
31         )
32         // Visible Buy Now <a> — do not include input[data-plan-name], a hidden
33         // input.url-picker with the same attribute sits first in the DOM and
34         // makes .first() / toBeVisible() wait on a hidden radio.
35         this.monthlyBuyNowButton = page
36             .locator("a[data-plan-name='Monthly']")
37             .filter({ visible: true })
38         this.annuallyBuyNowButton = page
39             .locator("a[data-plan-name='Annually']")
40             .filter({ visible: true })
41         this.quarterlyBuyNowButton = page
42             .locator("a[data-plan-name='Quarterly']")
43             .filter({ visible: true })
44         this.continueButton = page.locator('#user_plan_url')
45 
46         this.currentPriceValue = page.locator("//input[@id='dpfPlan1']")
47         // Responsive design locators
48         this.tabletAnnuallyBuyNowButton = page.locator(
49             "input[data-plan-name='Annually']"
50         )
51         this.tabletBuyNowButton = page.locator('div.mobile-buy-btn-wrp a')
52 
53         this.specialDealBanner = page.getByText('Special Deal')
54         this.plansPricingHeader = page
55             .locator('h1.plan-header')
56             .filter({ hasText: /Plans & Pricing/i })
57         // Live OQG and QCI both use h2.package-header-heading; keep h3.main-price
58         // as a fallback for older markup.
59         this.mainPriceHeadings = page.locator(
60             'h2.package-header-heading, h3.main-price'
61         )
62         this.discountPlanHeader = page
63             .locator('h1.plan-header')
64             .filter({ hasText: /Discounted Plan|One Time Payment/i })
65         // Amount only — never fall back to the h1 title (it matches first and
66         // toContainText('8.99') then times out on "Discounted Plan").
67         this.discountCardPrice = page
68             .locator(
69                 'h2.price-reverse-order, h2.package-header-heading, h3.main-price'
70             )
71             .filter({ hasText: /\d/ })
